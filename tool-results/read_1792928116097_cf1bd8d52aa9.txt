    44→/** Honest negative-marking copy: "−0.25 per wrong answer" or "No negative marking". */
    45→export function negativeMarkText(negativeMark: number): string {
    46→  return negativeMark > 0
    47→    ? `\u2212${negativeMark} per wrong answer`
    48→    : 'No negative marking'
    49→}

    51→function NegativeMarkBadge({ value }: { value: number }) {
    52→  return (
    53→    <span
    54→      className={cn(
    55→        'inline-flex items-center gap-1 rounded-full border px-2 py-0.5 text-xs',
    56→        value > 0
    57→          ? 'border-gold/40 bg-gold/10 text-gold'
    58→          : 'border-success/40 bg-success/10 text-success'

   372→// ─── Official document card (Layer 1 — Study Material) ───────────────────────
   373→// Honest behaviour: PORTAL/WEBPAGE open as pages (never fake PDF downloads);
   374→// PDF entries (only when a stable direct file is confirmed) offer a download.
   375→export function DocumentCard({ resource: r }: { resource: ResourceDTO }) {
   376→  const unreachable = r.verificationStatus === 'UNREACHABLE'
   377→  const kind = r.fileKind ?? 'WEBPAGE'
   378→
   379→  const Icon = kind === 'PDF' ? FileDown : kind === 'PORTAL' ? Landmark : ScrollText
   380→  const actionLabel = kind === 'PDF' ? 'Download PDF' : kind === 'PORTAL' ? 'Open portal' : 'Open page'
   381→  const ActionIcon = kind === 'PDF' ? FileDown : ExternalLink
   382→
   383→  return (
   384→    <li className={cn('flex h-full flex-col rounded-xl border border-border bg-card p-4', unreachable && 'opacity-80')}>
   385→      <div className="flex items-start gap-3">
   386→        <span className="flex h-9 w-9 shrink-0 items-center justify-center rounded-lg border border-gold/30 bg-gold/10">
   387→          <Icon className="h-4 w-4 text-gold" aria-hidden />
   388→        </span>
   389→        <div className="min-w-0">
   390→          <p className="text-sm font-medium leading-snug text-foreground">{r.title}</p>
   391→          <p className="mt-0.5 text-xs text-muted-foreground">
   392→            {r.provider} · {DOC_TYPE_LABELS[r.type] ?? r.type} · {r.language}
   393→          </p>
   394→        </div>
   395→      </div>
   396→
   397→      <div className="mt-2.5 flex flex-wrap items-center gap-1.5">
   398→        <ResourceStatusBadge status={r.verificationStatus} />
   399→        <AccessTypeBadge access={r.accessType} />
   400→        {r.docDate && (
   401→          <span className="inline-flex items-center gap-1 rounded-full border border-border bg-elevated px-2 py-0.5 text-[11px] text-muted-foreground">
   402→            Dated {r.docDate}
   403→          </span>
   404→        )}
   405→      </div>
   406→
   407→      {r.note && <p className="mt-2 text-xs leading-relaxed text-muted-foreground">{r.note}</p>}
   408→      {unreachable && (
   409→        <p className="mt-2 text-xs italic leading-relaxed text-muted-foreground">
   410→          This official site did not respond from our build environment ({r.lastChecked ? new Date(r.lastChecked).toLocaleDateString('en-IN', { year: 'numeric', month: 'short', day: 'numeric' }) : 'recently'}) — it normally works from student networks in India.
   411→        </p>
   412→      )}
   413→
   414→      <Button
   415→        asChild
   416→        variant="outline"
   417→        size="sm"
   418→        className={cn(
   419→          'mt-3 w-full shrink-0 sm:w-auto sm:self-start',
   420→          unreachable ? 'border-border text-muted-foreground' : 'border-gold/40 text-gold hover:bg-gold/10',
   421→        )}
   422→      >
   423→        <a
   424→          href={r.url}
   425→          target="_blank"
   426→          rel="noopener noreferrer"
   427→          download={kind === 'PDF' ? true : undefined}
   428→          aria-label={`${actionLabel}: ${r.title} (opens in a new tab)`}
   429→        >
   430→          {actionLabel} <ActionIcon className="ml-1.5 h-3.5 w-3.5" aria-hidden />
   431→        </a>
   432→      </Button>
   433→    </li>
   434→  )
   435→}

   477→function InlineTopicPanel({
   478→  topicId,
   479→  examName,
   480→  subjectName,
   481→  onAskBean,
   482→  onPractice,
   483→  startingTopicId,
   484→}: {
   485→  topicId: string
   486→  examName: string
   487→  subjectName: string
   488→  onAskBean: (t: FocusTopic, noteTitle: string | undefined) => void
   489→  onPractice: (topicId: string) => void
   490→  startingTopicId: string | null
   491→}) {
   492→  const navigate = useAppStore(s => s.navigate)
   493→  const [dto, setDto] = useState<TopicDetailDTO | null>(null)
   494→  const [error, setError] = useState<string | null>(null)
   495→
   496→  if (!dto && !error) {
   497→    api.topic(topicId).then(setDto).catch(e => setError(e instanceof Error ? e.message : 'Failed to load this topic'))
   498→  }
   499→
   500→  if (error) {
   501→    return (
   502→      <p role="alert" className="rounded-lg border border-destructive/40 bg-destructive/10 px-3 py-2 text-xs text-destructive">
   503→        {error}
   504→      </p>
   505→    )
   506→  }
   507→
   508→  if (!dto) {
   509→    return (
   510→      <div className="space-y-2" aria-busy="true" aria-label="Loading study material">
   511→        <Skeleton className="h-4 w-3/4 bg-elevated" />
   512→        <Skeleton className="h-4 w-full bg-elevated" />
   513→        <Skeleton className="h-4 w-5/6 bg-elevated" />
   514→        <Skeleton className="h-4 w-2/3 bg-elevated" />
   515→      </div>
   516→    )
   517→  }
   518→
   519→  const focus: FocusTopic = { id: dto.topic.id, name: dto.topic.name, stageName: '', subjectName }
   520→
   521→  return (

   541→      )}
   542→
   543→      {/* Documents / resources */}
   544→      {dto.resources.length > 0 && (
   545→        <div className="mt-3 border-t border-border pt-3">
   546→          <p className="text-xs font-semibold uppercase tracking-wider text-muted-foreground">

   563→              </li>

   570→          </ul>
   571→        </div>

   574→      {/* Actions — keep the selected topic context across layers */}

   587→            disabled={startingTopicId === topicId}
   588→            onClick={() => onPractice(dto.topic.id)}
   589→            className="bg-gold text-primary-foreground hover:bg-gold-hover"
   590→          >
   591→            {startingTopicId === topicId
   592→              ? <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" aria-hidden />
   593→              : <Play className="mr-1.5 h-3.5 w-3.5" aria-hidden />}

   598→          size="sm"
   599→          variant="ghost"
   600→          className="text-muted-foreground hover:text-foreground"
   601→          onClick={() => navigate({ view: 'topic', topicId: dto.topic.id })}
   602→        >
   603→          Full topic workspace <ExternalLink className="ml-1.5 h-3 w-3" aria-hidden />
   604→        </Button>
   605→      </div>
   606→      <p className="mt-2 text-[11px] text-muted-foreground">
   607→        Context carried: {examName} › {subjectName} › {dto.topic.name}
