   115→function formatBytes(n: number): string {
   116→  if (!Number.isFinite(n) || n <= 0) return '0 B'
   117→  if (n < 1024) return `${n} B`
   118→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(n < 10 * 1024 ? 1 : 0)} KB`
   119→  return `${(n / (1024 * 1024)).toFixed(1)} MB`
   120→}
   121→
   122→function formatDate(iso: string): string {
   123→  const d = new Date(iso)
   124→  if (Number.isNaN(d.getTime())) return iso
   125→  return d.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })
   126→}

   128→/** [1,2,3,7] → "pages 1–3, 7" · [5] → "page 5" */
   129→function formatPageRefs(pages: number[]): string {
   130→  const sorted = [...new Set(pages)].filter(Number.isFinite).sort((a, b) => a - b)
   131→  if (sorted.length === 0) return ''
   132→  const parts: string[] = []
   133→  let start = sorted[0]
   134→  let prev = sorted[0]
   135→  for (let i = 1; i <= sorted.length; i++) {
   136→    const cur = sorted[i]
   137→    if (cur === prev + 1) {
   138→      prev = cur
   139→      continue
   140→    }
   141→    parts.push(start === prev ? `${start}` : `${start}–${prev}`)
   142→    start = cur
   143→    prev = cur
   144→  }
   145→  const joined = parts.join(', ')
   146→  return parts.length === 1 && !joined.includes('–') ? `page ${joined}` : `pages ${joined}`
   147→}

   149→/** Same scaling the backend uses for the distribution preview. */
   150→function distributionLine(count: number): string {
   151→  const scale = count / 50
   152→  const d = {
   153→    f: Math.max(1, Math.round(10 * scale)),
   154→    a: Math.max(1, Math.round(15 * scale)),
   155→    s: Math.max(1, Math.round(10 * scale)),
   156→    m: Math.max(1, Math.round(10 * scale)),
   157→    x: Math.max(1, Math.round(5 * scale)),
   158→  }
   159→  return `${d.f} foundational · ${d.a} application · ${d.s} statement · ${d.m} multi-step · ${d.x} misconception`
   160→}

   433→
   434→  useEffect(() => {
   435→    if (!guestId) return

   438→      .material(docId, guestId)
   439→      .then(d => {
   440→        if (!alive) return
   441→        setLoaded({ key: loadKey, data: d })
   442→        setFailure(null)
   443→      })
   444→      .catch((e: unknown) => {
   445→        if (!alive) return
   446→        setFailure({ key: loadKey, message: e instanceof Error ? e.message : 'Could not load this material.' })
   447→      })
   448→    return () => {
   449→      alive = false
   450→    }
   451→  }, [guestId, docId, loadKey])
   452→
   453→  const doc = loaded && loaded.key === loadKey ? loaded.data : null
   454→  const error = failure && failure.key === loadKey ? failure.message : null
   455→  const loading = !doc && !error
   456→  const notFound = !!error && /not found/i.test(error)
   457→
   458→  // Poll the material every 5s ONLY while it (or one of its tests) is in flight.
   459→  const shouldPoll =
   460→    !!doc && (isProcessing(doc.status) || doc.tests.some(t => t.status === 'QUEUED' || t.status === 'GENERATING'))

   462→    if (!shouldPoll || !guestId || !docId) return

   464→      api
   465→        .material(docId, guestId)
   466→        .then(d => setLoaded({ key: loadKey, data: d }))
   467→        .catch(() => {
   468→          /* keep the last honest state; the next tick retries */
   469→        })
   470→    }, 5000)
   471→    return () => clearInterval(timer)
   472→  }, [shouldPoll, guestId, docId, loadKey])
   473→
   474→  // Default tab: land on Tests when the document already has a READY test to take.
   475→  const [tab, setTab] = useState<WorkspaceTab>('material')
   476→  const docKey = loaded?.key ?? null
   477→  const [appliedKey, setAppliedKey] = useState<string | null>(null)
   478→  if (doc && docKey && docKey !== appliedKey) {
   479→    setAppliedKey(docKey)
   480→    setTab(doc.status === 'READY' && doc.tests.some(t => t.status === 'READY') ? 'tests' : 'material')
   481→  }
   482→
   483→  const [renameOpen, setRenameOpen] = useState(false)
   484→  const [renameValue, setRenameValue] = useState('')
   485→  const [renaming, setRenaming] = useState(false)
   486→
   487→  // Integrated reader — hosted at workspace level so page references anywhere
   488→  // (analysis, explanations, material panel) can deep-link into it.
   489→  const [readerPage, setReaderPage] = useState<number | null>(null)
   490→  const openReaderAt = (page?: number) => setReaderPage(page ?? 1)
   491→  const closeReader = () => setReaderPage(null)
   492→
   493→  const refresh = () => setReloadTick(t => t + 1)
   494→
   495→  // ── Not found ──
   496→  if (notFound) {
   497→    return (
   498→      <div className="mx-auto w-full max-w-3xl px-4 py-16 sm:px-6 lg:px-8">
   499→        <EmptyState
   500→          icon={FileWarning}
