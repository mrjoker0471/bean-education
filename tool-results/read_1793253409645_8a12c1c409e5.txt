     1→// ─── Riishe document structure detection (deterministic, no AI) ──────────────
     2→// Infers a document's real organisation from the extracted text itself:
     3→//   • table of contents (detected + validated against the body)
     4→//   • headings (numbered, chapter-style, ALL-CAPS, Title-Case) with levels
     5→//   • question-paper layouts (so question banks are not exploded into "topics")
     6→//   • printed page numbers vs physical PDF pages (arabic offset + roman front
     7→//     matter) so page references can show both honestly
     8→// and turns all of that into structure-aware ANALYSIS UNITS that cover the
     9→// whole document — never an arbitrary "first N pages" slice.
    10→//
    11→// Everything here is pure and deterministic: the same extraction always
    12→// produces the same structure, so re-analysis is repeatable and unit-testable.
    13→

    93→function linesOf(page: ExtractedPage): string[] {

    97→function letters(s: string): string {
    98→  return s.replace(/[^A-Za-z]/g, '')
    99→}
   100→
   101→function upperRatio(s: string): number {
   102→  const ls = letters(s)
   103→  if (ls.length === 0) return 0
   104→  let up = 0
   105→  for (const c of ls) if (c === c.toUpperCase() && c !== c.toLowerCase()) up++
   106→  return up / ls.length
   107→}
   108→
   109→function titleCaseRatio(s: string): number {
   110→  const words = s.split(/\s+/).filter(w => WORD_RE.test(w))
   111→  if (words.length === 0) return 0
   112→  let cap = 0
   113→  for (const w of words) {
   114→    const first = letters(w)[0]
   115→    if (first && first === first.toUpperCase()) cap++
   116→  }
   117→  return cap / words.length
   118→}

   120→const NON_HEADING_RE = /^(page|fig(?:ure)?|table|source|note|answer|marks?|time\s*[:–-]|total|name|roll|date|reg(istration)?\s*(no|number))/i

   122→// ─── Structure groups ─────────────────────────────────────────────────────────
   123→
   124→const GROUP_WORD_RE =
   125→  /^(chapter|unit|lesson|module|part|section|appendix|annexure|annex)\s*[-–—:]?\s*([0-9]{1,3}|[ivxlcdm]{1,7}|[a-z])\b/i
   126→const APPENDIX_WORD_RE = /^(appendix|annexure|annex|exhibit|schedule)\b/i
   127→const BARE_ROMAN_TOKEN_RE = /^[ivxlcdm]{1,7}$/i
   128→const CHAPTER_PREFIX_WORD_RE = /^(chapter|unit|lesson|module|part|section)\b/i
   129→
   130→/**
   131→ * Which part of the document's own organisation does a contents entry belong
   132→ * to? Deterministic, purely from the entry's own numbering token and what the
   133→ * contents page announces — never from hardcoded titles.
   134→ *   • "Chapter-4", "Unit 3", "Lesson 2" … → CHAPTER
   135→ *   • "Appendix VII", "Annexure II" …    → APPENDIX
   136→ *   • a bare numeral ("I", "VII") on a contents page that announced itself as
   137→ *     an appendix list → APPENDIX
   138→ *   • entries before the first chapter-numbered row → FRONT (preliminary info)
   139→ *   • everything else → OTHER
   140→ */
   141→export function classifyEntryGroup(
   142→  number: string | null,
   143→  title: string,
   144→  appendixAnnounced: boolean,
   145→  seenChapterNumbering: boolean,
   146→): StructureGroup {
   147→  const token = number?.trim() ?? ''
   148→  if (token) {
   149→    if (/^appendix|^annexure|^annex/i.test(token)) return 'APPENDIX'
   150→    if (CHAPTER_PREFIX_WORD_RE.test(token)) return 'CHAPTER'
   151→    if (BARE_ROMAN_TOKEN_RE.test(token) && appendixAnnounced) return 'APPENDIX'
   152→  }
   153→  if (APPENDIX_WORD_RE.test(title.trim())) return 'APPENDIX'
   154→  if (GROUP_WORD_RE.test(title.trim())) {
   155→    return APPENDIX_WORD_RE.test(title.trim()) ? 'APPENDIX' : 'CHAPTER'
   156→  }
   157→  return seenChapterNumbering ? 'OTHER' : 'FRONT'
   158→}

   160→/** Same classification for a body heading found outside a contents table. */
   161→export function classifyHeadingGroup(title: string): StructureGroup {
   162→  const t = title.trim()
   163→  if (APPENDIX_WORD_RE.test(t)) return 'APPENDIX'
   164→  if (/^(chapter|unit|lesson|module)\s*[-–—:]?\s*([0-9]{1,3}|[ivxlcdm]{1,7}|[a-z])\b/i.test(t)) return 'CHAPTER'
   165→  if (/^(section|part)\s+[-–—:]?\s*[A-Z0-9IVX]{1,4}\b/.test(t)) return 'CHAPTER'
   166→  return 'OTHER'
   167→}

   171→const TOC_LEADER_RE = /^(.{2,90}?)[\s.·⋅•‣⁃…]{2,}(\d{1,3})(?:\s*[-–—]\s*(\d{1,3}))?$/
   172→const TOC_GAP_RE = /^(.{2,90}?)\s{3,}(\d{1,3})(?:\s*[-–—]\s*(\d{1,3}))?$/
   173→// Real contents tables often use a single space: "Chapter-1 Introduction 13-14"
   174→const TOC_TAIL_RE = /^(.{2,90}?)\s+(\d{1,3})(?:\s*[-–—]\s*(\d{1,3}))?$/
   175→const TOC_PAGE_HINT_RE = /^(index|contents?|table of contents|content page no\.?|appendices?|list of appendices?)\s*$/i
   176→const TOC_APPENDIX_HINT_RE = /^appendices?$/i
   177→/** Running footer/header furniture: "Page |2", "- 12 -", "12" */
   178→const TOC_PAGE_LABEL_RE = /^page\s*[|:\-]?\s*\d+/i
   179→const ROW_NUMBER_PREFIX_RE = /^\d{1,2}\.\s/

   181→interface TocPageResult {
   182→  page: number
   183→  entries: TocEntry[]
   184→  appendixAnnounced: boolean
   185→}
   186→
   187→/**
   188→ * Parse one contents page into entries.
   189→ *
   190→ * Real TOCs wrap long rows across several physical lines:
   191→ *   "6. VI Procedure to be adopted for compilation of NTA Score"
   192→ *   "for Multi Shifts Papers in CUET (UG) 2026 58-67"
   193→ *   "8. VIII Certificate regarding physical limitation in an Examinee"
   194→ *   "to Write 71"
   195→ * A naive per-line parse turns those continuations into bogus standalone
   196→ * sections ("for Multi Shifts Papers…", "to Write") and loses the real ones.
   197→ * So lines are walked in order with a PENDING title buffer:
   198→ *   • a line with a trailing page number completes an entry (pending text is
   199→ *     prepended; a pure-number "title" like "2026 74-94" is treated as the
   200→ *     wrapped page of the pending text);
   201→ *   • a plain title-looking line without a page number is held as pending;
   202→ *   • row-number restarts ("1. I List of …"), headers/footers (repeated lines),
   203→ *     page labels and contents-hint lines reset or skip the buffer.
   204→ * As a final safety net, an entry whose title begins lowercase is merged into
   205→ * the previous entry (it is the tail of a wrapped row that carried its own
   206→ * page number).
   207→ */
   208→/** "9. IX" / "10." / "IX" — a row-number (optionally followed by a numeral) with no title text yet.
   209→ *  These START a wrapped TOC row whose title follows on the next physical lines; dropping them
   210→ *  loses the entry's numbering ("9. IX Certificate for person…" became numberless). */
   211→const ROW_NUMBER_NUMERAL_RE = /^(\d{1,2}\s*[.)]\s*)?([ivxlcdm]{1,7})\.?$/i
   212→
   213→function detectTocOnPage(page: ExtractedPage, furniture: Set<string>): { entries: TocEntry[]; appendixAnnounced: boolean } {
   214→  const lines = linesOf(page)
   215→  if (lines.length < TOC_MIN_ENTRIES) return { entries: [], appendixAnnounced: false }
   216→  const entries: TocEntry[] = []
   217→  let pending = ''
   218→  let hintSeen = false
   219→  const appendixAnnounced = lines.slice(0, 6).some(l => TOC_APPENDIX_HINT_RE.test(l))
   220→
   221→  const pushEntry = (rawTitle: string, num: number | null, numEnd: number | null) => {
   222→    const title = cleanTitle(rawTitle, 300)
   223→    if (title.length < 2 || !WORD_RE.test(title)) return
   224→    entries.push({
   225→      title,
   226→      page: Number.isFinite(num) ? num : null,
   227→      pageEnd: numEnd !== null && Number.isFinite(numEnd) ? numEnd : null,
   228→      // the row's numbering token: subsection numbers ("4.1") live in the row
   229→      // PREFIX that stripEntryTitle removes — fall back to the full title so
   230→      // the document's own numbering survives on the entry
   231→      number: extractNumberToken(stripEntryTitle(title)) ?? extractNumberToken(title),
   232→      sourcePage: page.page,
   233→      appendixAnnounced,
   234→    })
   235→  }
   236→
   237→  for (const raw of lines) {
   238→    const line = raw.trim()
   239→    if (!line) continue
   240→    if (TOC_PAGE_HINT_RE.test(line)) { hintSeen = true; pending = ''; continue }
   241→
   242→    const m = TOC_LEADER_RE.exec(line) ?? TOC_GAP_RE.exec(line) ?? TOC_TAIL_RE.exec(line)
   243→    if (m) {
   244→      const rawTitle = m[1].replace(/[\s.·⋅•‣⁃…]+$/g, '').trim()
   245→      const num = Number(m[2])
   246→      const numEnd = m[3] ? Number(m[3]) : null
   247→      // A row-number restart ("1. I List of Central Universities 35") is a NEW
   248→      // row, never the continuation of the pending text above it.
   249→      const startsNewRow = ROW_NUMBER_PREFIX_RE.test(rawTitle) || NUMBERED_L2_RE.test(rawTitle)
   250→      if (PURE_NUMBER_RE.test(rawTitle)) {
   251→        // "2026 74-94" — the wrapped row's tail is part of the title and the
   252→        // number completes the pending entry ("… CUET (UG) – 2026  74-94").
   253→        if (pending) { pushEntry(`${pending} ${rawTitle}`, num, numEnd); pending = '' }
   254→        continue
   255→      }
   256→      if (rawTitle.length < 2 || rawTitle.length > 160 || !WORD_RE.test(rawTitle)) continue
   257→      // Data rows of reference lists ("101  Ahmedabad  Zone  2") start with a
   258→      // multi-digit code and are NOT contents entries — they would flood the
   259→      // TOC with junk sections on list-heavy documents.
   260→      if (/^\d{2,}\s/.test(rawTitle)) continue
   261→      if (startsNewRow || !pending) {
   262→        if (pending) { pending = ''; }
   263→        // lowercase-start titles are wrapped tails of the previous entry
   264→        const prev = entries[entries.length - 1]
   265→        if (prev && /^[a-z(]/.test(rawTitle)) {
   266→          prev.title = cleanTitle(`${prev.title} ${rawTitle}`, 300)
   267→          if (num !== null && !Number.isNaN(num)) prev.page = num
   268→          if (numEnd !== null && !Number.isNaN(numEnd)) prev.pageEnd = numEnd
   269→          continue
   270→        }
   271→        pushEntry(rawTitle, num, numEnd)
   272→      } else {
   273→        pending = `${pending} ${rawTitle}`.slice(0, 400)
   274→        pushEntry(pending, num, numEnd)
   275→        pending = ''
   276→      }
   277→      continue
   278→    }
   279→
   280→    // No page number on this line — hold as pending title text when it can
   281→    // plausibly be one (running headers repeat across pages and are excluded).
   282→    if (TOC_PAGE_LABEL_RE.test(line) || PURE_NUMBER_RE.test(line)) {
   283→      // A bare number after pending title text is the wrapped row's page
   284→      // ("9. IX / Certificate for person … / … in writing / 72").
   285→      if (PURE_NUMBER_RE.test(line) && pending && !TOC_PAGE_LABEL_RE.test(line)) {
   286→        pushEntry(pending, Number(line), null)
   287→        pending = ''
   288→      }
   289→      continue
   290→    }
   291→    // A lone "9. IX" row starts a wrapped entry whose title follows on the
   292→    // next lines — keep it as the pending prefix instead of dropping it.
   293→    if ((entries.length > 0 || hintSeen) && !pending && line.length <= 12 && ROW_NUMBER_NUMERAL_RE.test(line)) {
   294→      pending = line
   295→      continue
   296→    }
   297→    if (furniture.has(normKey(line))) continue
   298→    // Wrapped-row lines inside an established contents table can be long
   299→    // (single-line titles up to ~300 chars); outside one, stay strict.
   300→    const maxPendingLine = entries.length > 0 || hintSeen ? 300 : 120
   301→    if (line.length < 8 || line.length > maxPendingLine) continue
   302→    if (!WORD_RE.test(line)) continue
   303→    if (entries.length === 0 && !hintSeen) continue // title/header before any entry
   304→    pending = pending ? `${pending} ${line}` : line
   305→  }
   306→
   307→  // A page is a TOC page when either a clear majority of its lines are entries,
   308→  // or it announces itself (Index / Contents / Appendices) and has several.
   309→  const announces = hintSeen || lines.slice(0, 6).some(l => TOC_PAGE_HINT_RE.test(l))
   310→  const threshold = announces
   311→    ? Math.max(TOC_MIN_ENTRIES, Math.ceil(lines.length * 0.25))
   312→    : Math.max(TOC_MIN_ENTRIES, Math.ceil(lines.length * 0.5))
   313→  if (entries.length < threshold) return { entries: [], appendixAnnounced: false }
   314→  // Real contents tables point FORWARD through the document; a page whose
   315→  // claimed numbers cycle (1,2,3,4,1,2…) is a data list, not a contents page.
   316→  const numbered = entries.filter(e => e.page !== null) as (TocEntry & { page: number })[]
   317→  if (numbered.length >= TOC_MIN_ENTRIES) {
   318→    let downs = 0
   319→    for (let i = 1; i < numbered.length; i++) if (numbered[i].page < numbered[i - 1].page - 2) downs++
   320→    if (downs / (numbered.length - 1) > 0.25) return { entries: [], appendixAnnounced: false }
   321→  }
   322→  return { entries, appendixAnnounced }
   323→}
   324→
   325→/** Validate a TOC: entry pages must be plausible and broadly increasing. */
   326→function validateToc(entries: TocEntry[], numPages: number): boolean {
   327→  const withPages = entries.filter(e => e.page !== null) as (TocEntry & { page: number })[]
   328→  if (withPages.length < TOC_MIN_ENTRIES) return false
   329→  let inRange = 0
   330→  for (const e of withPages) if (e.page >= 1 && e.page <= numPages) inRange++
   331→  if (inRange / withPages.length < 0.8) return false
   332→  // Broadly increasing: allow local noise (multi-column TOCs reorder), but the
   333→  // overall trend must go up — otherwise the "TOC" is likely a name list.
   334→  let up = 0
   335→  let total = 0
   336→  for (let i = 1; i < withPages.length; i++) {
   337→    total++
   338→    if (withPages[i].page >= withPages[i - 1].page - 2) up++
   339→  }
   340→  return total === 0 || up / total >= 0.75
   341→}

   346→  /^(chapter|unit|section|part|module|appendix|annexure|annex|experiment|activity|practical|lesson)\s*[-–—:]?\s*([0-9ivxlcdm]+|[a-z])\b[\s.:–—-]*(.*)$/i

   353→/** "CHAPTER 10", "Appendix VII", "UNIT 3" — a bare label with no title text */
   354→const BARE_CHAPTER_RE = /^(chapter|unit|part|section|appendix|annexure|annex)\s*[-–—:]?\s*([0-9ivxlcdm]{1,5}|[a-z])\.?$/i

   364→function headingScore(line: string, inTocTitles: Set<string>): { level: 1 | 2; score: number; title: string } | null {
   365→  const raw = line.trim()
   366→  if (raw.length < 3 || raw.length > HEADING_MAX_LEN) return null
   367→  if (PURE_NUMBER_RE.test(raw)) return null
   368→  if (/[.,;:]$/.test(raw) && !NUMBERED_L1_RE.test(raw) && !NUMBERED_L2_RE.test(raw)) return null
   369→  if (NON_HEADING_RE.test(raw)) return null
   370→
   371→  // x.y numbered subsection — strongest hierarchical signal. Long sentence-
   372→  // like titles (numbered instruction paragraphs "8.1 The candidate must show…")
   373→  // are not headings; only TOC-corroborated long rows pass.
   374→  const l2 = NUMBERED_L2_RE.exec(raw)
   375→  if (l2 && l2[3].length >= 3 && WORD_RE.test(l2[3])) {
   376→    const title = l2[3].trim()
   377→    if (title.length <= 60 || inTocTitles.has(normKey(title))) {
   378→      return { level: 2, score: 6, title }
   379→    }
   380→    return null
   381→  }
   382→
   383→  // Chapter / Unit / Appendix … style — keep the document's own wording
   384→  const ch = CHAPTER_WORD_RE.exec(raw)
   385→  if (ch) {
   386→    return { level: 1, score: 7, title: cleanTitle(raw) }
   387→  }
   388→
   389→  // "Section A", "PART — I"
   390→  const sec = SECTION_LABEL_RE.exec(raw)
   391→  if (sec && (sec[3] ?? '').trim().length < 40) {
   392→    return { level: 1, score: 5, title: cleanTitle(raw) }
   393→  }
   394→
   395→  // Numbered top-level heading "12. Reservation Policy". A single-word title
   396→  // with no contents-table corroboration is usually a table row fragment
   397→  // ("1. Total", "2. Marks") — reject it.
   398→  const l1 = NUMBERED_L1_RE.exec(raw)
   399→  if (l1) {
   400→    const title = l1[2].trim()
   401→    // Numbered lines also match QUESTION_LINE_RE (question papers number every
   402→    // item). A numbered line is a question ITEM, not a heading, when it reads
   403→    // like a sentence: asks something, runs long, or is not title/caps-shaped.
   404→    // "1. Interference of Light" is a heading even without a contents table.
   405→    if (QUESTION_LINE_RE.test(raw)) {
   406→      const words = title.split(/\s+/)
   407→      const looksLikeHeading =
   408→        !/[?:,;]$/.test(raw) && words.length <= 8 &&
   409→        (titleCaseRatio(title) >= 0.6 || upperRatio(title) >= 0.8)
   410→      if (!looksLikeHeading) return null
   411→    }
   412→    if (title.length >= 3 && WORD_RE.test(title)) {
   413→      const corroborated = inTocTitles.has(normKey(title))
   414→      const lastWord = title.split(/\s+/).pop() ?? ''
   415→      // a title ending in a preposition is a wrapped fragment ("Choice of"), not a heading
   416→      const fragmentTail = /^(of|and|for|in|with|to|the|a|an|by|on)$/i.test(lastWord)
   417→      if ((title.split(/\s+/).length >= 2 || corroborated) && !fragmentTail) {
   418→        return { level: 1, score: 4 + (corroborated ? 2 : 0), title: cleanTitle(title) }
   419→      }
   420→    }
   421→    return null
   422→  }
   423→
   424→  // ALL-CAPS short line
   425→  const ls = letters(raw)
   426→  if (ls.length >= 4 && upperRatio(raw) >= 0.8 && raw.split(/\s+/).length <= 12) {
   427→    const bonus = inTocTitles.has(normKey(raw)) ? 2 : 0
   428→    return { level: 1, score: 3 + bonus, title: cleanTitle(raw) }
   429→  }
   430→
   431→  // Title-Case short line (two+ capitalised words, no ending punctuation)
   432→  const words = raw.split(/\s+/)
   433→  if (words.length >= 2 && words.length <= 10 && titleCaseRatio(raw) >= 0.7 && raw.length >= 8) {
   434→    const bonus = inTocTitles.has(normKey(raw)) ? 3 : 0
   435→    // Without a TOC match, plain Title-Case lines are weak signals — they are
   436→    // often names, table headers or list items. Only TOC-corroborated ones or
   437→    // clearly heading-ish lines (no verb-ish trailing punctuation) pass.
   438→    if (bonus > 0) return { level: 1, score: 3 + bonus, title: cleanTitle(raw) }
   439→    return null
   440→  }
   441→
   442→  return null
   443→}
   444→
   445→function cleanTitle(t: string, maxLen = 90): string {
   446→  return t.replace(/\s+/g, ' ').replace(/[.:]\s*$/, '').trim().slice(0, maxLen)
   447→}

   449→/** Remove leading row numbers from a TOC entry title: "1.1 About NTA" → "About NTA". */
   450→export function stripEntryTitle(t: string): string {
   451→  return t
   452→    .replace(/^\d{1,2}\.\d{1,2}\s*[.)]?\s*/, '')
   453→    .replace(/^\d{1,2}\s*[.)]\s+/, '')
   454→    .replace(/\s+/g, ' ')
   455→    .trim()
   456→}
   457→
   458→/**
   459→ * Extract the document's own numbering token from a TOC row / heading title,
   460→ * exactly as printed: "Chapter-4 Reservation" → "Chapter-4", "1.1 About NTA" →
   461→ * "1.1", "I List of Central Universities" → "I", "Appendix – VII" → "Appendix – VII".
   462→ * Null when the title carries no explicit numbering.
   463→ */
   464→const NUMBER_TOKEN_RE =
   465→  /^((?:chapter|appendix|annexure|annex|unit|part|section|paper|lesson|module)\s*[-–—:]?\s*(?:[0-9]{1,3}|[ivxlcdm]{1,7}|[a-z])\b|\d{1,2}(?:\.\d{1,2})?\b|[ivxlcdm]{1,7}\b)/i
   466→
   467→export function extractNumberToken(title: string): string | null {
   468→  const m = NUMBER_TOKEN_RE.exec(title.trim())
   469→  if (!m) return null
   470→  const token = m[1].replace(/\s+/g, ' ').trim()
   471→  // A bare roman numeral must be a real numeral row, not the word "I" or "mix"
   472→  if (/^[ivxlcdm]+$/i.test(token) && token.length > 1 && !/^(ii|iii|iv|vi|vii|viii|ix|xi|xii|xiii|xiv|xv|vi{0,3}|i[xv]|xl|xc|cd|cm|dl)/i.test(token)) {
   473→    return null
   474→  }
   475→  return token
   476→}
   477→
   478→/** x.y subsection rows of a contents table ("1.1 About NTA  13") */
   479→const SUBSECTION_ROW_RE = /^\d{1,2}\.\d{1,2}\b/
   480→
   481→/** Normalisation used to correlate TOC titles with body headings. */
   482→export function normKey(s: string): string {
   483→  return s
   484→    .toLowerCase()
   485→    .replace(/[^a-z0-9]+/g, ' ')
   486→    .replace(/\b(the|a|an|of|for|and|in|to|on)\b/g, ' ')
   487→    .replace(/\s+/g, ' ')
   488→    .trim()
   489→}

   491→// ─── Printed page number detection ────────────────────────────────────────────
   492→
   493→const ROMAN_VALUES: Record<string, number> = { i: 1, v: 5, x: 10, l: 50, c: 100, d: 500, m: 1000 }
   494→
   495→function romanToInt(s: string): number | null {
   496→  const str = s.toLowerCase()
   497→  if (!/^[ivxlcdm]+$/.test(str)) return null
   498→  let total = 0
   499→  let prev = 0
   500→  for (let i = str.length - 1; i >= 0; i--) {
   501→    const v = ROMAN_VALUES[str[i]]
   502→    if (v < prev) total -= v
   503→    else {
   504→      total += v
   505→      prev = v
   506→    }
   507→  }
   508→  return total >= 1 ? total : null
   509→}
   510→
   511→function printedLabelFromEdge(lines: string[]): string | null {
   512→  const edges = [...lines.slice(0, 2), ...lines.slice(-2)]
   513→  for (const line of edges) {
   514→    if (line.length > 40) continue
   515→    // "Page 12 of 96", "Page |12", "Page | 12", "12", "- 12 -", "12."
   516→    const m1 = /^page\s*[|:\-]?\s*(\d{1,4})(?:\s*(?:of|\/)\s*\d{1,4})?$/i.exec(line)
   517→    if (m1) return m1[1]
   518→    const m2 = /^[-–—\s]*(\d{1,4})[-–—\s.]*$/.exec(line)
   519→    if (m2) return m2[1]
   520→    if (ROMAN_RE.test(line) && line.length <= 7) {
   521→      const v = romanToInt(line)
   522→      if (v !== null && v <= 100) return line.toLowerCase()
   523→    }
   524→  }
   525→  return null
   526→}
   527→
   528→export function detectPrintedPages(pages: ExtractedPage[]): PrintedPageInfo {
   529→  const pairs: { physical: number; printed: number | null; roman: boolean }[] = []
   530→  const labels: Record<number, string> = {}
   531→  for (const p of pages) {
   532→    if (p.scanned || !p.text) {
   533→      pairs.push({ physical: p.page, printed: null, roman: false })
   534→      continue
   535→    }
   536→    const label = printedLabelFromEdge(linesOf(p))
   537→    if (label === null) {
   538→      pairs.push({ physical: p.page, printed: null, roman: false })
   539→    } else if (/^\d+$/.test(label)) {
   540→      pairs.push({ physical: p.page, printed: Number(label), roman: false })
   541→      labels[p.page] = label
   542→    } else {
   543→      pairs.push({ physical: p.page, printed: romanToInt(label), roman: true })
   544→      labels[p.page] = label
   545→    }
   546→  }
   547→
   548→  // Roman front matter anywhere in the first 15% of the document
   549→  const headCount = Math.max(1, Math.ceil(pairs.length * 0.15))
   550→  const romanFrontMatter = pairs.slice(0, headCount).some(p => p.roman && p.printed !== null)
   551→
   552→  // Longest consistent arabic run: printed = physical + offset
   553→  let bestOffset: number | null = null
   554→  let bestRun = 0
   555→  const byOffset = new Map<number, number>()
   556→  for (const p of pairs) {
   557→    if (p.printed === null || p.roman) continue
   558→    const offset = p.physical - p.printed
   559→    if (offset < 0 || offset > 60) continue // cover/plate offsets beyond 60 are implausible
   560→    byOffset.set(offset, (byOffset.get(offset) ?? 0) + 1)
   561→  }
   562→  for (const [offset, count] of byOffset) {
   563→    if (count > bestRun) {
   564→      bestRun = count
   565→      bestOffset = offset
   566→    }
   567→  }
   568→  // Require the offset to hold for a meaningful share of pages (≥5 and ≥25%)
   569→  const arabicPages = pairs.filter(p => p.printed !== null && !p.roman).length
   570→  if (bestOffset !== null && !(bestRun >= 5 && bestRun >= arabicPages * 0.5 && arabicPages >= pages.length * 0.25)) {
   571→    bestOffset = null
   572→  }
   573→  return { offset: bestOffset, romanFrontMatter, labels }
   574→}

   578→export function detectStructure(pages: ExtractedPage[], numPages: number, opts: BuildUnitsOptions): DetectedStructure {
   579→  const readable = pages.filter(p => !p.scanned && p.text.length > 0)
   580→  const printed = detectPrintedPages(pages)
   581→
   582→  // Running headers/footers repeat across pages; inside a contents table such
   583→  // lines must never be mistaken for a wrapped title row.
   584→  const furniture = new Set<string>()
   585→  {
   586→    const freq = new Map<string, Set<number>>()
   587→    for (const p of readable.slice(0, 15)) {
   588→      for (const l of linesOf(p)) {
   589→        if (l.length < 4 || l.length > 70) continue
   590→        const key = normKey(l)
   591→        if (!key) continue
   592→        const set = freq.get(key) ?? new Set<number>()
   593→        set.add(p.page)
   594→        freq.set(key, set)
   595→      }
   596→    }
   597→    for (const [key, pageSet] of freq) if (pageSet.size >= 2) furniture.add(key)
   598→  }
   599→
   600→  // 1 ── TOC pages
   601→  const tocPages: TocPageResult[] = []
   602→  for (const p of readable.slice(0, 15)) {
   603→    const { entries, appendixAnnounced } = detectTocOnPage(p, furniture)
   604→    if (entries.length > 0) tocPages.push({ page: p.page, entries, appendixAnnounced })
   605→  }
   606→  const allTocEntries = tocPages.flatMap(t => t.entries)
   607→  const tocValidated = validateToc(allTocEntries, numPages)
   608→  // Titles stripped of leading row numbers ("1.1 About NTA" → "About NTA") so
   609→  // TOC entries correlate with the body's own headings.
   610→  const tocTitleKeys = new Set(allTocEntries.map(e => normKey(stripEntryTitle(e.title))))
   611→
   612→  // 2 ── Heading candidates over all readable lines
   613→  const hits: LineHit[] = []
   614→  let questionLines = 0
   615→  let contentLines = 0
   616→  for (const p of readable) {
   617→    const onTocPage = tocPages.some(t => t.page === p.page)
   618→    const lines = linesOf(p)
   619→    for (let i = 0; i < lines.length; i++) {
   620→      const line = lines[i]
   621→      if (line.length >= 12) contentLines++
   622→      if (QUESTION_LINE_RE.test(line)) questionLines++
   623→      if (onTocPage) continue // TOC lines look like headings but are not
   624→      const h = headingScore(line, tocTitleKeys)
   625→      if (!h) continue
   626→      // Bare "CHAPTER 10" / "APPENDIX VII" headings: the real title often sits
   627→      // on the next line — merge it so the topic keeps the document's wording.
   628→      let title = h.title
   629→      let consumedExtra = 0
   630→      const bare = BARE_CHAPTER_RE.exec(h.title)
   631→      if (bare) {
   632→        const next = lines[i + 1]
   633→        if (
   634→          next && next.length >= 3 && next.length <= 90 && !PURE_NUMBER_RE.test(next) &&
   635→          !NON_HEADING_RE.test(next) && !headingScore(next, tocTitleKeys) && !QUESTION_LINE_RE.test(next)
   636→        ) {
   637→          title = cleanTitle(`${h.title} ${next}`)
   638→          consumedExtra = 1
   639→        }
   640→      }
   641→      hits.push({ title, level: h.level, score: h.score, page: p.page, lineIdx: i })
   642→      i += consumedExtra
   643→    }
   644→  }
   645→
   646→  // 3 ── Question-paper layout? High numbered-line density + explicit marks
   647→  // instructions near the front — both together, so ordinary numbered lists in
   648→  // textbooks (or a phone directory) are not misclassified.
   649→  const marksMention = readable
   650→    .slice(0, 8)
   651→    .some(p => /marks?\s*[:：]|max(?:imum)?\s*marks|time\s*(?:allowed|:)\s*\d/i.test(p.text))
   652→  const questionPaper =
   653→    contentLines > 30 && questionLines / Math.max(1, contentLines) >= QUESTION_LINE_RATIO && marksMention
   654→
   655→  // 4 ── Heading density sanity: too many "headings" ⇒ they are not headings
   656→  let headingHits = hits.filter(h => h.score >= 4)
   657→  const density = contentLines > 0 ? headingHits.length / contentLines : 0
   658→  if (density > HEADING_DENSITY_LIMIT) headingHits = headingHits.filter(h => h.score >= 6)
   659→  if (contentLines > 0 && headingHits.length / contentLines > HEADING_DENSITY_LIMIT) headingHits = []
   660→
   661→  // 5 ── Corroborate with the TOC: boost + mark
   662→  const headings: HeadingHit[] = []
   663→  let lastKey = ''
   664→  for (const h of headingHits) {
   665→    const key = normKey(h.title)
   666→    if (!key || key === lastKey) continue // consecutive duplicate line
   667→    lastKey = key
   668→    headings.push({
   669→      title: h.title,
   670→      page: h.page,
   671→      lineIdx: h.lineIdx,
   672→      level: h.level,
   673→      score: h.score,
   674→      fromToc: tocTitleKeys.has(key),
   675→    })
   676→  }
   677→
   678→  // Drop weak one-off headings when a TOC exists (they are usually noise);
   679→  // keep everything when the TOC corroborates them.
   680→  const filtered = headings.filter(h => !tocValidated || h.fromToc || h.score >= 4)
   681→
   682→  // 6 ── Units. When a validated TOC exists, the document's own contents table
   683→  // (validated against the body) defines the major sections — that is more
   684→  // faithful than any heading heuristic. Headings are the fallback.
   685→  let units: AnalysisUnit[] | null = null
   686→  let kind: StructureKind
   687→  if (questionPaper) {
   688→    kind = 'QUESTION_PAPER'
   689→  } else {
   690→    if (tocValidated) units = buildTocUnits(readable, allTocEntries, printed.offset, opts)
   691→    if (units && units.length >= 2) {
   692→      kind = 'TOC_GUIDED'
   693→    } else {
   694→      units = null
   695→      kind = filtered.length >= 3 ? (tocValidated ? 'TOC_GUIDED' : 'HEADING_GUIDED') : 'UNSTRUCTURED'
   696→    }
   697→  }
   698→  if (!units) units = buildUnits(readable, filtered, kind, opts)
   699→
   700→  return {
   701→    kind,
   702→    toc: allTocEntries,
   703→    tocValidated,
   704→    headings: filtered,
   705→    printed,
   706→    units,
   707→    tocPages: tocPages.map(t => t.page),
   708→    tocLayer: verifyTocLayer(allTocEntries, readable, printed.offset, numPages),
   709→  }
   710→}
   711→
   712→// ─── Contentless pages: extracted text that is only running furniture ────────
   713→
   714→/**
   715→ * Detect pages whose extracted text carries NO real content — only running
   716→ * headers/footers and page labels (form facsimiles, Gazette reprints, image
   717→ * plates that still print the document's furniture). These pages pass the
   718→ * naive char-count threshold but contain nothing analysable; the coverage
   719→ * report must count them as their own bucket instead of folding them into
   720→ * "readable content".
   721→ *
   722→ * Deterministic: a short line that appears (normalised) on ≥3 different pages
   723→ * or on ≥8% of pages is furniture; a page whose remaining text has fewer than
   724→ * CONTENTLESS_CHARS characters of non-furniture content is contentless.
   725→ */
   726→export function detectContentlessPages(pages: ExtractedPage[]): number[] {
   727→  const candidates = pages.filter(p => !p.scanned && p.text.length > 0)
   728→  if (candidates.length < 4) return []
   729→  // Frequency of short lines across pages (running headers/footers repeat)
   730→  const freq = new Map<string, Set<number>>()
   731→  for (const p of candidates) {
   732→    for (const l of linesOf(p)) {
   733→      if (l.length < 4 || l.length > 80) continue
   734→      const key = normKey(l)
   735→      if (!key) continue
   736→      const set = freq.get(key) ?? new Set<number>()
   737→      set.add(p.page)
   738→      freq.set(key, set)
   739→    }
   740→  }
   741→  const minPages = Math.max(3, Math.ceil(candidates.length * 0.08))
   742→  const furniture = new Set([...freq.entries()].filter(([, set]) => set.size >= minPages).map(([key]) => key))
   743→  if (furniture.size === 0) return []
   744→  const CONTENTLESS_CHARS = 32
   745→  const out: number[] = []
   746→  for (const p of candidates) {
   747→    let content = 0
   748→    for (const l of linesOf(p)) {
   749→      // page-label lines ("Page |12", "12", "- 12 -") are furniture everywhere
   750→      if (TOC_PAGE_LABEL_RE.test(l) || PURE_NUMBER_RE.test(l)) continue
   751→      if (furniture.has(normKey(l))) continue
   752→      content += l.length
   753→    }
   754→    if (content < CONTENTLESS_CHARS) out.push(p.page)
   755→  }
   756→  return out
   757→}

   759→// ─── Analysis units ───────────────────────────────────────────────────────────
   760→
   761→/**
   762→ * Verify one contents entry against the body text.
   763→ * Returns the physical page that carries the entry's words, or null.
   764→ *
   765→ * The check is evidence-first and never trusts the claimed number blindly:
   766→ *   1. the entry's meaningful tokens must appear on the claimed physical page;
   767→ *   2. failing that, they may appear anywhere in the entry's plausible window
   768→ *      (claimed page … page before the next entry's claim) — headings in real
   769→ *      documents often sit on an image-only claimed page while their text
   770→ *      starts on the next readable page (a Gazette reprint, a fold-out table);
   771→ *   3. an entry whose claimed page has NO readable text is never "disproven" —
   772→ *      it is unverifiable, and the caller decides honestly what to record.
   773→ */
   774→function matchEntryInBody(
   775→  title: string,
   776→  claimedPhysical: number | null,
   777→  windowTo: number,
   778→  pageByNumber: Map<number, ExtractedPage>,
   779→  numberToken?: string | null,
   780→): { matchPage: number | null; claimedReadable: boolean; occurrences: number[] } {
   781→  if (claimedPhysical === null) return { matchPage: null, claimedReadable: false, occurrences: [] }
   782→  const claimedPage = pageByNumber.get(claimedPhysical)
   783→  const claimedReadable = !!claimedPage && !claimedPage.scanned && claimedPage.text.trim().length >= 40
   784→  const tokens = normKey(title).split(' ').filter(t => t.length > 3)
   785→  if (tokens.length === 0) return { matchPage: claimedPhysical, claimedReadable, occurrences: claimedReadable ? [claimedPhysical] : [] }
   786→  // A claimed page that carries the section's own heading line ("Appendix VII",
   787→  // "Chapter-4") is a positive match even when the rest of the page is an
   788→  // image-only reprint — the heading is body evidence the section starts there.
   789→  const numeral = numberToken
   790→    ? /^(?:chapter|appendix|annexure|annex|unit|part|section|paper|lesson|module)?\s*[-–—:]?\s*([ivxlcdm]{1,7}|\d{1,3})$/i.exec(numberToken.trim())?.[1]
   791→    : null
   792→  const headingLineRe = numeral
   793→    ? new RegExp(`^(?:append(?:ix|ance)|annexure|annex|chapter|unit|part|section|paper|lesson|module)\\s*[-–—:]?\\s*${numeral}\\b`, 'i')
   794→    : null
   795→  const from = claimedPhysical
   796→  const to = Math.max(claimedPhysical, Math.min(windowTo, claimedPhysical + 30))
   797→  const occurrences: number[] = []
   798→  for (let p = from; p <= to; p++) {
   799→    const page = pageByNumber.get(p)
   800→    if (!page || page.scanned || page.text.trim().length === 0) continue
   801→    // Token evidence needs substantial text; a heading-line-only page (an
   802→    // image-only appendix reprint that still prints its own heading) can
   803→    // still prove the section starts exactly where the contents says.
   804→    if (page.text.trim().length >= 40) {
   805→      const hay = normText(page.text)
   806→      if (tokens.some(t => hay.includes(t))) occurrences.push(p)
   807→    }
   808→    if (headingLineRe && p === claimedPhysical && !occurrences.includes(p)) {
   809→      const hasHeadingLine = page.text.split('\n').some(l => headingLineRe.test(l.trim()))
   810→      if (hasHeadingLine) occurrences.push(p)
   811→    }
   812→  }
   813→  return { matchPage: occurrences[0] ?? null, claimedReadable, occurrences }
   814→}
   815→
   816→const normText = (s: string) => s.toLowerCase().replace(/[^a-z0-9]+/g, ' ')
   817→
   818→/**
   819→ * Build the ORIGINAL-STRUCTURE record of the document's own contents table:
   820→ * every entry with its printed claim, physical claim, body match status, an
   821→ * honest note when it could not be verified, its group (front matter / main
   822→ * chapters / appendices / other) and the boundary evidence used to establish
   823→ * its page span. Unmatched entries are recorded, never silently dropped, and
   824→ * never treated as unquestionable ground truth.
   825→ */
   826→export function verifyTocLayer(
   827→  entries: TocEntry[],
   828→  readable: ExtractedPage[],
   829→  printedOffset: number | null,
   830→  numPages: number,
   831→  cap = 200,
   832→): StructureTocEntry[] {
   833→  const pageByNumber = new Map(readable.map(p => [p.page, p] as const))
   834→  const physicalOf = (printed: number) => (printedOffset !== null ? printed + printedOffset : printed)
   835→  const lastPage = numPages > 0 ? numPages : (readable[readable.length - 1]?.page ?? 0)
   836→  const out: StructureTocEntry[] = []
   837→  let seenChapterNumbering = false
   838→  let parentGroup: StructureGroup = 'FRONT'
   839→  for (let i = 0; i < entries.length && out.length < cap; i++) {
   840→    const e = entries[i]
   841→    const physical = e.page !== null ? physicalOf(e.page) : null
   842→    // window: up to the next major entry's claim
   843→    let windowTo = physical ?? 0
   844→    for (let j = i + 1; j < entries.length; j++) {
   845→      const nxt = entries[j]
   846→      if (nxt.page === null) continue
   847→      const nxtPhysical = physicalOf(nxt.page)
   848→      if (Number.isFinite(nxtPhysical) && nxtPhysical > (physical ?? 0)) {
   849→        windowTo = nxtPhysical - 1
   850→        break
   851→      }
   852→    }
   853→    const level: 1 | 2 = SUBSECTION_ROW_RE.test(e.title.trim()) ? 2 : 1
   854→    let group = classifyEntryGroup(e.number, e.title, e.appendixAnnounced, seenChapterNumbering)
   855→    if (group === 'CHAPTER') seenChapterNumbering = true
   856→    // a numbered subsection belongs to the part of the document its parent
   857→    // section lives in ("4.1 Reservations" is part of the chapters layer)
   858→    if (level === 2 && group === 'OTHER') group = parentGroup
   859→    if (level === 1) parentGroup = group
   860→
   861→    const { matchPage, claimedReadable, occurrences } = matchEntryInBody(
   862→      stripEntryTitle(e.title), physical, windowTo, pageByNumber, e.number,
   863→    )
   864→
   865→    // ── start page + how it was established
   866→    const startPage = matchPage ?? physical
   867→    let startStatus: PageMappingStatus
   868→    let note: string | undefined
   869→    if (startPage === null) {
   870→      startStatus = 'UNRESOLVED'
   871→      note = 'The contents entry states no page number, and no body page carried its words.'
   872→    } else if (matchPage !== null && physical !== null && matchPage === physical) {
   873→      startStatus = 'VERIFIED'
   874→    } else if (matchPage !== null) {
   875→      // words found, but not at the claimed page — record the discrepancy
   876→      startStatus = 'INFERRED'
   877→      note = `The entry's words were found on physical page ${matchPage}, while the contents table claims ${physical ?? '?'} (printed ${e.page ?? '?'}).`
   878→    } else if (!claimedReadable) {
   879→      startStatus = 'INFERRED'
   880→      note = `The claimed page (physical ${physical}) has no readable text — image-only or blank — so the start could not be verified against the body; it follows from the contents claim${printedOffset !== null ? ` (+${printedOffset}-page offset)` : ''}.`
   881→    } else {
   882→      startStatus = 'INFERRED'
   883→      note = `No body page carried this entry's words at its claimed page (physical ${physical}); the start follows from the contents claim.`
   884→    }
   885→
   886→    // ── end boundary: the contents row's own printed range when it has one,
   887→    // bounded by where the next entry AT THE SAME OR HIGHER LEVEL starts
   888→    // (whichever is stricter). A child subsection ("4.1" inside Chapter-4)
   889→    // must never cut its parent's range short.
   890→    const printedEndPhysical = e.pageEnd !== null && printedOffset !== null ? e.pageEnd + printedOffset : null
   891→    let nextStart: number | null = null
   892→    for (let j = i + 1; j < entries.length; j++) {
   893→      const nxt = entries[j]
   894→      if (nxt.page === null) continue
   895→      const nxtPhysical = physicalOf(nxt.page)
   896→      if (!Number.isFinite(nxtPhysical) || nxtPhysical <= (startPage ?? 0)) continue
   897→      const nxtLevel: 1 | 2 = SUBSECTION_ROW_RE.test(nxt.title.trim()) ? 2 : 1
   898→      if (level === 1 && nxtLevel === 2) continue // a child never bounds its parent
   899→      nextStart = nxtPhysical
   900→      break
   901→    }
   902→    let endPage: number | null = null
   903→    let boundaryStatus: PageMappingStatus
   904→    const evidenceBits: string[] = []
   905→    if (printedEndPhysical !== null && nextStart !== null) {
   906→      endPage = Math.min(printedEndPhysical, nextStart - 1)
   907→      boundaryStatus = 'VERIFIED'
   908→      evidenceBits.push(`the contents row prints pages ${e.page}–${e.pageEnd}`, `the next entry starts at physical ${nextStart}`)
   909→    } else if (printedEndPhysical !== null) {
   910→      endPage = printedEndPhysical
   911→      boundaryStatus = 'VERIFIED'
   912→      evidenceBits.push(`the contents row prints pages ${e.page}–${e.pageEnd}`)
   913→    } else if (nextStart !== null) {
   914→      endPage = nextStart - 1
   915→      boundaryStatus = 'INFERRED'
   916→      evidenceBits.push(`ends where the next contents entry starts (physical ${nextStart})`)
   917→    } else {
   918→      endPage = startPage !== null ? Math.max(startPage, lastPage) : null
   919→      boundaryStatus = endPage !== null && endPage === lastPage ? 'INFERRED' : 'UNRESOLVED'
   920→      evidenceBits.push(endPage === lastPage ? 'runs to the end of the document' : 'no end evidence available')
   921→    }
   922→    if (endPage !== null && startPage !== null && endPage < startPage) {
   923→      endPage = startPage // same-page siblings share one physical page
   924→      if (!evidenceBits.some(b => b.includes('same physical page'))) {
   925→        evidenceBits.push(`the next entry starts on the same physical page (${startPage})`)
   926→      }
   927→    }
   928→    // sanity: never run past the actual document
   929→    if (endPage !== null) endPage = Math.min(endPage, lastPage)
   930→    if (occurrences.length > 1) {
   931→      evidenceBits.push(`the entry's words appear on ${occurrences.length} pages (${occurrences.slice(0, 4).join(', ')}${occurrences.length > 4 ? '…' : ''})`)
   932→    }
   933→
   934→    out.push({
   935→      title: e.title,
   936→      number: e.number,
   937→      printedPage: e.page,
   938→      printedPageEnd: e.pageEnd,
   939→      physicalPage: physical,
   940→      level,
   941→      matched: matchPage !== null,
   942→      matchPage,
   943→      note,
   944→      group,
   945→      tocPage: e.sourcePage,
   946→      startPage,
   947→      endPage,
   948→      startStatus,
   949→      boundaryStatus,
   950→      boundaryEvidence: evidenceBits.join('; '),
   951→      occurrences: occurrences.length,
   952→    })
   953→  }
   954→  return out
   955→}

   957→/**
   958→ * Build units directly from a VALIDATED table of contents: each major (non
   959→ * x.y) entry becomes a section whose physical start page is the entry's page
   960→ * (printed + detected offset) — but only when the body page actually contains
   961→ * the entry's words, OR the claimed page is unreadable (image-only reprint:
   962→ * the entry stays, honestly unverified — dropping it erased whole appendices
   963→ * from the analysis). If too few entries verify, returns null so the caller
   964→ * falls back to heading-anchored units. This is what makes page references
   965→ * land on the right page in documents with a real contents table.
   966→ */
   967→export function buildTocUnits(
   968→  readable: ExtractedPage[],
   969→  entries: TocEntry[],
   970→  printedOffset: number | null,
   971→  opts: BuildUnitsOptions,
   972→): AnalysisUnit[] | null {
   973→  const physicalOf = (printed: number) => (printedOffset !== null ? printed + printedOffset : printed)
   974→  const pageByNumber = new Map(readable.map(p => [p.page, p] as const))
   975→  const normText = (s: string) => s.toLowerCase().replace(/[^a-z0-9]+/g, ' ')
   976→
   977→  const majors = entries
   978→    .filter(e => e.page !== null && !SUBSECTION_ROW_RE.test(e.title.trim()))
   979→    .map(e => ({ title: stripEntryTitle(e.title), physical: physicalOf(e.page as number), number: e.number }))
   980→    .filter(x => x.title.length >= 2 && x.title.length <= 300)
   981→  if (majors.length < TOC_MIN_ENTRIES) return null
   982→
   983→  // Wrapped TOC rows continue on the next line claiming the same page —
   984→  // "…General Aptitude" + "Test under CUET (UG) – 2026 37-38" are ONE section.
   985→  const mergedMajors: { title: string; physical: number; number: string | null }[] = []
   986→  for (const m of majors) {
   987→    const prev = mergedMajors[mergedMajors.length - 1]
   988→    if (prev && prev.physical === m.physical) {
   989→      prev.title = cleanTitle(`${prev.title} ${m.title}`, 300)
   990→      continue
   991→    }
   992→    mergedMajors.push({ ...m })
   993→  }
   994→  if (mergedMajors.length < TOC_MIN_ENTRIES) return null
   995→
   996→  // Verify each entry against the body. An entry whose claimed page carries
   997→  // its words is verified; an entry whose claimed page has NO readable text
   998→  // is unverifiable but KEPT (the document's own claim — dropping it erased
   999→  // image-reprint appendices); only an entry contradicted by a readable
  1000→  // claimed page is dropped here (the TOC layer records it either way).
  1001→  // The entry's own numbering token is passed so an image-only claimed page
  1002→  // that still prints its heading line ("Appendix VII") verifies — without
  1003→  // it, such an entry vanished and its pages were absorbed by the previous
  1004→  // section's range.
  1005→  const verified: { title: string; physical: number }[] = []
  1006→  for (let i = 0; i < mergedMajors.length; i++) {
  1007→    const m = mergedMajors[i]
  1008→    const windowTo = i + 1 < mergedMajors.length ? mergedMajors[i + 1].physical - 1 : m.physical + 30
  1009→    const { matchPage, claimedReadable } = matchEntryInBody(m.title, m.physical, windowTo, pageByNumber, m.number)
  1010→    if (matchPage !== null || !claimedReadable) verified.push(m)
  1011→  }
  1012→  if (verified.length < TOC_MIN_ENTRIES || verified.length < Math.ceil(mergedMajors.length * 0.5)) return null
  1013→
  1014→  verified.sort((a, b) => a.physical - b.physical)
  1015→  const cleaned = verified.filter((v, i) => i === 0 || v.physical > verified[i - 1].physical)
  1016→  if (cleaned.length < TOC_MIN_ENTRIES) return null
  1017→
  1018→  const units: AnalysisUnit[] = []
  1019→  const textOf = (from: number, to: number) =>
  1020→    readable
  1021→      .filter(p => p.page >= from && p.page <= to)
  1022→      .map(p => `[page ${p.page}]\n${p.text}`)
  1023→      .join('\n\n')
  1024→
  1025→  const pushRange = (title: string | null, from: number, to: number, kind: AnalysisUnit['kind']) => {
  1026→    const pagesSlice = readable.filter(p => p.page >= from && p.page <= to)
  1027→    if (pagesSlice.length === 0) return
  1028→    // oversize ranges split into consecutive chunks under the same title
  1029→    let start = pagesSlice[0]
  1030→    let prev = pagesSlice[0]
  1031→    let chars = 0
  1032→    const flush = () => {
  1033→      units.push({
  1034→        index: units.length, title, level: title ? 1 : null,
  1035→        fromPage: start.page, toPage: prev.page, text: textOf(start.page, prev.page),
  1036→        chars, kind,
  1037→      })
  1038→    }
  1039→    for (const p of pagesSlice) {
  1040→      if (chars > 0 && chars + p.text.length > opts.maxUnitChars) {
  1041→        flush()
  1042→        start = p
  1043→        chars = 0
  1044→      }
  1045→      chars += p.text.length
  1046→      prev = p
  1047→    }
  1048→    if (chars > 0) flush()
  1049→  }
  1050→
  1051→  // Front matter: everything before the first verified entry
  1052→  const first = cleaned[0].physical
  1053→  const front = readable.filter(p => p.page < first)
  1054→  if (front.length > 0) pushRange(null, front[0].page, front[front.length - 1].page, 'FRONT')
  1055→
  1056→  for (let i = 0; i < cleaned.length; i++) {
  1057→    const from = cleaned[i].physical
  1058→    const to = i + 1 < cleaned.length ? Math.min(cleaned[i + 1].physical - 1, readable[readable.length - 1].page) : readable[readable.length - 1].page
  1059→    if (to < from) continue
  1060→    pushRange(cleaned[i].title, from, to, 'SECTION')
  1061→  }
  1062→
  1063→  return units.length >= 2 ? units.map((u, i) => ({ ...u, index: i })) : null
  1064→}
  1065→
  1066→/**
  1067→ * Build analysis units that cover every readable page exactly once.
  1068→ * TOC/heading-guided documents are sliced at real heading boundaries (level-1
  1069→ * sections, split at level-2 subsections when oversize); everything else falls
  1070→ * back to whole-document page chunks. Tiny neighbouring sections are merged so
  1071→ * one-line headings don't each cost a model call, and oversize sections are
  1072→ * split further so no single call exceeds the char budget.
  1073→ */
  1074→export function buildUnits(
  1075→  readable: ExtractedPage[],
  1076→  headings: HeadingHit[],
  1077→  kind: StructureKind,
  1078→  opts: BuildUnitsOptions,
  1079→): AnalysisUnit[] {
  1080→  const units: AnalysisUnit[] = []
  1081→
  1082→  const pushUnit = (u: Omit<AnalysisUnit, 'index'>) => {
  1083→    units.push({ ...u, index: units.length })
  1084→  }
  1085→
  1086→  const textOf = (from: ExtractedPage, to: ExtractedPage) =>
  1087→    readable
  1088→      .slice(readable.indexOf(from), readable.indexOf(to) + 1)
  1089→      .map(p => `[page ${p.page}]\n${p.text}`)
  1090→      .join('\n\n')
  1091→
  1092→  if (kind === 'TOC_GUIDED' || kind === 'HEADING_GUIDED') {
  1093→    // Anchor each level-1 heading to its page; sections span to the next level-1 anchor.
  1094→    const anchors = headings
  1095→      .filter(h => h.level === 1)
  1096→      .sort((a, b) => a.page - b.page || a.lineIdx - b.lineIdx)
  1097→    // Collapse anchors that land on the same page with the same key (repeat prints)
  1098→    const deduped: HeadingHit[] = []
  1099→    for (const a of anchors) {
  1100→      const prev = deduped[deduped.length - 1]
  1101→      if (prev && prev.page === a.page && normKey(prev.title) === normKey(a.title)) continue
  1102→      deduped.push(a)
  1103→    }
  1104→
  1105→
  1106→    // Front matter: readable pages before the first anchor
  1107→    const firstAnchorPage = deduped[0]?.page ?? null
  1108→    const frontPages = firstAnchorPage ? readable.filter(p => p.page < firstAnchorPage) : []
  1109→    if (frontPages.length > 0) {
  1110→      const from = frontPages[0]
  1111→      const to = frontPages[frontPages.length - 1]
  1112→      pushUnit({ title: null, level: null, fromPage: from.page, toPage: to.page, text: textOf(from, to), chars: frontPages.reduce((s, p) => s + p.text.length, 0), kind: 'FRONT' })
  1113→    }
  1114→
  1115→    for (let i = 0; i < deduped.length; i++) {
  1116→      const anchor = deduped[i]
  1117→      const nextAnchor = deduped[i + 1] ?? null
  1118→      const nextPage = nextAnchor?.page ?? null
  1119→      // Same-page sibling sections ("2. Diffraction" ends where "3. Photo-
  1120→      // electric Effect" begins on the SAME page — dense notes, handouts):
  1121→      // the page-level span would be empty, so split the shared page by the
  1122→      // anchors' line positions instead of dropping the section.
  1123→      if (nextPage === anchor.page && nextAnchor) {
  1124→        const page = readable.find(p => p.page === anchor.page)
  1125→        if (page) {
  1126→          const lines = linesOf(page)
  1127→          const text = lines.slice(anchor.lineIdx, nextAnchor.lineIdx).join('\n')
  1128→          if (text.trim().length > 0) {
  1129→            pushUnit({
  1130→              title: anchor.title,
  1131→              level: 1,
  1132→              fromPage: anchor.page,
  1133→              toPage: anchor.page,
  1134→              text: `[page ${anchor.page}]\n${text}`,
  1135→              chars: text.length,
  1136→              kind: 'SECTION',
  1137→            })
  1138→          }
  1139→        }
  1140→        continue
  1141→      }
  1142→      const sectionPagesList = readable.filter(p => p.page >= anchor.page && (nextPage === null || p.page < nextPage))
  1143→      if (sectionPagesList.length === 0) continue
  1144→      // Split oversize sections at level-2 heading pages within the section
  1145→      const subAnchors = headings
  1146→        .filter(h => h.level === 2 && h.page >= anchor.page && (nextPage === null || h.page < nextPage))
  1147→        .sort((a, b) => a.page - b.page)
  1148→      const segments: { from: ExtractedPage; to: ExtractedPage; title: string | null }[] = []
  1149→      let cursorPage = anchor.page
  1150→      for (const sub of subAnchors) {
  1151→        if (sub.page <= cursorPage) continue
  1152→        const from = readable.find(p => p.page === cursorPage)
  1153→        const to = readable.find(p => p.page === sub.page - 1) ?? readable.find(p => p.page === cursorPage)
  1154→        if (from && to) segments.push({ from, to, title: null })
  1155→        cursorPage = sub.page
  1156→      }
  1157→      {
  1158→        const from = readable.find(p => p.page === cursorPage)
  1159→        const to = sectionPagesList[sectionPagesList.length - 1]
  1160→        if (from && to) segments.push({ from, to, title: null })
  1161→      }
  1162→
  1163→      let pendingFrom: ExtractedPage | null = null
  1164→      let pendingChars = 0
  1165→      let pendingStartPage = anchor.page
  1166→      const flushPending = (upto: ExtractedPage) => {
  1167→        if (!pendingFrom) return
  1168→        pushUnit({
  1169→          title: anchor.title,
  1170→          level: 1,
  1171→          fromPage: pendingStartPage,
  1172→          toPage: upto.page,
  1173→          text: textOf(pendingFrom, upto),
  1174→          chars: pendingChars,
  1175→          kind: 'SECTION',
  1176→        })
  1177→        pendingFrom = null
  1178→        pendingChars = 0
  1179→      }
  1180→      for (const seg of segments) {
  1181→        const segPages = readable.filter(p => p.page >= seg.from.page && p.page <= seg.to.page)
  1182→        const segChars = segPages.reduce((s, p) => s + p.text.length, 0)
  1183→        if (!pendingFrom) {
  1184→          pendingFrom = segPages[0]
  1185→          pendingStartPage = pendingFrom.page
  1186→          pendingChars = segChars
  1187→        } else {
  1188→          pendingChars += segChars
  1189→        }
  1190→        if (pendingChars >= opts.maxUnitChars) flushPending(seg.to)
  1191→      }
  1192→      flushPending(segments[segments.length - 1]?.to ?? sectionPagesList[sectionPagesList.length - 1])
  1193→    }
  1194→
  1195→    // Sanity: any readable pages beyond the last anchor (appendices etc.)
  1196→    const lastCovered = units.reduce((m, u) => Math.max(m, u.toPage), 0)
  1197→    const tail = readable.filter(p => p.page > lastCovered)
  1198→    if (tail.length > 0) {
  1199→      pushUnit({
  1200→        title: deduped.length > 0 ? 'Appendix & back matter' : null,
  1201→        level: deduped.length > 0 ? 1 : null,
  1202→        fromPage: tail[0].page,
  1203→        toPage: tail[tail.length - 1].page,
  1204→        text: textOf(tail[0], tail[tail.length - 1]),
  1205→        chars: tail.reduce((s, p) => s + p.text.length, 0),
  1206→        kind: 'SECTION',
  1207→      })
  1208→    }
  1209→
  1210→    // Merge tiny units forward (a 1-line heading should not cost a model call)
  1211→    return mergeTinyUnits(units, opts.maxUnitChars)
  1212→  }
  1213→
  1214→  // Fallback: UNSTRUCTURED / QUESTION_PAPER — page chunks across the WHOLE doc
  1215→  let cur: ExtractedPage[] = []
  1216→  let curChars = 0
  1217→  const flush = () => {
  1218→    if (cur.length === 0) return
  1219→    const from = cur[0]
  1220→    const to = cur[cur.length - 1]
  1221→    pushUnit({
  1222→      title: null,
  1223→      level: null,
  1224→      fromPage: from.page,
  1225→      toPage: to.page,
  1226→      text: textOf(from, to),
  1227→      chars: curChars,
  1228→      kind: kind === 'QUESTION_PAPER' ? 'CHUNK' : 'CHUNK',
  1229→    })
  1230→    cur = []
  1231→    curChars = 0
  1232→  }
  1233→  for (const p of readable) {
  1234→    if (p.text.length > opts.maxUnitChars) {
  1235→      flush()
  1236→      const text = `[page ${p.page}]\n${p.text.slice(0, opts.maxUnitChars)}`
  1237→      pushUnit({ title: null, level: null, fromPage: p.page, toPage: p.page, text, chars: text.length, kind: 'OVERFLOW' })
  1238→      continue
  1239→    }
  1240→    if (curChars + p.text.length > opts.maxUnitChars) flush()
  1241→    cur.push(p)
  1242→    curChars += p.text.length
  1243→  }
  1244→  flush()
  1245→  return units
  1246→}
  1247→
  1248→function mergeTinyUnits(units: AnalysisUnit[], maxUnitChars: number): AnalysisUnit[] {
  1249→  const out: AnalysisUnit[] = []
  1250→  for (const u of units) {
  1251→    const prev = out[out.length - 1]
  1252→    const tiny = u.chars < MIN_UNIT_CHARS
  1253→    if (
  1254→      tiny && prev && prev.kind === 'SECTION' && u.kind === 'SECTION' &&
  1255→      prev.title === u.title && // never fuse two different sections — a short chapter is still a chapter
  1256→      prev.chars + u.chars <= maxUnitChars
  1257→    ) {
  1258→      prev.toPage = u.toPage
  1259→      prev.text = `${prev.text}\n\n${u.text}`
  1260→      prev.chars += u.chars
  1261→      continue
  1262→    }
  1263→    out.push({ ...u })
  1264→  }
  1265→  return out.map((u, i) => ({ ...u, index: i }))
  1266→}
