     1→// ─── Riishe document pipeline: extraction → structure → grounded analysis ────
     2→// Runs as a background job.
     3→//   Stage 1  extraction (local pdf.js, per-page text with real line breaks)
     4→//   Stage 2  deterministic structure detection (TOC, headings, printed pages)
     5→//   Stage 3  AI analysis over EVERY structure-aware section of the document —
     6→//            there is no "first N pages" limit; a section that fails is
     7→//            recorded and retried without discarding the sections that worked
     8→//   Stage 4  deterministic merge: duplicate topics merged at the data level,
     9→//            page references verified against the extracted page text,
    10→//            coverage accounted honestly (extracted / OCR-needed / failed)
    11→// The uploaded material is untrusted data — every model call fences it.

    46→  printedPage: number | null
    47→  matchPage: number | null

    51→export function buildTocInfoMap(tocLayer: StructureTocEntry[]): Map<string, TocInfo> {
    52→  const map = new Map<string, TocInfo>()
    53→  for (const e of tocLayer) {
    54→    const key = normalizeTitle(stripEntryTitle(e.title))
    55→    if (key && !map.has(key)) {
    56→      map.set(key, { number: e.number, printedPage: e.printedPage, matchPage: e.matchPage, matched: e.matched })
    57→    }
    58→  }
    59→  return map
    60→}

    62→export function collectTopicInputs(
    63→  parts: (SectionAnalysis | null)[],
    64→  units: AnalysisUnit[],
    65→  anchoredPages: Map<number, number | null>,
    66→  tocPages: number[] = [],
    67→  tocInfo: Map<string, TocInfo> = new Map(),
    68→): TopicInput[] {
    69→  const inputs: TopicInput[] = []
    70→  void anchoredPages
    71→  // Heading-anchored topics: each headed section contributes its own heading
    72→  // as a topic — that is the document's real organisation, not model opinion.
    73→  // The topic keeps the document's exact wording, numbering and (when the
    74→  // document has a contents table) the entry's own match record.
    75→  for (const u of units) {
    76→    if (u.kind === 'SECTION' && u.title) {
    77→      const info = tocInfo.get(normalizeTitle(u.title))
    78→      inputs.push({
    79→        name: u.title,
    80→        subtopics: [],
    81→        pages: pagesBetween(u.fromPage, u.toPage),
    82→        unitIndex: u.index,
    83→        unitTitle: u.title,
    84→        unitFromPage: u.fromPage,
    85→        unitToPage: u.toPage,
    86→        anchored: true,
    87→        origin: 'SOURCE_HEADING',
    88→        titleSource: info ? 'toc' : 'body',
    89→        numbering: extractNumberToken(u.title),
    90→        tocPrintedPage: info?.printedPage ?? null,
    91→        tocMatchPage: info?.matchPage ?? null,
    92→        structureVerified: info ? info.matched : true,
    93→      })
    94→    }
    95→  }
    96→  const tocPageSet = new Set(tocPages)
    97→  for (const u of units) {
    98→    const part = parts[u.index]
    99→    if (!part) continue
   100→    // Topics the model claims ONLY on the document's own contents pages are
   101→    // INDEX ECHOES ("Eligibility Criteria" from the contents line) — a map of
   102→    // the document, not its content. Page-level check: real front-matter
   103→    // content around the contents table (vision/mission pages) is still kept.
   104→    const isFront = u.kind === 'FRONT'
   105→    for (const t of part.topics ?? []) {
   106→      const name = String(t?.name ?? '').trim()
   107→      if (!name) continue
   108→      const claimed = (t.pages ?? []).filter(Number.isFinite)
   109→      if (isFront && claimed.length > 0 && claimed.every(p => tocPageSet.has(p))) continue
   110→      inputs.push({
   111→        name,
   112→        subtopics: (t.subtopics ?? []).map(s => String(s)).filter(Boolean),
   113→        pages: claimed,
   114→        unitIndex: u.index,
   115→        unitTitle: u.title,
   116→        unitFromPage: u.fromPage,
   117→        unitToPage: u.toPage,
   118→        origin: 'AI_INFERRED',
   119→        titleSource: 'inferred',
   120→      })
   121→    }
   122→  }
   123→  return inputs
   124→}

   126→function pagesBetween(from: number, to: number): number[] {
   127→  const out: number[] = []
   128→  for (let p = from; p <= to; p++) out.push(p)
   129→  return out
   130→}

   132→function mergeFacts(parts: (SectionAnalysis | null)[], cap: number, pick: (p: SectionAnalysis) => string[]): string[] {
   133→  const seen = new Set<string>()
   134→  const out: string[] = []
   135→  for (const part of parts) {
   136→    if (!part) continue
   137→    for (const raw of pick(part)) {
   138→      const f = String(raw ?? '').trim()
   139→      if (!f) continue
   140→      const key = normalizeTitle(f).slice(0, 100)
   141→      if (key && seen.has(key)) continue
   142→      seen.add(key)
   143→      out.push(f.slice(0, 400))
   144→      if (out.length >= cap) return out
   145→    }
   146→  }
   147→  return out
   148→}
   149→
   150→function mergeDefinitions(parts: (SectionAnalysis | null)[], cap: number): DocumentAnalysis['definitions'] {
   151→  const seen = new Set<string>()
   152→  const out: DocumentAnalysis['definitions'] = []
   153→  for (const part of parts) {
   154→    if (!part) continue
   155→    for (const d of part.definitions ?? []) {
   156→      if (!d?.term || !d?.meaning) continue
   157→      const key = normalizeTitle(String(d.term))
   158→      if (!key || seen.has(key)) continue
   159→      seen.add(key)
   160→      out.push({
   161→        term: String(d.term).slice(0, 120),
   162→        meaning: String(d.meaning).slice(0, 500),
   163→        page: Number(d.page) || 0,
   164→      })
   165→      if (out.length >= cap) return out
   166→    }
   167→  }
   168→  return out
   169→}
   170→
   171→function mergeMisconceptions(parts: (SectionAnalysis | null)[], cap: number): DocumentAnalysis['misconceptions'] {
   172→  const seen = new Set<string>()
   173→  const out: DocumentAnalysis['misconceptions'] = []
   174→  for (const part of parts) {
   175→    if (!part) continue
   176→    for (const m of part.misconceptions ?? []) {
   177→      if (!m?.misconception || !m?.correction) continue
   178→      const key = normalizeTitle(String(m.misconception)).slice(0, 100)
   179→      if (!key || seen.has(key)) continue
   180→      seen.add(key)
   181→      out.push({
   182→        misconception: String(m.misconception).slice(0, 300),
   183→        correction: String(m.correction).slice(0, 400),
   184→        page: Number.isFinite(m.page) ? Number(m.page) : undefined,
   185→      })
   186→      if (out.length >= cap) return out
   187→    }
   188→  }
   189→  return out
   190→}
   191→
   192→function mergeUnreadable(parts: (SectionAnalysis | null)[], cap: number): string[] {
   193→  const out: string[] = []
   194→  for (const part of parts) {
   195→    if (!part) continue
   196→    for (const u of part.unreadable ?? []) {
   197→      const s = String(u ?? '').trim()
   198→      if (!s) continue
   199→      out.push(s.slice(0, 200))
   200→      if (out.length >= cap) return out
   201→    }
   202→  }
   203→  return out
   204→}

   206→function buildCoverage(
   207→  extraction: { numPages: number; scannedPages: number },
   208→  readablePages: ExtractedPage[],
   209→  units: AnalysisUnit[],
   210→  outcomes: SectionOutcome[],
   211→): CoverageMetrics {
   212→  const unitPages = (u: AnalysisUnit) => pagesBetween(u.fromPage, u.toPage)
   213→  const okUnits = units.filter((u, i) => outcomes[i]?.status === 'OK')
   214→  const failedUnits = units.filter((u, i) => outcomes[i]?.status === 'FAILED')
   215→  const skippedUnits = units.filter((u, i) => outcomes[i]?.status === 'SKIPPED')
   216→  const accountedPageSet = new Set<number>()
   217→  for (const u of okUnits) for (const p of unitPages(u)) accountedPageSet.add(p)
   218→  const skippedPageSet = new Set<number>()
   219→  for (const u of skippedUnits) for (const p of unitPages(u)) skippedPageSet.add(p)
   220→  const readableSet = new Set(readablePages.map(p => p.page))
   221→  const unanalysed = [...readableSet].filter(p => !accountedPageSet.has(p) && !skippedPageSet.has(p)).length
   222→  return {
   223→    totalPages: extraction.numPages,
   224→    readablePages: readablePages.length,
   225→    scannedPages: extraction.scannedPages,
   226→    analysedPages: accountedPageSet.size,
   227→    unanalysedPages: unanalysed,
   228→    skippedPages: [...readableSet].filter(p => skippedPageSet.has(p)).length,
   229→    unitsTotal: units.length,
   230→    unitsOk: okUnits.length,
   231→    unitsFailed: failedUnits.length,
   232→    unitsSkipped: skippedUnits.length,
   233→    complete: unanalysed === 0 && failedUnits.length === 0 && units.length > 0,
   234→  }
   235→}
   236→
   237→function buildCoverageNote(cov: CoverageMetrics): string {
   238→  const bits: string[] = []
   239→  if (cov.unitsFailed > 0) {
   240→    bits.push(`Analysed ${cov.analysedPages} of ${cov.readablePages} readable page${cov.readablePages === 1 ? '' : 's'} — ${cov.unitsFailed} section${cov.unitsFailed === 1 ? '' : 's'} failed and can be retried without losing the sections that succeeded.`)
   241→  } else if (cov.unitsSkipped > 0) {
   242→    bits.push(`All readable pages accounted for: ${cov.analysedPages} analysed across ${cov.unitsOk} section${cov.unitsOk === 1 ? '' : 's'}, ${cov.skippedPages} page${cov.skippedPages === 1 ? '' : 's'} in ${cov.unitsSkipped} section${cov.unitsSkipped === 1 ? '' : 's'} skipped with a recorded reason (see below).`)
   243→  } else {
   244→    bits.push(`All ${cov.readablePages} readable page${cov.readablePages === 1 ? '' : 's'} analysed across ${cov.unitsTotal} section${cov.unitsTotal === 1 ? '' : 's'}.`)
   245→  }
   246→  if (cov.scannedPages > 0) {
   247→    bits.push(`${cov.scannedPages} image-only page${cov.scannedPages === 1 ? '' : 's'} would need OCR, which is not available yet — they are excluded from the analysis.`)
   248→  }
   249→  return bits.join(' ')
   250→}

   254→/**
   255→ * Background job: extract text, detect the document's real structure, analyse
   256→ * every section, merge deterministically.
   257→ *  opts.force      — full re-analysis of a READY document (fresh extraction)
   258→ *  partial retry   — a READY v2 analysis with failed sections re-runs ONLY the
   259→ *                    failed sections and keeps every successful result
   260→ */
   261→export async function processDocument(documentId: string, opts?: { force?: boolean }): Promise<void> {
   262→  const force = opts?.force === true
   263→  const doc = await db.studyDocument.findUnique({ where: { id: documentId } })
   264→  if (!doc) return
   265→
   266→  const stored = safeParse<StoredAnalysis | null>(doc.analysisJson, null)
   267→  const canPartialRetry =
   268→    !force &&
   269→    doc.status === 'READY' &&
   270→    stored?.pipelineVersion === PIPELINE_VERSION &&
   271→    Array.isArray(stored?._parts) &&
   272→    Array.isArray(stored?.sectionOutcomes) &&
   273→    stored?.coverage?.complete === false
   274→
   275→  const allowed = force ? ['QUEUED', 'FAILED', 'NEEDS_OCR', 'READY'] : ['QUEUED', 'FAILED', 'NEEDS_OCR']
   276→  if (!allowed.includes(doc.status) && !canPartialRetry) return
   277→
   278→  // Identical-file reuse — only from the SAME pipeline version, so an upgrade
   279→  // always re-analyses instead of copying a stale analysis.
   280→  let twinAnalysisJson: string | null = null
   281→  if (!force && (doc.status === 'FAILED' || doc.status === 'QUEUED')) {
   282→    const twin = await db.studyDocument.findFirst({
   283→      where: { contentHash: doc.contentHash, status: 'READY', id: { not: doc.id } },
   284→      select: { analysisJson: true },
   285→    })
   286→    const twinJson = twin?.analysisJson ?? null
   287→    const twinParsed = safeParse<StoredAnalysis | null>(twinJson, null)
   288→    if (twinParsed?.pipelineVersion === PIPELINE_VERSION && twinParsed.topics) twinAnalysisJson = twinJson
   289→  }
   290→
   291→  await db.studyDocument.update({
   292→    where: { id: documentId },
   293→    data: {
   294→      status: 'EXTRACTING',
   295→      statusDetail: canPartialRetry ? 'Preparing to re-analyse the failed sections…' : 'Reading the uploaded file…',
   296→      updatedAt: new Date(),
   297→    },
   298→  })
   299→
   300→  let buffer: Buffer
   301→  try {
   302→    buffer = await readFile(doc.storedPath)
   303→  } catch {
   304→    await db.studyDocument.update({
   305→      where: { id: documentId },
   306→      data: { status: 'FAILED', statusDetail: 'The stored file could not be read. Please upload it again.' },
   307→    })
   308→    return
   309→  }
   310→
   311→  // Verify the stored content still matches its hash (integrity)
   312→  const hash = createHash('sha256').update(buffer).digest('hex')
   313→  if (hash !== doc.contentHash) {
   314→    await db.studyDocument.update({
   315→      where: { id: documentId },
   316→      data: { status: 'FAILED', statusDetail: 'Stored file failed its integrity check. Please upload it again.' },
   317→    })
   318→    return
   319→  }
   320→
   321→  // ── Stage 1: extraction (re-extract for force / first runs; reuse for partial retry)
   322→  let extraction: Awaited<ReturnType<typeof extractPdfPages>>
   323→  let storedPages: ExtractedPage[] | null = null
   324→  if (canPartialRetry && doc.extractionJson) {
   325→    const parsed = safeParse<ExtractedPage[] | null>(doc.extractionJson, null)
   326→    if (parsed && parsed.length > 0) storedPages = parsed
   327→  }
   328→  if (storedPages) {
   329→    const readable = storedPages.filter(p => !p.scanned && p.text.length > 0)
   330→    extraction = {
   331→      pages: storedPages,
   332→      numPages: doc.pageCount ?? storedPages.length,
   333→      textChars: readable.reduce((s, p) => s + p.text.length, 0),
   334→      scannedPages: storedPages.filter(p => p.scanned).length,
   335→      titleGuess: null,
   336→    }
   337→  } else {
   338→    try {
   339→      extraction = await extractPdfPages(new Uint8Array(buffer))
   340→    } catch (err) {
   341→      console.error('[riishe extract]', err)
   342→      await db.studyDocument.update({
   343→        where: { id: documentId },
   344→        data: {
   345→          status: 'FAILED',
   346→          statusDetail: `This file could not be parsed as a PDF (${err instanceof Error ? err.name : 'unknown error'}). It may be corrupted or not a real PDF.`,
   347→        },
   348→      })
   349→      return
   350→    }
   351→  }
   352→
   353→  const readablePages = extraction.pages.filter(p => !p.scanned && p.text.length > 0)
   354→  await db.studyDocument.update({
   355→    where: { id: documentId },
   356→    data: {
   357→      pageCount: extraction.numPages,
   358→      extractionJson: JSON.stringify(extraction.pages),
   359→      status: readablePages.length === 0 ? 'NEEDS_OCR' : 'ANALYSING',
   360→      statusDetail: readablePages.length === 0
   361→        ? 'No extractable text found — every page looks like a scanned image. OCR is not available in BEAN EXAMS yet, so Riishe cannot read this document. Questions cannot be generated from it.'
   362→        : 'Riishe is mapping the document structure…',
   363→    },
   364→  })
   365→
   366→  if (readablePages.length === 0) return
   367→
   368→  // ── Identical-file reuse (same pipeline version only)
   369→  if (!canPartialRetry && twinAnalysisJson) {
   370→    const analysis = safeParse<StoredAnalysis | null>(twinAnalysisJson, null)
   371→    if (analysis) {
   372→      analysis.title = analysis.title || doc.title
   373→      await db.studyDocument.update({
   374→        where: { id: documentId },
   375→        data: {
   376→          status: 'READY',
   377→          statusDetail: `Ready — analysis reused from an identical upload (${extraction.numPages} pages).`,
   378→          analysisJson: JSON.stringify(analysis),
   379→        },
   380→      })
   381→      return
   382→    }
   383→  }
   384→
   385→  // ── Stage 2: deterministic structure detection
   386→  const structure = detectStructure(extraction.pages, extraction.numPages, { maxUnitChars: UNIT_CHARS })
   387→  const units = structure.units
   388→
   389→  if (units.length === 0) {
   390→    await db.studyDocument.update({
   391→      where: { id: documentId },
   392→      data: {
   393→        status: 'NEEDS_OCR',
   394→        statusDetail: 'The PDF has pages but no readable text that Riishe can analyse. OCR is not available yet.',
   395→      },
   396→    })
   397→    return
   398→  }
   399→
   400→  // ── Partial retry: reuse stored successful parts, re-run only failures.
   401→  // Previously SKIPPED units are terminal (their refusal reason is recorded);
   402→  // carrying the decision over keeps retries convergent instead of re-asking
   403→  // the provider to refuse the same text again.
   404→  const skipReasons = new Map<number, string>()
   405→  let storedParts: (SectionAnalysis | null)[] | null = null
   406→  if (canPartialRetry && stored) {
   407→    const prev = stored._parts ?? []
   408→    const prevOutcomes = stored.sectionOutcomes ?? []
   409→    if (prev.length === units.length && prevOutcomes.length === units.length) {
   410→      storedParts = prev
   411→      for (let i = 0; i < prevOutcomes.length; i++) {
   412→        if (prevOutcomes[i]?.status === 'SKIPPED' && prevOutcomes[i]?.error) {
   413→          skipReasons.set(i, prevOutcomes[i].error as string)
   414→        }
   415→      }
   416→    }
   417→  }
   418→
   419→  // ── Stage 3: analyse every unit (failures recorded, never fatal)
   420→  const parts: (SectionAnalysis | null)[] = new Array(units.length).fill(null)
   421→  if (storedParts) for (let i = 0; i < units.length; i++) parts[i] = storedParts[i]
   422→
   423→  let done = 0
   424→  let failedCount = 0
   425→  let skippedCount = 0
   426→  for (const unit of units) {
   427→    if (parts[unit.index]) continue // preserved from the previous run
   428→    if (skipReasons.has(unit.index)) continue // terminal skip carried over
   429→    const res = await analyseSection(doc.title, unit, units.length)
   430→    if (res.part) {
   431→      parts[unit.index] = res.part
   432→      done++
   433→    } else if (res.skippedReason) {
   434→      skippedCount++
   435→      skipReasons.set(unit.index, res.skippedReason)
   436→    } else {
   437→      failedCount++
   438→    }
   439→    const settled = done + failedCount + skippedCount
   440→    if (settled % 3 === 0 || settled === units.length) {
   441→      const extras = [
   442→        failedCount > 0 ? `${failedCount} failed` : null,
   443→        skippedCount > 0 ? `${skippedCount} skipped` : null,
   444→      ].filter(Boolean).join(', ')
   445→      await db.studyDocument.update({
   446→        where: { id: documentId },
   447→        data: {
   448→          statusDetail: extras
   449→            ? `Riishe is analysing the content — section ${settled} of ${units.length} (${extras})…`
   450→            : `Riishe is analysing the content — section ${settled} of ${units.length}…`,
   451→        },
   452→      }).catch(() => {})
   453→    }
   454→  }
   455→
   456→  const outcomes: SectionOutcome[] = units.map((u, i) => {
   457→    const range = `${u.fromPage}${u.toPage !== u.fromPage ? `–${u.toPage}` : ''}`
   458→    const label = u.title ? `${u.title} (${range})` : `pages ${range}`
   459→    if (parts[i]) return { index: i, title: u.title, label, fromPage: u.fromPage, toPage: u.toPage, status: 'OK' as const }
   460→    if (skipReasons.has(i)) return { index: i, title: u.title, label, fromPage: u.fromPage, toPage: u.toPage, status: 'SKIPPED' as const, error: skipReasons.get(i) }
   461→    return { index: i, title: u.title, label, fromPage: u.fromPage, toPage: u.toPage, status: 'FAILED' as const, error: 'AI analysis failed for this section' }
   462→  })
   463→
   464→  if (parts.every(p => p === null)) {
   465→    // Nothing succeeded — honest, retryable failure (extraction is preserved).
   466→    // A skip-only outcome (every unit skipped with reason) is still honest
   467→    // READY-partial, not FAILED: the extraction worked, the provider refused.
   468→    const allSkipped = skipReasons.size === units.length
   469→    if (!allSkipped) {
   470→      await db.studyDocument.update({
   471→        where: { id: documentId },
   472→        data: {
   473→          status: 'FAILED',
   474→          statusDetail: `Extraction succeeded (${readablePages.length} readable pages) but AI analysis failed for all ${units.length} section${units.length === 1 ? '' : 's'}: ${AI_UNAVAILABLE_MESSAGE}`,
   475→          extractionJson: JSON.stringify(extraction.pages),
   476→        },
   477→      })
   478→      return
   479→    }
   480→  }
   481→
   482→  // ── Stage 4: deterministic merge + verification + coverage
   483→  const pageTextIndex = new Map<number, string>()
   484→  for (const p of extraction.pages) {
   485→    if (!p.scanned && p.text) pageTextIndex.set(p.page, p.text.toLowerCase().replace(/[^a-z0-9]+/g, ' '))
   486→  }
   487→
   488→  const topicInputs = collectTopicInputs(parts, units, new Map(), structure.tocPages, buildTocInfoMap(structure.tocLayer))
   489→  const topics: AnalyzedTopic[] = canonicalizeTopics(topicInputs, { pageTextIndex, maxTopics: 200, maxSubtopicsPerTopic: 12 })
   490→
   491→  // Printed page labels: per-page labels actually READ from the page edges
   492→  // take precedence (evidence, never derived); the detected offset is the
   493→  // fallback for pages whose label was not printed. Nothing is invented.
   494→  {
   495→    const edgeLabels = structure.printed.labels ?? {}
   496→    const off = structure.printed.offset
   497→    for (const t of topics) {
   498→      if (t.pages.length === 0) continue
   499→      const direct = t.pages.map(p => edgeLabels[p]).filter((s): s is string => !!s)
   500→      if (direct.length === t.pages.length) {
   501→        t.printedPages = direct
   502→        continue
   503→      }
   504→      if (off !== null) {
   505→        const labels = t.pages.map(p => p - off).filter(n => n >= 1).map(String)
   506→        if (labels.length === t.pages.length) t.printedPages = labels
   507→      }
   508→    }
   509→  }
   510→
   511→  const cov = buildCoverage(
   512→    { numPages: extraction.numPages, scannedPages: extraction.scannedPages },
   513→    readablePages,
   514→    units,
   515→    outcomes,
   516→  )
   517→
   518→  const merged: StoredAnalysis = {
   519→    title: doc.title,
   520→    summary: '',
   521→    topics,
   522→    definitions: mergeDefinitions(parts, 60),
   523→    keyFacts: mergeFacts(parts, 80, p => p.keyFacts ?? []),
   524→    formulas: mergeFacts(parts, 50, p => p.formulas ?? []),
   525→    misconceptions: mergeMisconceptions(parts, 40),
   526→    keyPages: [...new Set(topics.flatMap(t => t.pages.slice(0, 3)))].filter(p => p >= 1).sort((a, b) => a - b).slice(0, 60),
   527→    unreadableSections: mergeUnreadable(parts, 16),
   528→    studyGaps: [],
   529→    coverageNote: buildCoverageNote(cov),
   530→    pipelineVersion: PIPELINE_VERSION,
   531→    coverage: cov,
   532→    sectionOutcomes: outcomes,
   533→    structure: {
   534→      kind: structure.kind,
   535→      tocValidated: structure.tocValidated,
   536→      tocEntries: structure.toc.length,
   537→      headingCount: structure.headings.length,
   538→      printedPageOffset: structure.printed.offset,
   539→      printedRomanFrontMatter: structure.printed.romanFrontMatter,
   540→      // the ORIGINAL-STRUCTURE layer: the document's own contents entries with
   541→      // body-match status, its body headings, and the printed page labels that
   542→      // were actually read — recorded, never silently dropped
   543→      toc: structure.tocLayer.slice(0, 200),
   544→      headings: structure.headings.slice(0, 120).map(h => ({ title: h.title, page: h.page, level: h.level, fromToc: h.fromToc })),
   545→      tocMatched: structure.tocLayer.filter(e => e.matched).length,
   546→      tocUnmatched: structure.tocLayer.filter(e => !e.matched).length,
   547→      printedLabels: structure.printed.labels,
   548→      tocPages: structure.tocPages,
   549→    },
   550→    _parts: parts, // retained so a partial retry keeps every successful section
   551→  }
   552→  merged.summary = await summarise(doc.title, extraction.pages, merged)
   553→  merged.studyGaps = deriveStudyGaps(merged, cov)
   554→
   555→  const partial = !cov.complete
   556→  await db.studyDocument.update({
   557→    where: { id: documentId },
   558→    data: {
   559→      status: 'READY',
   560→      statusDetail: partial
   561→        ? `Ready (partial) — analysed ${cov.unitsOk}/${cov.unitsTotal} sections covering ${cov.analysedPages} of ${cov.readablePages} readable pages${cov.unitsSkipped > 0 ? `; ${cov.unitsSkipped} section${cov.unitsSkipped === 1 ? '' : 's'} skipped with reason` : ''}. ${cov.unitsFailed > 0 ? 'Retry analysis to cover the rest.' : ''}`.trim()
   562→        : `Ready — ${merged.topics.length} topic${merged.topics.length === 1 ? '' : 's'} across ${cov.unitsTotal} section${cov.unitsTotal === 1 ? '' : 's'}; all ${cov.readablePages} readable page${cov.readablePages === 1 ? '' : 's'} accounted for${cov.unitsSkipped > 0 ? ` (${cov.unitsSkipped} skipped with reason)` : ''}.`,
   563→      analysisJson: JSON.stringify(merged),
   564→    },
   565→  })
   566→}

   570→type SectionResult =
   571→  | { part: SectionAnalysis; skippedReason?: undefined }
   572→  | { part: null; skippedReason: string }
   573→  | { part: null; skippedReason?: undefined }
   574→
   575→const RATE_LIMIT_BACKOFF_MS = 15_000
   576→
   577→/**
   578→ * Analyse one unit with provider-failure strategies that keep the pipeline
   579→ * honest AND convergent:
   580→ *   • 429 rate limiting → ONE bounded 15 s backoff, then give up (the section
   581→ *     stays FAILED and the existing partial-retry path covers it);
   582→ *   • provider content filter (some legitimate dense name/term lists trip it
   583→ *     and an identical retry can NEVER succeed) → split the unit into smaller
   584→ *     page-range sub-chunks and analyse each independently, merging what the
   585→ *     provider accepts; if every sub-chunk is refused, the unit is SKIPPED
   586→ *     with a recorded reason instead of looping as FAILED forever.
   587→ */
   588→async function analyseSection(docTitle: string, unit: AnalysisUnit, totalUnits: number): Promise<SectionResult> {
   589→  try {
   590→    return { part: await analyseUnit(docTitle, unit, totalUnits) }
   591→  } catch (err) {
   592→    if (isRateLimitError(err)) {
   593→      console.warn(`[riishe analyse unit ${unit.index + 1}/${totalUnits}] rate limited — backing off ${RATE_LIMIT_BACKOFF_MS / 1000}s once`)
   594→      await new Promise(r => setTimeout(r, RATE_LIMIT_BACKOFF_MS))
   595→      try {
   596→        return { part: await analyseUnit(docTitle, unit, totalUnits) }
   597→      } catch (err2) {
   598→        if (!isContentFilterError(err2)) {
   599→          console.error(`[riishe analyse unit ${unit.index + 1}/${totalUnits} ${unit.title ?? `pages ${unit.fromPage}-${unit.toPage}`}]`, err2)
   600→          return { part: null }
   601→        }
   602→        err = err2
   603→      }
   604→    }
   605→    if (!isContentFilterError(err)) {
   606→      console.error(`[riishe analyse unit ${unit.index + 1}/${totalUnits} ${unit.title ?? `pages ${unit.fromPage}-${unit.toPage}`}]`, err)
   607→      return { part: null }
   608→    }
   609→  }
   610→
   611→  // Content-filter refusal → try smaller page-range sub-chunks of this unit.
   612→  const subChunks = splitUnitPages(unit)
   613→  if (subChunks.length <= 1) {
   614→    return {
   615→      part: null,
   616→      skippedReason: 'The AI provider\'s safety filter refused to process this section\'s text (dense lists of names or categories are a known trigger). Its pages are accounted as skipped, not analysed — no content was invented for them.',
   617→    }
   618→  }
   619→  console.warn(`[riishe analyse unit ${unit.index + 1}/${totalUnits}] content-filtered — retrying as ${subChunks.length} sub-chunks`)
   620→  const mergedParts: SectionAnalysis[] = []
   621→  for (const [i, chunk] of subChunks.entries()) {
   622→    try {
   623→      mergedParts.push(await analyseUnit(docTitle, chunk, totalUnits))
   624→    } catch (chunkErr) {
   625→      if (!isContentFilterError(chunkErr)) {
   626→        console.error(`[riishe analyse unit ${unit.index + 1}/${totalUnits} sub-chunk ${i + 1}/${subChunks.length}]`, chunkErr)
   627→      }
   628→      // filter-refused sub-chunks are dropped; remaining chunks still count
   629→    }
   630→  }
   631→  if (mergedParts.length === 0) {
   632→    return {
   633→      part: null,
   634→      skippedReason: `The AI provider's safety filter refused this section's text in full and in ${subChunks.length} smaller parts (dense lists of names or categories are a known trigger). Its pages are accounted as skipped, not analysed — no content was invented for them.`,
   635→    }
   636→  }
   637→  return { part: mergeSectionAnalyses(mergedParts) }
   638→}
   639→
   640→/** Split a unit into 2–4 page-range sub-chunks (halves, quarters for 8+ pages). */
   641→export function splitUnitPages(unit: AnalysisUnit): AnalysisUnit[] {
   642→  const pageCount = unit.toPage - unit.fromPage + 1
   643→  if (pageCount < 2) return [unit]
   644→  const parts = pageCount >= 8 ? 4 : 2
   645→  const size = Math.ceil(pageCount / parts)
   646→  const out: AnalysisUnit[] = []
   647→  for (let from = unit.fromPage; from <= unit.toPage; from += size) {
   648→    const to = Math.min(from + size - 1, unit.toPage)
   649→    const subText = unit.text
   650→      .split(/\n\n+/)
   651→      .filter(block => {
   652→        const m = /^\[page (\d+)\]/.exec(block)
   653→        return m ? Number(m[1]) >= from && Number(m[1]) <= to : true
   654→      })
   655→      .join('\n\n')
   656→    out.push({ ...unit, fromPage: from, toPage: to, text: subText, chars: subText.length })
   657→  }
   658→  return out.length > 0 ? out : [unit]
   659→}

   661→export function mergeSectionAnalyses(parts: SectionAnalysis[]): SectionAnalysis {
   662→  return {
   663→    topics: parts.flatMap(p => p.topics ?? []).slice(0, 12),
   664→    definitions: parts.flatMap(p => p.definitions ?? []).slice(0, 20),
   665→    keyFacts: parts.flatMap(p => p.keyFacts ?? []).slice(0, 20),

   667→    misconceptions: parts.flatMap(p => p.misconceptions ?? []).slice(0, 10),

   672→function unitDescriptor(unit: AnalysisUnit): string {
   673→  const range = unit.fromPage === unit.toPage ? `page ${unit.fromPage}` : `pages ${unit.fromPage}–${unit.toPage}`
   674→  if (unit.kind === 'SECTION' && unit.title) return `the section "${unit.title}" (${range} of the uploaded PDF)`
   675→  if (unit.kind === 'FRONT') return `the front matter (${range} of the uploaded PDF)`
   676→  if (unit.kind === 'OVERFLOW') return `an excerpt of ${range} of the uploaded PDF`
   677→  return `${range} of the uploaded PDF`
   678→}

   680→async function analyseUnit(docTitle: string, unit: AnalysisUnit, totalUnits: number): Promise<SectionAnalysis> {
   681→  return chatJson<SectionAnalysis>({
   682→    system: SYSTEM,
   683→    user: `Analyse ${unitDescriptor(unit)} — part ${unit.index + 1} of ${totalUnits} covering the whole document ("${docTitle}"). Extract ONLY what is present in the text below.
   684→
   685→Return JSON with this exact shape:
   686→{
   687→  "topics": [{"name": string, "subtopics": string[], "pages": number[]}],
   688→  "definitions": [{"term": string, "meaning": string, "page": number}],
   689→  "keyFacts": string[],
   690→  "formulas": string[],
   691→  "misconceptions": [{"misconception": string, "correction": string, "page": number}],
   692→  "unreadable": string[]
   693→}
   694→
   695→Rules:
   696→- "topics": the concrete topics/concepts this text actually covers, using the document's own headings and terminology (3-8 at most; fewer is fine). Topic names must be SHORT noun phrases (≤ 8 words) — never full sentences. Do NOT invent topics, do NOT echo the section title unless the text genuinely develops it, and do NOT promote every heading you see.
   697→- REFERENCE LISTS (abbreviations, glossaries, indexes, rosters, fee/centre/name tables): return exactly ONE topic named after the section's own heading or the list's own title, with the ACTUAL entries from the list as its subtopics (up to 12, verbatim, most study-relevant). NEVER invent thematic groupings the document did not write (e.g. "Government Bodies", "Educational Abbreviations" for an abbreviations list).
   698→- "subtopics": concepts or key terms under each topic, found in this text — they are verified against the pages and invented or garbled ones are dropped.
   699→- "pages" must cite page numbers from the [page N] markers in the text — never other numbers.
   700→- If this text is from a question paper, answer key or marks scheme, extract the subject areas the questions test (not the questions themselves) and note it in keyFacts.
   701→- "unreadable": list page labels whose text is garbled, truncated or incomprehensible ([] if none).
   702→- "misconceptions": only if the text supports a correction (common wrong belief vs what the text says).
   703→- "formulas": mathematical/scientific formulas or rules stated in the text ([] if none).
   704→
   705→${fenceDocumentText(docTitle, unit.text)}`,
   706→  })
   707→}

   709→async function summarise(title: string, pages: ExtractedPage[], merged: StoredAnalysis): Promise<string> {
   710→  // Grounded summary from the first readable pages + identified topic names
   711→  const head = pages
   712→    .filter(p => !p.scanned && p.text)
   713→    .slice(0, 3)
   714→    .map(p => `[page ${p.page}]\n${p.text}`)
   715→    .join('\n\n')
   716→    .slice(0, 3000)
   717→  try {
   718→    const res = await chatJson<{ summary: string }>({
   719→      system: SYSTEM,
   720→      user: `Write a 2–3 sentence summary of this study document ("${title}") for a student, strictly from the text. Return JSON: {"summary": string}
   721→
   722→Known topics identified: ${merged.topics.slice(0, 12).map(t => t.name).join(', ') || '(none)'}.
   723→
   724→${fenceDocumentText(title, head)}`,
   725→    })
   726→    return String(res.summary ?? '').slice(0, 800)
   727→  } catch {
   728→    // Summary is cosmetic — fall back to a factual non-AI line rather than failing the job
   729→    return `Study document with ${merged.topics.length} identified topic${merged.topics.length === 1 ? '' : 's'}.`
   730→  }
   731→}

   733→function deriveStudyGaps(a: DocumentAnalysis, cov: CoverageMetrics): string[] {
   734→  const gaps: string[] = []
   735→  if (a.topics.length === 0) gaps.push('No clear topics were identified — the text may be too sparse for question generation.')
   736→  if (a.definitions.length === 0) gaps.push('No key definitions were found — you may need supporting material for terminology-heavy questions.')
   737→  if (a.formulas.length === 0) gaps.push('No formulas or rules were detected — calculation-based questions will not be generated.')
   738→  if (a.unreadableSections.length > 0) gaps.push(`${a.unreadableSections.length} section${a.unreadableSections.length === 1 ? '' : 's'} were unreadable and excluded from the analysis.`)
   739→  if (cov.scannedPages > 0) gaps.push(`${cov.scannedPages} image-only page${cov.scannedPages === 1 ? ' needs' : 's need'} OCR to be analysed — OCR is not available yet.`)
   740→  if (cov.unitsFailed > 0) gaps.push(`${cov.unanalysedPages} readable page${cov.unanalysedPages === 1 ? ' is' : 's are'} not yet covered (failed sections) — retry the analysis to include them.`)
   741→  if (cov.unitsSkipped > 0) gaps.push(`${cov.unitsSkipped} section${cov.unitsSkipped === 1 ? '' : 's'} (${cov.skippedPages} page${cov.skippedPages === 1 ? '' : 's'}) were skipped because the AI provider's safety filter refused their text — see the section breakdown for the exact reason.`)
   742→  return gaps
   743→}
