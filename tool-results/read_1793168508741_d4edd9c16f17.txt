     3→//   1. Retrieve page-aligned sections relevant to the focus topics.
     4→//   2. Generate per-category batches (configurable distribution) with strict
     5→//      grounding rules — every question must cite pages + a verbatim excerpt.
     6→//   3. Programmatic validation: structure, duplicates, evidence-in-document
     7→//      check, page-range check.
     8→//   4. LLM validation pass: is the keyed answer actually correct and supported?
     9→//   5. Top-up rounds until the requested count is validated or the batch
    10→//      budget is spent.
    11→//   6. Publish only validated questions. Short/poor sources get a smaller
    12→//      validated test with an honest note — never 50 unchecked outputs.

    45→// ─── Normalisation / duplicate detection ──────────────────────────────────────
    46→
    47→function normalize(s: string): string {
    48→  return s.toLowerCase().replace(/[^a-z0-9]+/g, ' ').trim()
    49→}
    50→
    51→function tokenSet(s: string): Set<string> {
    52→  return new Set(normalize(s).split(' ').filter(w => w.length > 2))
    53→}
    54→
    55→function jaccard(a: Set<string>, b: Set<string>): number {
    56→  if (a.size === 0 || b.size === 0) return 0
    57→  let inter = 0
    58→  for (const w of a) if (b.has(w)) inter++
    59→  return inter / (a.size + b.size - inter)
    60→}
    61→
    62→/** Loose containment check for evidence: whitespace/punctuation insensitive. */
    63→function evidenceInText(evidence: string, text: string): boolean {
    64→  const norm = (x: string) => x.toLowerCase().replace(/[^a-z0-9]+/g, ' ').trim()
    65→  const ev = norm(evidence)
    66→  if (ev.length < 12) return false
    67→  const hay = norm(text)
    68→  if (hay.includes(ev)) return true
    69→  // Sliding window: evidence may reflow across line breaks
    70→  const words = ev.split(' ')
    71→  if (words.length >= 4) {
    72→    const half = Math.floor(words.length / 2)
    73→    return hay.includes(words.slice(0, half).join(' ')) || hay.includes(words.slice(half).join(' '))
    74→  }
    75→  return false
    76→}

    93→/**
    94→ * Build the generation section pool from the WHOLE document using the same
    95→ * deterministic structure detection the analysis pipeline uses (so retrieval
    96→ * can never be silently limited to the first few pages). When the student
    97→ * picked focus topics, the pool leads with the ACTUAL pages those topics sit
    98→ * on (from the verified analysis page references), padded with balanced
    99→ * coverage so batches always have enough material.
   100→ */
   101→export function buildGenerationSections(
   102→  pages: ExtractedPage[],
   103→  numPages: number,
   104→  analysis: DocumentAnalysis | null,
   105→  focus: string[],
   106→): BaseGenSection[] {
   107→  const readable = pages.filter(p => !p.scanned && p.text.length > 0)
   108→  const pageByNumber = new Map(readable.map(p => [p.page, p] as const))
   109→  const textOf = (from: number, to: number) =>
   110→    readable
   111→      .filter(p => p.page >= from && p.page <= to)
   112→      .map(p => `[page ${p.page}]\n${p.text}`)
   113→      .join('\n\n')
   114→
   115→  const fromUnits = (): BaseGenSection[] => {
   116→    const structure = detectStructure(pages, numPages, { maxUnitChars: GEN_SECTION_CHARS })
   117→    if (structure.units.length === 0) return []
   118→    return structure.units.map(u => ({
   119→      label: u.title ? `${u.title} — pages ${u.fromPage}${u.toPage !== u.fromPage ? `–${u.toPage}` : ''}` : u.fromPage === u.toPage ? `page ${u.fromPage}` : `pages ${u.fromPage}–${u.toPage}`,
   120→      text: textOf(u.fromPage, u.toPage),
   121→      fromPage: u.fromPage,
   122→      toPage: u.toPage,
   123→    }))
   124→  }
   125→
   126→  const fallbackSections = (): BaseGenSection[] =>
   127→    sectionPages(pages, { maxChars: GEN_SECTION_CHARS, maxSections: 400 }).map(s => ({
   128→      label: s.label,
   129→      text: s.text,
   130→      fromPage: s.fromPage,
   131→      toPage: s.toPage,
   132→    }))
   133→
   134→  let sections = fromUnits()
   135→  if (sections.length === 0) sections = fallbackSections()
   136→  if (sections.length === 0) return []
   137→
   138→  if (focus.length > 0 && analysis?.topics?.length) {
   139→    const normFocus = new Set(focus.map(f => normalizeTitle(f)))
   140→    const matched = analysis.topics.filter(t =>
   141→      normFocus.has(normalizeTitle(t.name)) ||
   142→      (t.aliases ?? []).some(a => normFocus.has(normalizeTitle(a))),
   143→    )
   144→    // include children of matched sections so the focus covers the sub-topics too
   145→    const byParent = new Map(analysis.topics.map(t => [t.id ?? '', t] as const))
   146→    for (const t of analysis.topics) {
   147→      if (t.parentId && byParent.get(t.parentId) && matched.some(m => m.id === t.parentId)) matched.push(t)
   148→    }
   149→    const pageSet = new Set<number>()
   150→    for (const t of matched) for (const p of t.pages) if (pageByNumber.has(p)) pageSet.add(p)
   151→    if (pageSet.size > 0) {
   152→      // expand by one page for context, then group into contiguous runs
   153→      for (const p of [...pageSet]) {
   154→        if (pageByNumber.has(p - 1)) pageSet.add(p - 1)
   155→        if (pageByNumber.has(p + 1)) pageSet.add(p + 1)
   156→      }
   157→      const sorted = [...pageSet].sort((a, b) => a - b)
   158→      const runs: { from: number; to: number }[] = []
   159→      for (const p of sorted) {
   160→        const last = runs[runs.length - 1]
   161→        if (last && p <= last.to + 1) last.to = p
   162→        else runs.push({ from: p, to: p })
   163→      }
   164→  const focusSections: BaseGenSection[] = runs
   165→        .map(r => ({
   166→          label: r.from === r.to ? `page ${r.from}` : `pages ${r.from}–${r.to}`,
   167→          text: textOf(r.from, r.to).slice(0, GEN_SECTION_CHARS * 2),
   168→          fromPage: r.from,
   169→          toPage: r.to,
   170→        }))
   171→        .filter(s => s.text.length > 0)
   172→      const totalChars = focusSections.reduce((s, x) => s + x.text.length, 0)
   173→      if (focusSections.length > 0 && totalChars >= 1500) {
   174→        // top up with balanced coverage if the focus area is thin
   175→        const out = [...focusSections]
   176→        let chars = totalChars
   177→        for (const sec of sections) {
   178→          if (chars >= 6000) break
   179→          if (out.some(o => o.fromPage <= sec.toPage && sec.fromPage <= o.toPage)) continue
   180→          out.push(sec)
   181→          chars += sec.text.length
   182→        }
   183→        return out
   184→      }
   185→    }
   186→  }
   187→  return sections
   188→}
   189→
   190→function scoreSection(sectionText: string, focus: string[]): number {
   191→  if (focus.length === 0) return 0
   192→  const hay = normalize(sectionText)
   193→  let score = 0
   194→  for (const f of focus) {
   195→    const nf = normalize(f)
   196→    if (nf && hay.includes(nf)) score += 2
   197→    for (const w of nf.split(' ')) {
   198→      if (w.length > 4 && hay.includes(w)) score += 1
   199→    }
   200→  }
   201→  return score
   202→}
   203→
   204→function pickSections(
   205→  sections: GenSection[],
   206→  focus: string[],
   207→  cursor: { i: number },
   208→  count: number,
   209→): GenSection[] {
   210→  if (focus.length > 0) {
   211→    const ranked = [...sections].sort((a, b) => scoreSection(b.text, focus) - scoreSection(a.text, focus))
   212→    const top = ranked.slice(0, count)
   213→    if (scoreSection(top[top.length - 1]?.text ?? '', focus) > 0) return top
   214→  }
   215→  // Balanced round-robin over the document
   216→  const out: typeof sections = []
   217→  for (let k = 0; k < count && sections.length > 0; k++) {
   218→    out.push(sections[cursor.i % sections.length])
   219→    cursor.i++
   220→  }
   221→  return out
   222→}

   234→function generationPrompt(opts: {
   235→  category: QuestionCategory
   236→  count: number
   237→  sections: { label: string; text: string }[]
   238→  focus: string[]
   239→  avoid: string[]
   240→  examLabel: string | null
   241→  questionPaper?: boolean
   242→}): string {
   243→  const docText = opts.sections
   244→    .map(s => `--- SOURCE EXCERPT (${s.label}) ---\n${s.text}`)
   245→    .join('\n\n')
   246→  return `You are Riishe, the exam-question engine inside BEAN EXAMS. Generate EXACTLY ${opts.count} challenging multiple-choice questions of the category "${CATEGORY_LABELS[opts.category]}" for an Indian competitive-exam student${opts.examLabel ? ` preparing for ${opts.examLabel}` : ''}.
   247→
   248→${CATEGORY_RULES[opts.category]}
   249→
   250→STRICT GROUNDING RULES:
   251→- Every question, every option and the correct answer MUST be supported by the SOURCE EXCERPTS below. Do not use outside facts.
   252→- "evidence" = a verbatim quote (max 300 characters) copied EXACTLY from the excerpts that supports the correct answer.
   253→- "pageRefs" = the [page N] numbers whose excerpts support the answer.
   254→- If the excerpts don't contain enough for ${opts.count} good questions, generate fewer — quality over quantity.
   255→- 4 options each, exactly ONE correct. Distractors must be plausible (same topic, subtle differences) but clearly wrong to someone who knows the material.
   256→- "topic" = the specific subtopic this question tests.
   257→- Language: clear English, exam-style stems, no ambiguity, no trick wording that punishes careful readers.
   258→- Vary stems; do not reword the same question twice.
   259→${opts.questionPaper ? '\nNOTE: this source is a question paper / marks scheme. Use its questions to inform the STYLE and difficulty of your questions only — factual support still comes from the content the paper itself contains (its questions, options, instructions and data). Never treat a question\'s wording as evidence for a fact.' : ''}
   260→${opts.focus.length ? `\nThe student asked to focus on: ${opts.focus.join('; ')}. Skip content unrelated to these topics.` : ''}
   261→${opts.avoid.length ? `\nDo NOT duplicate or reword these existing question stems:\n${opts.avoid.map(a => `- ${a}`).join('\n')}` : ''}
   262→
   263→${fenceDocumentText('student study material', docText)}
   264→
   265→Return ONLY a JSON array; each element:
   266→{"text": string, "options": [string, string, string, string], "correctIndex": 0|1|2|3, "explanation": string, "difficulty": "EASY"|"MEDIUM"|"HARD", "topic": string, "pageRefs": number[], "evidence": string}
   267→
   268→"explanation" must teach: why the correct answer is right AND why the tempting wrong options fail (2-4 sentences).`
   269→}

   288→// ─── Programmatic validation ───────────────────────────────────────────────────
   289→
   290→function programmaticValidate(
   291→  raw: RawQuestion,
   292→  category: QuestionCategory,
   293→  sections: GenSection[],
   294→): CandidateQuestion | { reject: string } {
   295→  const text = String(raw.text ?? '').trim()
   296→  const options = (raw.options ?? []).map(o => String(o ?? '').trim())
   297→  const correctIndex = Number(raw.correctIndex)
   298→  const explanation = String(raw.explanation ?? '').trim()
   299→  const evidence = String(raw.evidence ?? '').trim()
   300→  const pageRefs = (raw.pageRefs ?? []).map(Number).filter(n => Number.isFinite(n))
   301→  const difficulty = ['EASY', 'MEDIUM', 'HARD'].includes(String(raw.difficulty)) ? String(raw.difficulty) : 'MEDIUM'
   302→  const topic = String(raw.topic ?? '').trim().slice(0, 120) || 'General'
   303→
   304→  if (text.length < 20 || text.length > 800) return { reject: 'stem too short/long' }
   305→  if (options.length !== 4 || options.some(o => o.length === 0 || o.length > 300)) return { reject: 'options invalid' }
   306→  const normed = options.map(o => normalize(o))
   307→  if (new Set(normed).size !== 4) return { reject: 'duplicate options' }
   308→  if (!Number.isInteger(correctIndex) || correctIndex < 0 || correctIndex > 3) return { reject: 'bad correctIndex' }
   309→  if (explanation.length < 40) return { reject: 'explanation too thin' }
   310→  if (!evidence) return { reject: 'no evidence' }
   311→  if (evidence.length > 600) return { reject: 'evidence too long' }
   312→
   313→  // Evidence must actually exist in the fed sections (grounding check)
   314→  const covered = sections.find(s => evidenceInText(evidence, s.text))
   315→  if (!covered) return { reject: 'evidence not found in source text' }
   316→  // Page refs must point into the fed section range
   317→  const validPages = pageRefs.filter(p => covered.fromPage <= p && p <= covered.toPage)
   318→  if (validPages.length === 0) return { reject: 'page refs out of range' }
   319→
   320→  return {
   321→    text, options, correctIndex, explanation,
   322→    difficulty: difficulty as GeneratedQuestion['difficulty'],
   323→    category, topic, pageRefs: validPages, evidence,
   324→    sectionIdx: sections.indexOf(covered),
   325→    docId: covered.docId,
   326→    dupKey: normalize(text).slice(0, 140),
   327→  }
   328→}

   332→export async function generateTestJob(testId: string): Promise<void> {
   333→  const test = await db.riisheTest.findUnique({ where: { id: testId } })
   334→  if (!test || !['QUEUED', 'FAILED'].includes(test.status)) return
   335→
   336→  // Every source document behind this test (multi-document practice) —
   337→  // ownership was checked by the route that created the test; here we only
   338→  // load what was recorded. The primary document is always first.
   339→  const docIds = [test.documentId, ...jsonIdArr(test.sourceDocIdsJson).filter(id => id !== test.documentId)]
   340→  const docs = await db.studyDocument.findMany({ where: { id: { in: docIds } } })
   341→  const ordered = docIds.map(id => docs.find(d => d.id === id)).filter((d): d is NonNullable<typeof d> => !!d)
   342→  const primary = ordered[0]
   343→  if (!primary || primary.status !== 'READY' || !primary.extractionJson || ordered.some(d => d.status !== 'READY' || !d.extractionJson)) {
   344→    await db.riisheTest.update({
   345→      where: { id: testId },
   346→      data: { status: 'FAILED', statusDetail: 'The document is not analysed yet. Wait for processing to finish, then try again.' },
   347→    })
   348→    return
   349→  }
   350→
   351→  await db.riisheTest.update({
   352→    where: { id: testId },
   353→    data: { status: 'GENERATING', statusDetail: 'Selecting relevant sections…', updatedAt: new Date() },
   354→  })
   355→
   356→  const focus = safeParse<string[]>(test.focusTopicsJson ?? '[]', []).filter(Boolean)
   357→
   358→  // Whole-document, structure-aware retrieval per source document (never
   359→  // limited to the first pages). Sections carry their own docId so every
   360→  // generated question is attributed to the document that actually supports it.
   361→  const sections: GenSection[] = []
   362→  const analysisOf = new Map<string, DocumentAnalysis | null>()
   363→  for (const d of ordered) {
   364→    const pages = safeParse(d.extractionJson, [] as ExtractedPage[])
   365→    const analysis = safeParse<DocumentAnalysis | null>(d.analysisJson, null)
   366→    analysisOf.set(d.id, analysis)
   367→    const docSections = buildGenerationSections(pages, d.pageCount ?? pages.length, analysis, focus)
   368→    for (const s of docSections) sections.push({ ...s, docId: d.id, docTitle: d.title })
   369→  }
   370→  if (sections.length === 0) {
   371→    await db.riisheTest.update({
   372→      where: { id: testId },
   373→      data: { status: 'FAILED', statusDetail: 'The document has no analysable text, so questions cannot be generated from it.' },
   374→    })
   375→    return
   376→  }
   377→
   378→  // Distribution: the route computed the student's requested mix (count +
   379→  // category selection). Fall back to the scaled default for legacy rows.
   380→  const requested = Math.min(50, Math.max(4, test.requestedCount || 50))
   381→  const storedDist = safeParse<Record<string, number> | null>(test.distributionJson, null)
   382→  const dist: Record<QuestionCategory, number> = { ...DEFAULT_DISTRIBUTION }
   383→  if (storedDist && typeof storedDist === 'object') {
   384→    const valid = (Object.keys(dist) as QuestionCategory[]).filter(k => Number.isFinite(storedDist[k]) && storedDist[k] > 0)
   385→    if (valid.length > 0) {
   386→      for (const k of Object.keys(dist) as QuestionCategory[]) dist[k] = 0
   387→      for (const k of valid) dist[k] = Math.max(1, Math.round(storedDist[k]))
   388→    }
   389→  } else if (requested !== 50) {
   390→    const scale = requested / 50
   391→    for (const k of Object.keys(dist) as QuestionCategory[]) dist[k] = Math.max(1, Math.round(dist[k] * scale))
   392→  }
   393→  const focusCat: QuestionCategory[] = (Object.keys(dist) as QuestionCategory[]).filter(k => dist[k] > 0)
   394→
   395→  // Build the initial batch plan (category → chunk of ≤ BATCH_SIZE)
   396→  const plan: { category: QuestionCategory; count: number }[] = []
   397→  for (const cat of focusCat) {
   398→    let remaining = dist[cat]
   399→    while (remaining > 0) {
   400→      plan.push({ category: cat, count: Math.min(BATCH_SIZE, remaining) })
   401→      remaining -= BATCH_SIZE
   402→    }
   403→  }
   404→
   405→  const candidates: CandidateQuestion[] = []
   406→  const seenKeys = new Set<string>()
   407→  let llmCalls = 0
   408→  let batches = 0
   409→  let providerFailed = false
   410→  /** true when batches were refused by rate limiting — the failure message must
   411→   *  say so instead of falsely claiming the document lacks content. */
   412→  let providerThrottled = false
   413→
   414→  const cursor = { i: 0 }
   415→
   416→  const runBatch = async (category: QuestionCategory, count: number): Promise<void> => {
   417→    if (batches >= MAX_BATCHES) return
   418→    batches++
   419→    const secs = pickSections(sections, focus, cursor, category === 'MISCONCEPTION' ? 3 : 2)
   420→    const avoid = candidates.slice(-30).map(c => c.text.slice(0, 90))
   421→    // A batch that draws on a question-paper document only informs STYLE — the
   422→    // prompt is told so it never treats question wording as factual evidence.
   423→    const questionPaper = secs.some(s => analysisOf.get(s.docId)?.structure?.kind === 'QUESTION_PAPER')
   424→    const callGenerate = async (): Promise<RawQuestion[]> => {
   425→      llmCalls++
   426→      const raws = await chatJson<RawQuestion[]>({
   427→        system: 'You are Riishe, a meticulous exam-question generator. Output only valid JSON.',
   428→        user: generationPrompt({
   429→          category, count,
   430→          sections: secs.map(s => ({ label: `${s.docTitle} · ${s.label}`, text: s.text })),
   431→          focus, avoid,
   432→          examLabel: primary.examLabel,
   433→          questionPaper,
   434→        }),
   435→      })
   436→      return Array.isArray(raws) ? raws : []
   437→    }
   438→    let raws: RawQuestion[]
   439→    try {
   440→      raws = await callGenerate()
   441→    } catch (err) {
   442→      if (isRateLimitError(err)) {
   443→        // One bounded backoff per batch — sustained throttling still ends the
   444→        // run honestly (see the publish message), it is never hidden.
   445→        providerThrottled = true
   446→        console.warn('[riishe generate] rate limited — backing off 15s once')
   447→        await new Promise(r => setTimeout(r, RATE_LIMIT_BACKOFF_MS))
   448→        try {
   449→          raws = await callGenerate()
   450→        } catch (err2) {
   451→          providerFailed = providerFailed || isProviderUnavailable(err2)
   452→          console.error('[riishe generate] batch failed after rate-limit backoff:', err2)
   453→          return
   454→        }
   455→      } else {
   456→        if (isProviderUnavailable(err)) providerFailed = true
   457→        console.error('[riishe generate] batch failed:', err)
   458→        return
   459→      }
   460→    }
   461→
   462→    // Programmatic validation
   463→    const passed: CandidateQuestion[] = []
   464→    for (const raw of raws) {
   465→      const v = programmaticValidate(raw, category, secs)
   466→      if ('reject' in v) continue
   467→      if (seenKeys.has(v.dupKey)) continue
   468→      // Near-duplicate check against everything so far
   469→      const tokens = tokenSet(v.text)
   470→      if (candidates.some(c => jaccard(tokenSet(c.text), tokens) > 0.78)) continue
   471→      seenKeys.add(v.dupKey)
   472→      candidates.push(v)
   473→      passed.push(v)
   474→    }
   475→
   476→    // LLM validation pass over the survivors of this batch
   477→    if (passed.length > 0) {
   478→      try {
   479→        llmCalls++
   480→        const verdicts = await chatJson<ValidationVerdict[]>({
   481→          system: 'You are Riishe\'s question validator. You verify MCQs against source text. Output only valid JSON.',
   482→          user: `Verify each question below against the SOURCE EXCERPTS. Check: (1) the keyed answer is factually correct per the source, (2) exactly one clearly correct option, (3) distractors are wrong, (4) the stem is clear and answerable from the source alone.
   483→
   484→Return a JSON array: [{"index": <position in the list starting at 0>, "verdict": "VALID"|"INVALID", "note": short reason}]
   485→
   486→${fenceDocumentText('student study material', secs.map(s => `--- (${s.docTitle} · ${s.label}) ---\n${s.text}`).join('\n\n'))}
   487→
   488→QUESTIONS:
   489→${passed.map((q, i) => `Q${i}. ${q.text}\nOptions: ${q.options.map((o, j) => `${'ABCD'[j]}. ${o}`).join(' | ')}\nKeyed correct: ${'ABCD'[q.correctIndex]}\nEvidence: "${q.evidence.slice(0, 250)}"`).join('\n\n')}`,
   490→        })
   491→        const byIndex = new Map<number, ValidationVerdict>()
   492→        for (const v of (Array.isArray(verdicts) ? verdicts : [])) {
   493→          if (v && typeof v.index === 'number') byIndex.set(v.index, v)
   494→        }
   495→        for (let i = passed.length - 1; i >= 0; i--) {
   496→          const verdict = byIndex.get(i)
   497→          const q = passed[i]
   498→          const idx = candidates.indexOf(q)
   499→          if (verdict?.verdict === 'INVALID') {
   500→            if (idx >= 0) candidates.splice(idx, 1)
   501→          } else if (verdict?.note) {
   502→            q.validationNote = String(verdict.note).slice(0, 200)
   503→          }
   504→        }
   505→      } catch { /* validator failure keeps the programmatic verdicts — honest middle ground */ }
   506→    }
   507→  }
   508→
   509→  // Initial batches
   510→  for (const step of plan) {
   511→    if (countValid() >= requested) break
   512→    await runBatch(step.category, step.count)
   513→    await db.riisheTest.update({
   514→      where: { id: testId },
   515→      data: { statusDetail: `Generated ${candidates.length} candidate questions so far…` },
   516→    }).catch(() => {})
   517→  }
   518→
   519→  // Top-up rounds (categories with the largest deficits first)
   520→  let guard = 0
   521→  while (countValid() < requested && batches < MAX_BATCHES && guard < 4 && !providerFailed) {
   522→    guard++
   523→    const deficit = categoryDeficit(dist, candidates, requested)
   524→    const cat = deficit[0]?.category ?? 'APPLICATION'
   525→    await runBatch(cat, Math.min(BATCH_SIZE, requested - countValid()))
   526→  }
   527→
   528→  function countValid(): number {
   529→    return candidates.length
   530→  }
   531→
   532→  // ── Publish ──
   533→  const published = candidates.slice(0, requested)
   534→  if (published.length < MIN_PUBLISHABLE) {
   535→    const throttleNote = providerThrottled
   536→      ? 'The AI provider kept rate-limiting generation, so no validated questions could be produced yet — this says nothing about the document itself. Your analysis is safe; please try generating again in a few minutes.'
   537→      : `Only ${published.length} question${published.length === 1 ? '' : 's'} passed validation — the document does not contain enough reliable, testable content for a full test. Try a more detailed document or a different topic.`
   538→    await db.riisheTest.update({
   539→      where: { id: testId },
   540→      data: {
   541→        status: 'FAILED',
   542→        statusDetail: providerFailed
   543→          ? `Question generation failed: ${AI_UNAVAILABLE_MESSAGE}`
   544→          : throttleNote,
   545→        statsJson: JSON.stringify(statsSnapshot(candidates, llmCalls, batches)),
   546→      },
   547→    })
   548→    return
   549→  }
   550→
   551→  // Persist questions + publish (each question keeps the document that
   552→  // actually supports it, so provenance and page chips deep-link correctly)
   553→  await db.riisheQuestion.deleteMany({ where: { testId } })
   554→  const rows = published.map((q, i) => ({
   555→    testId,
   556→    documentId: q.docId,
   557→    order: i + 1,
   558→    text: q.text,
   559→    optionsJson: JSON.stringify(q.options),
   560→    correctIndex: q.correctIndex,
   561→    explanation: q.explanation,
   562→    difficulty: q.difficulty,
   563→    category: q.category,
   564→    topic: q.topic,
   565→    pageRefsJson: JSON.stringify(q.pageRefs),
   566→    evidence: q.evidence.slice(0, 600),
   567→    status: 'VALIDATED',
   568→    validationNote: q.validationNote ?? null,
   569→  }))
   570→  await db.riisheQuestion.createMany({ data: rows })
   571→  const saved = await db.riisheQuestion.findMany({ where: { testId }, select: { id: true }, orderBy: { order: 'asc' } })
   572→
   573→  const rejected = llmCalls > 0 ? Math.max(0, batches * BATCH_SIZE - candidates.length) : 0
   574→  const honestNote = published.length < requested
   575→    ? providerThrottled
   576→      ? `${published.length} of the requested ${requested} questions validated — the AI provider rate-limited some batches, so the test is smaller than requested. Try generating again later for more.`
   577→      : `${published.length} of the requested ${requested} questions validated — the material supports a test of this size.`
   578→    : null
   579→
   580→  await db.riisheTest.update({
   581→    where: { id: testId },
   582→    data: {
   583→      status: 'READY',
   584→      statusDetail: honestNote,
   585→      questionIdsJson: JSON.stringify(saved.map(q => q.id)),
   586→      statsJson: JSON.stringify(statsSnapshot(published, llmCalls, batches, rejected)),
   587→    },
   588→  })
   589→
   590→  function statsSnapshot(list: CandidateQuestion[], calls: number, batchCount: number, rejectedCount?: number): RiisheTestStats {
   591→    const byCategory: RiisheTestStats['byCategory'] = {}
   592→    const byDifficulty = { EASY: 0, MEDIUM: 0, HARD: 0 }
   593→    for (const q of list) {
   594→      byCategory[q.category] = (byCategory[q.category] ?? 0) + 1
   595→      byDifficulty[q.difficulty as 'EASY' | 'MEDIUM' | 'HARD']++
   596→    }
   597→    return {
   598→      generated: list.length,
   599→      validated: list.length,
   600→      rejected: rejectedCount ?? Math.max(0, batchCount * BATCH_SIZE - list.length),
   601→      byCategory,
   602→      byDifficulty,
   603→      llmCalls: calls,
   604→      batches: batchCount,
   605→    }
   606→  }
   607→
   608→  function categoryDeficit(distMap: Record<QuestionCategory, number>, list: CandidateQuestion[], total: number): { category: QuestionCategory; deficit: number }[] {
   609→    return (Object.keys(distMap) as QuestionCategory[])
   610→      .map(cat => ({ category: cat, deficit: Math.round(distMap[cat] / 50 * total) - list.filter(q => q.category === cat).length }))
   611→      .sort((a, b) => b.deficit - a.deficit)
   612→  }
   613→}

   617→  try {

   619→    return Array.isArray(v) ? v.map(String) : []
   620→  } catch {
   621→    return []
   622→  }
