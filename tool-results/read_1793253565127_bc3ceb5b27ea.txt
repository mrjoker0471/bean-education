     1→// ─── Riishe test generation: 50 challenging, VALIDATED questions ─────────────
     2→// Pipeline (all server-side, background job):
     3→//   1. Retrieve page-aligned sections relevant to the focus topics.
     4→//   2. Generate per-category batches (configurable distribution) with strict
     5→//      grounding rules — every question must cite pages + a verbatim excerpt.
     6→//   3. Programmatic validation: structure, duplicates, evidence-in-document
     7→//      check, page-range check.
     8→//   4. LLM validation pass: is the keyed answer actually correct and supported?
     9→//   5. Top-up rounds until the requested count is validated or the batch
    10→//      budget is spent.
    11→//   6. Publish only validated questions. Short/poor sources get a smaller
    12→//      validated test with an honest note — never 50 unchecked outputs.
    13→

    25→const GEN_SECTION_CHARS = 4200
    26→const BATCH_SIZE = 10
    27→const MIN_BATCHES = 9 // floor keeps small tests as generous as the original budget
    28→const MAX_BATCHES = 30 // absolute ceiling — always bounded, never unbounded
    29→const MIN_PUBLISHABLE = 4
    30→const RATE_LIMIT_BACKOFF_MS = 15_000
    31→
    32→/**
    33→ * Batch budget for one run: enough batches to reach the requested count plus
    34→ * top-up headroom, bounded both ways. 50 → 9 (the original budget), 100 → 14,
    35→ * 200 → 24. A bigger request buys more attempts; it never removes the ceiling.
    36→ */
    37→export function batchBudgetFor(requested: number): number {
    38→  return Math.max(MIN_BATCHES, Math.min(MAX_BATCHES, Math.ceil(requested / BATCH_SIZE) + 4))
    39→}

    41→/** Provider outages that make further batches pointless this run. */
    42→function isProviderUnavailable(err: unknown): boolean {
    43→  const msg = err instanceof Error ? err.message : String(err)
    44→  return /empty provider|temporar|timeout|unavailable|ECONN|ENOTFOUND|network/i.test(msg)
    45→}

    55→// ─── Normalisation / duplicate detection ──────────────────────────────────────
    56→
    57→function normalize(s: string): string {
    58→  return s.toLowerCase().replace(/[^a-z0-9]+/g, ' ').trim()
    59→}
    60→
    61→function tokenSet(s: string): Set<string> {
    62→  return new Set(normalize(s).split(' ').filter(w => w.length > 2))
    63→}
    64→
    65→function jaccard(a: Set<string>, b: Set<string>): number {
    66→  if (a.size === 0 || b.size === 0) return 0
    67→  let inter = 0
    68→  for (const w of a) if (b.has(w)) inter++
    69→  return inter / (a.size + b.size - inter)
    70→}
    71→
    72→/** Loose containment check for evidence: whitespace/punctuation insensitive. */
    73→function evidenceInText(evidence: string, text: string): boolean {
    74→  const norm = (x: string) => x.toLowerCase().replace(/[^a-z0-9]+/g, ' ').trim()
    75→  const ev = norm(evidence)
    76→  if (ev.length < 12) return false
    77→  const hay = norm(text)
    78→  if (hay.includes(ev)) return true
    79→  // Sliding window: evidence may reflow across line breaks
    80→  const words = ev.split(' ')
    81→  if (words.length >= 4) {
    82→    const half = Math.floor(words.length / 2)
    83→    return hay.includes(words.slice(0, half).join(' ')) || hay.includes(words.slice(half).join(' '))
    84→  }
    85→  return false
    86→}

   103→/**
   104→ * Build the generation section pool from the WHOLE document using the same
   105→ * deterministic structure detection the analysis pipeline uses (so retrieval
   106→ * can never be silently limited to the first few pages). When the student
   107→ * picked focus topics, the pool leads with the ACTUAL pages those topics sit
   108→ * on (from the verified analysis page references), padded with balanced
   109→ * coverage so batches always have enough material.
   110→ */
   111→export function buildGenerationSections(
   112→  pages: ExtractedPage[],
   113→  numPages: number,
   114→  analysis: DocumentAnalysis | null,
   115→  focus: string[],
   116→): BaseGenSection[] {
   117→  const readable = pages.filter(p => !p.scanned && p.text.length > 0)
   118→  const pageByNumber = new Map(readable.map(p => [p.page, p] as const))
   119→  const textOf = (from: number, to: number) =>
   120→    readable
   121→      .filter(p => p.page >= from && p.page <= to)
   122→      .map(p => `[page ${p.page}]\n${p.text}`)
   123→      .join('\n\n')
   124→
   125→  const fromUnits = (): BaseGenSection[] => {
   126→    const structure = detectStructure(pages, numPages, { maxUnitChars: GEN_SECTION_CHARS })
   127→    if (structure.units.length === 0) return []
   128→    return structure.units.map(u => ({
   129→      label: u.title ? `${u.title} — pages ${u.fromPage}${u.toPage !== u.fromPage ? `–${u.toPage}` : ''}` : u.fromPage === u.toPage ? `page ${u.fromPage}` : `pages ${u.fromPage}–${u.toPage}`,
   130→      text: textOf(u.fromPage, u.toPage),
   131→      fromPage: u.fromPage,
   132→      toPage: u.toPage,
   133→    }))
   134→  }
   135→
   136→  const fallbackSections = (): BaseGenSection[] =>
   137→    sectionPages(pages, { maxChars: GEN_SECTION_CHARS, maxSections: 400 }).map(s => ({
   138→      label: s.label,
   139→      text: s.text,
   140→      fromPage: s.fromPage,
   141→      toPage: s.toPage,
   142→    }))
   143→
   144→  let sections = fromUnits()
   145→  if (sections.length === 0) sections = fallbackSections()
   146→  if (sections.length === 0) return []
   147→
   148→  if (focus.length > 0 && analysis?.topics?.length) {
   149→    const normFocus = new Set(focus.map(f => normalizeTitle(f)))
   150→    const matched = analysis.topics.filter(t =>
   151→      normFocus.has(normalizeTitle(t.name)) ||
   152→      (t.aliases ?? []).some(a => normFocus.has(normalizeTitle(a))),
   153→    )
   154→    // include children of matched sections so the focus covers the sub-topics too
   155→    const byParent = new Map(analysis.topics.map(t => [t.id ?? '', t] as const))
   156→    for (const t of analysis.topics) {
   157→      if (t.parentId && byParent.get(t.parentId) && matched.some(m => m.id === t.parentId)) matched.push(t)
   158→    }
   159→    const pageSet = new Set<number>()
   160→    for (const t of matched) for (const p of t.pages) if (pageByNumber.has(p)) pageSet.add(p)
   161→    if (pageSet.size > 0) {
   162→      // expand by one page for context, then group into contiguous runs
   163→      for (const p of [...pageSet]) {
   164→        if (pageByNumber.has(p - 1)) pageSet.add(p - 1)
   165→        if (pageByNumber.has(p + 1)) pageSet.add(p + 1)
   166→      }
   167→      const sorted = [...pageSet].sort((a, b) => a - b)
   168→      const runs: { from: number; to: number }[] = []
   169→      for (const p of sorted) {
   170→        const last = runs[runs.length - 1]
   171→        if (last && p <= last.to + 1) last.to = p
   172→        else runs.push({ from: p, to: p })
   173→      }
   174→  const focusSections: BaseGenSection[] = runs
   175→        .map(r => ({
   176→          label: r.from === r.to ? `page ${r.from}` : `pages ${r.from}–${r.to}`,
   177→          text: textOf(r.from, r.to).slice(0, GEN_SECTION_CHARS * 2),
   178→          fromPage: r.from,
   179→          toPage: r.to,
   180→        }))
   181→        .filter(s => s.text.length > 0)
   182→      const totalChars = focusSections.reduce((s, x) => s + x.text.length, 0)
   183→      if (focusSections.length > 0 && totalChars >= 1500) {
   184→        // top up with balanced coverage if the focus area is thin
   185→        const out = [...focusSections]
   186→        let chars = totalChars
   187→        for (const sec of sections) {
   188→          if (chars >= 6000) break
   189→          if (out.some(o => o.fromPage <= sec.toPage && sec.fromPage <= o.toPage)) continue
   190→          out.push(sec)
   191→          chars += sec.text.length
   192→        }
   193→        return out
   194→      }
   195→    }
   196→  }
   197→  return sections
   198→}
   199→
   200→function scoreSection(sectionText: string, focus: string[]): number {
   201→  if (focus.length === 0) return 0
   202→  const hay = normalize(sectionText)
   203→  let score = 0
   204→  for (const f of focus) {
   205→    const nf = normalize(f)
   206→    if (nf && hay.includes(nf)) score += 2
   207→    for (const w of nf.split(' ')) {
   208→      if (w.length > 4 && hay.includes(w)) score += 1
   209→    }
   210→  }
   211→  return score
   212→}

   214→function pickSections(
   215→  sections: GenSection[],
   216→  focus: string[],
   217→  cursor: { i: number },
   218→  count: number,
   219→): GenSection[] {
   220→  if (focus.length > 0) {
   221→    const ranked = [...sections].sort((a, b) => scoreSection(b.text, focus) - scoreSection(a.text, focus))
   222→    const top = ranked.slice(0, count)
   223→    if (scoreSection(top[top.length - 1]?.text ?? '', focus) > 0) return top
   224→  }
   225→  // Balanced round-robin over the document
   226→  const out: typeof sections = []
   227→  for (let k = 0; k < count && sections.length > 0; k++) {
   228→    out.push(sections[cursor.i % sections.length])
   229→    cursor.i++
   230→  }
   231→  return out
   232→}

   277→
   278→"explanation" must teach: why the correct answer is right AND why the tempting wrong options fail (2-4 sentences).`
   279→}

   298→// ─── Programmatic validation ───────────────────────────────────────────────────

   300→function programmaticValidate(
   301→  raw: RawQuestion,
   302→  category: QuestionCategory,
   303→  sections: GenSection[],
   304→): CandidateQuestion | { reject: string } {
   305→  const text = String(raw.text ?? '').trim()
   306→  const options = (raw.options ?? []).map(o => String(o ?? '').trim())
   307→  const correctIndex = Number(raw.correctIndex)
   308→  const explanation = String(raw.explanation ?? '').trim()
   309→  const evidence = String(raw.evidence ?? '').trim()
   310→  const pageRefs = (raw.pageRefs ?? []).map(Number).filter(n => Number.isFinite(n))
   311→  const difficulty = ['EASY', 'MEDIUM', 'HARD'].includes(String(raw.difficulty)) ? String(raw.difficulty) : 'MEDIUM'
   312→  const topic = String(raw.topic ?? '').trim().slice(0, 120) || 'General'
   313→
   314→  if (text.length < 20 || text.length > 800) return { reject: 'stem too short/long' }
   315→  if (options.length !== 4 || options.some(o => o.length === 0 || o.length > 300)) return { reject: 'options invalid' }
   316→  const normed = options.map(o => normalize(o))
   317→  if (new Set(normed).size !== 4) return { reject: 'duplicate options' }
   318→  if (!Number.isInteger(correctIndex) || correctIndex < 0 || correctIndex > 3) return { reject: 'bad correctIndex' }
   319→  if (explanation.length < 40) return { reject: 'explanation too thin' }
   320→  if (!evidence) return { reject: 'no evidence' }
   321→  if (evidence.length > 600) return { reject: 'evidence too long' }
   322→
   323→  // Evidence must actually exist in the fed sections (grounding check)
   324→  const covered = sections.find(s => evidenceInText(evidence, s.text))
   325→  if (!covered) return { reject: 'evidence not found in source text' }
   326→  // Page refs must point into the fed section range
   327→  const validPages = pageRefs.filter(p => covered.fromPage <= p && p <= covered.toPage)
   328→  if (validPages.length === 0) return { reject: 'page refs out of range' }
   329→
   330→  // Page-reference precision: pin the citation to the page(s) that actually
   331→  // carry the evidence excerpt. The model sometimes cites the section's first
   332→  // page while quoting text from a later page of the same section — the quote
   333→  // is real, but the chip would land one page off. Split the section into its
   334→  // [page N] blocks and keep only cited pages that hold the evidence; when
   335→  // none does, cite the pages where the evidence actually appears instead.
   336→  const pageTexts = new Map<number, string>()
   337→  for (const block of covered.text.split(/\n\n+/)) {
   338→    const m = /^\[page (\d+)\]/.exec(block)
   339→    if (m) pageTexts.set(Number(m[1]), block)
   340→  }
   341→  const evidencePages = [...pageTexts.entries()]
   342→    .filter(([, txt]) => evidenceInText(evidence, txt))
   343→    .map(([p]) => p)
   344→  const refined = validPages.filter(p => evidencePages.includes(p))
   345→  const finalPages = refined.length > 0
   346→    ? refined
   347→    : evidencePages.filter(p => covered.fromPage <= p && p <= covered.toPage)
   348→
   349→  return {
   350→    text, options, correctIndex, explanation,
   351→    difficulty: difficulty as GeneratedQuestion['difficulty'],
   352→    category, topic, pageRefs: finalPages.length > 0 ? finalPages : validPages, evidence,
   353→    sectionIdx: sections.indexOf(covered),
   354→    docId: covered.docId,
   355→    dupKey: normalize(text).slice(0, 140),
   356→  }
   357→}

   361→export async function generateTestJob(testId: string): Promise<void> {
   362→  const test = await db.riisheTest.findUnique({ where: { id: testId } })
   363→  if (!test || !['QUEUED', 'FAILED'].includes(test.status)) return
   364→
   365→  // Every source document behind this test (multi-document practice) —
   366→  // ownership was checked by the route that created the test; here we only
   367→  // load what was recorded. The primary document is always first.
   368→  const docIds = [test.documentId, ...jsonIdArr(test.sourceDocIdsJson).filter(id => id !== test.documentId)]
   369→  const docs = await db.studyDocument.findMany({ where: { id: { in: docIds } } })
   370→  const ordered = docIds.map(id => docs.find(d => d.id === id)).filter((d): d is NonNullable<typeof d> => !!d)
   371→  const primary = ordered[0]
   372→  if (!primary || primary.status !== 'READY' || !primary.extractionJson || ordered.some(d => d.status !== 'READY' || !d.extractionJson)) {
   373→    await db.riisheTest.update({
   374→      where: { id: testId },
   375→      data: { status: 'FAILED', statusDetail: 'The document is not analysed yet. Wait for processing to finish, then try again.' },
   376→    })
   377→    return
   378→  }
   379→
   380→  await db.riisheTest.update({
   381→    where: { id: testId },
   382→    data: { status: 'GENERATING', statusDetail: 'Selecting relevant sections…', updatedAt: new Date() },
   383→  })
   384→
   385→  const focus = safeParse<string[]>(test.focusTopicsJson ?? '[]', []).filter(Boolean)
   386→
   387→  // Whole-document, structure-aware retrieval per source document (never
   388→  // limited to the first pages). Sections carry their own docId so every
   389→  // generated question is attributed to the document that actually supports it.
   390→  const sections: GenSection[] = []
   391→  const analysisOf = new Map<string, DocumentAnalysis | null>()
   392→  for (const d of ordered) {
   393→    const pages = safeParse(d.extractionJson, [] as ExtractedPage[])
   394→    const analysis = safeParse<DocumentAnalysis | null>(d.analysisJson, null)
   395→    analysisOf.set(d.id, analysis)
   396→    const docSections = buildGenerationSections(pages, d.pageCount ?? pages.length, analysis, focus)
   397→    for (const s of docSections) sections.push({ ...s, docId: d.id, docTitle: d.title })
   398→  }
   399→  if (sections.length === 0) {
   400→    await db.riisheTest.update({
   401→      where: { id: testId },
   402→      data: { status: 'FAILED', statusDetail: 'The document has no analysable text, so questions cannot be generated from it.' },
   403→    })
   404→    return
   405→  }
   406→
   407→  // Distribution: the route computed the student's requested mix (count +
   408→  // category selection). Fall back to the scaled default for legacy rows.
   409→  // Requests beyond 50 (the practice builder offers up to 200) are honoured
   410→  // when the material supports them — the batch budget scales with the count,
   411→  // and validation still publishes only what actually passes.
   412→  const requested = Math.min(200, Math.max(4, test.requestedCount || 50))
   413→  const storedDist = safeParse<Record<string, number> | null>(test.distributionJson, null)
   414→  const dist: Record<QuestionCategory, number> = { ...DEFAULT_DISTRIBUTION }
   415→  if (storedDist && typeof storedDist === 'object') {
   416→    const valid = (Object.keys(dist) as QuestionCategory[]).filter(k => Number.isFinite(storedDist[k]) && storedDist[k] > 0)
   417→    if (valid.length > 0) {
   418→      for (const k of Object.keys(dist) as QuestionCategory[]) dist[k] = 0
   419→      for (const k of valid) dist[k] = Math.max(1, Math.round(storedDist[k]))
   420→    }
   421→  } else if (requested !== 50) {
   422→    const scale = requested / 50
   423→    for (const k of Object.keys(dist) as QuestionCategory[]) dist[k] = Math.max(1, Math.round(dist[k] * scale))
   424→  }
   425→  const focusCat: QuestionCategory[] = (Object.keys(dist) as QuestionCategory[]).filter(k => dist[k] > 0)
   426→  const maxBatches = batchBudgetFor(requested)
   427→
   428→  // Build the initial batch plan (category → chunk of ≤ BATCH_SIZE)
   429→  const plan: { category: QuestionCategory; count: number }[] = []
   430→  for (const cat of focusCat) {
   431→    let remaining = dist[cat]
   432→    while (remaining > 0) {
   433→      plan.push({ category: cat, count: Math.min(BATCH_SIZE, remaining) })
   434→      remaining -= BATCH_SIZE
   435→    }
   436→  }
   437→
   438→  const candidates: CandidateQuestion[] = []
   439→  const seenKeys = new Set<string>()
   440→  let llmCalls = 0
   441→  let batches = 0
   442→  let providerFailed = false
   443→  /** true when batches were refused by rate limiting — the failure message must
   444→   *  say so instead of falsely claiming the document lacks content. */
   445→  let providerThrottled = false
   446→
   447→  const cursor = { i: 0 }
   448→
   449→  const runBatch = async (category: QuestionCategory, count: number): Promise<void> => {
   450→    if (batches >= maxBatches) return
   451→    batches++
   452→    const secs = pickSections(sections, focus, cursor, category === 'MISCONCEPTION' ? 3 : 2)
   453→    const avoid = candidates.slice(-30).map(c => c.text.slice(0, 90))
   454→    // A batch that draws on a question-paper document only informs STYLE — the
   455→    // prompt is told so it never treats question wording as factual evidence.
   456→    const questionPaper = secs.some(s => analysisOf.get(s.docId)?.structure?.kind === 'QUESTION_PAPER')
   457→    const callGenerate = async (): Promise<RawQuestion[]> => {
   458→      llmCalls++
   459→      const raws = await chatJson<RawQuestion[]>({
   460→        system: 'You are Riishe, a meticulous exam-question generator. Output only valid JSON.',
   461→        user: generationPrompt({
   462→          category, count,
   463→          sections: secs.map(s => ({ label: `${s.docTitle} · ${s.label}`, text: s.text })),
   464→          focus, avoid,
   465→          examLabel: primary.examLabel,
   466→          questionPaper,
   467→        }),
   468→      })
   469→      return Array.isArray(raws) ? raws : []
   470→    }
   471→    let raws: RawQuestion[]
   472→    try {
   473→      raws = await callGenerate()
   474→    } catch (err) {
   475→      if (isRateLimitError(err)) {
   476→        // One bounded backoff per batch — sustained throttling still ends the
   477→        // run honestly (see the publish message), it is never hidden.
   478→        providerThrottled = true
   479→        console.warn('[riishe generate] rate limited — backing off 15s once')
   480→        await new Promise(r => setTimeout(r, RATE_LIMIT_BACKOFF_MS))
   481→        try {
   482→          raws = await callGenerate()
   483→        } catch (err2) {
   484→          providerFailed = providerFailed || isProviderUnavailable(err2)
   485→          console.error('[riishe generate] batch failed after rate-limit backoff:', err2)
   486→          return
   487→        }
   488→      } else {
   489→        if (isProviderUnavailable(err)) providerFailed = true
   490→        console.error('[riishe generate] batch failed:', err)
   491→        return
   492→      }
   493→    }
   494→
   495→    // Programmatic validation
   496→    const passed: CandidateQuestion[] = []
   497→    for (const raw of raws) {
   498→      const v = programmaticValidate(raw, category, secs)
   499→      if ('reject' in v) continue
   500→      if (seenKeys.has(v.dupKey)) continue
   501→      // Near-duplicate check against everything so far
   502→      const tokens = tokenSet(v.text)
   503→      if (candidates.some(c => jaccard(tokenSet(c.text), tokens) > 0.78)) continue
   504→      seenKeys.add(v.dupKey)
   505→      candidates.push(v)
   506→      passed.push(v)
   507→    }
   508→
   509→    // LLM validation pass over the survivors of this batch
   510→    if (passed.length > 0) {
   511→      try {
   512→        llmCalls++
   513→        const verdicts = await chatJson<ValidationVerdict[]>({
   514→          system: 'You are Riishe\'s question validator. You verify MCQs against source text. Output only valid JSON.',
   515→          user: `Verify each question below against the SOURCE EXCERPTS. Check: (1) the keyed answer is factually correct per the source, (2) exactly one clearly correct option, (3) distractors are wrong, (4) the stem is clear and answerable from the source alone.
   516→
   517→Return a JSON array: [{"index": <position in the list starting at 0>, "verdict": "VALID"|"INVALID", "note": short reason}]
   518→
   519→${fenceDocumentText('student study material', secs.map(s => `--- (${s.docTitle} · ${s.label}) ---\n${s.text}`).join('\n\n'))}
   520→
   521→QUESTIONS:
   522→${passed.map((q, i) => `Q${i}. ${q.text}\nOptions: ${q.options.map((o, j) => `${'ABCD'[j]}. ${o}`).join(' | ')}\nKeyed correct: ${'ABCD'[q.correctIndex]}\nEvidence: "${q.evidence.slice(0, 250)}"`).join('\n\n')}`,
   523→        })
   524→        const byIndex = new Map<number, ValidationVerdict>()
   525→        for (const v of (Array.isArray(verdicts) ? verdicts : [])) {
   526→          if (v && typeof v.index === 'number') byIndex.set(v.index, v)
   527→        }
   528→        for (let i = passed.length - 1; i >= 0; i--) {
   529→          const verdict = byIndex.get(i)
   530→          const q = passed[i]
   531→          const idx = candidates.indexOf(q)
   532→          if (verdict?.verdict === 'INVALID') {
   533→            if (idx >= 0) candidates.splice(idx, 1)
   534→          } else if (verdict?.note) {
   535→            q.validationNote = String(verdict.note).slice(0, 200)
   536→          }
   537→        }
   538→      } catch { /* validator failure keeps the programmatic verdicts — honest middle ground */ }
   539→    }
   540→  }
   541→
   542→  // Initial batches
   543→  for (const step of plan) {
   544→    if (countValid() >= requested) break
   545→    await runBatch(step.category, step.count)
   546→    await db.riisheTest.update({
   547→      where: { id: testId },
   548→      data: { statusDetail: `Generated ${candidates.length} candidate questions so far…` },
   549→    }).catch(() => {})
   550→  }
   551→
   552→  // Top-up rounds (categories with the largest deficits first)
   553→  let guard = 0
   554→  while (countValid() < requested && batches < maxBatches && guard < 4 && !providerFailed) {
   555→    guard++
   556→    const deficit = categoryDeficit(dist, candidates, requested)
   557→    const cat = deficit[0]?.category ?? 'APPLICATION'
   558→    await runBatch(cat, Math.min(BATCH_SIZE, requested - countValid()))
   559→  }
   560→
   561→  function countValid(): number {
   562→    return candidates.length
   563→  }
   564→
   565→  // ── Publish ──
   566→  const published = candidates.slice(0, requested)
   567→  if (published.length < MIN_PUBLISHABLE) {
   568→    const throttleNote = providerThrottled
   569→      ? 'The AI provider kept rate-limiting generation, so no validated questions could be produced yet — this says nothing about the document itself. Your analysis is safe; please try generating again in a few minutes.'
   570→      : `Only ${published.length} question${published.length === 1 ? '' : 's'} passed validation — the document does not contain enough reliable, testable content for a full test. Try a more detailed document or a different topic.`
   571→    await db.riisheTest.update({
   572→      where: { id: testId },
   573→      data: {
   574→        status: 'FAILED',
   575→        statusDetail: providerFailed
   576→          ? `Question generation failed: ${AI_UNAVAILABLE_MESSAGE}`
   577→          : throttleNote,
   578→        statsJson: JSON.stringify(statsSnapshot(candidates, llmCalls, batches)),
   579→      },
   580→    })
   581→    return
   582→  }
   583→
   584→  // Persist questions + publish (each question keeps the document that
   585→  // actually supports it, so provenance and page chips deep-link correctly)
   586→  await db.riisheQuestion.deleteMany({ where: { testId } })
   587→  const rows = published.map((q, i) => ({
   588→    testId,
   589→    documentId: q.docId,
   590→    order: i + 1,
   591→    text: q.text,
   592→    optionsJson: JSON.stringify(q.options),
   593→    correctIndex: q.correctIndex,
   594→    explanation: q.explanation,
   595→    difficulty: q.difficulty,
   596→    category: q.category,
   597→    topic: q.topic,
   598→    pageRefsJson: JSON.stringify(q.pageRefs),
   599→    evidence: q.evidence.slice(0, 600),
   600→    status: 'VALIDATED',
   601→    validationNote: q.validationNote ?? null,
   602→  }))
   603→  await db.riisheQuestion.createMany({ data: rows })
   604→  const saved = await db.riisheQuestion.findMany({ where: { testId }, select: { id: true }, orderBy: { order: 'asc' } })
   605→
   606→  const rejected = llmCalls > 0 ? Math.max(0, batches * BATCH_SIZE - candidates.length) : 0
   607→  const honestNote = published.length < requested
   608→    ? providerThrottled
   609→      ? `${published.length} of the requested ${requested} questions validated — the AI provider rate-limited some batches, so the test is smaller than requested. Try generating again later for more.`
   610→      : `${published.length} of the requested ${requested} questions validated — the material supports a test of this size.`
   611→    : null
   612→
   613→  await db.riisheTest.update({
   614→    where: { id: testId },
   615→    data: {
   616→      status: 'READY',
   617→      statusDetail: honestNote,
   618→      questionIdsJson: JSON.stringify(saved.map(q => q.id)),
   619→      statsJson: JSON.stringify(statsSnapshot(published, llmCalls, batches, rejected)),
   620→    },
   621→  })

   623→  function statsSnapshot(list: CandidateQuestion[], calls: number, batchCount: number, rejectedCount?: number): RiisheTestStats {
   624→    const byCategory: RiisheTestStats['byCategory'] = {}
   625→    const byDifficulty = { EASY: 0, MEDIUM: 0, HARD: 0 }
   626→    for (const q of list) {
   627→      byCategory[q.category] = (byCategory[q.category] ?? 0) + 1
   628→      byDifficulty[q.difficulty as 'EASY' | 'MEDIUM' | 'HARD']++
   629→    }
   630→    return {
   631→      generated: list.length,
   632→      validated: list.length,
   633→      rejected: rejectedCount ?? Math.max(0, batchCount * BATCH_SIZE - list.length),
   634→      byCategory,
   635→      byDifficulty,
   636→      llmCalls: calls,
   637→      batches: batchCount,
   638→    }
   639→  }
   640→
   641→  function categoryDeficit(distMap: Record<QuestionCategory, number>, list: CandidateQuestion[], total: number): { category: QuestionCategory; deficit: number }[] {
   642→    // Scale by the distribution's own total (it sums to the requested count),
   643→    // never a hard-coded 50 — otherwise top-ups skew for 20/30/100-question runs.
   644→    const distSum = (Object.keys(distMap) as QuestionCategory[]).reduce((s, k) => s + Math.max(0, distMap[k]), 0) || 1
   645→    return (Object.keys(distMap) as QuestionCategory[])
   646→      .map(cat => ({ category: cat, deficit: Math.round(distMap[cat] / distSum * total) - list.filter(q => q.category === cat).length }))
   647→      .sort((a, b) => b.deficit - a.deficit)
   648→  }
   649→}
