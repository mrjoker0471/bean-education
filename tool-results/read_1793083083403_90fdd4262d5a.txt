     1→     1→// ─── Riishe document pipeline: extraction → structure → grounded analysis ────
     2→     2→// Runs as a background job.
     3→     3→//   Stage 1  extraction (local pdf.js, per-page text with real line breaks)
     4→     4→//   Stage 2  deterministic structure detection (TOC, headings, printed pages)
     5→     5→//   Stage 3  AI analysis over EVERY structure-aware section of the document —
     6→     6→//            there is no "first N pages" limit; a section that fails is
     7→     7→//            recorded and retried without discarding the sections that worked
     8→     8→//   Stage 4  deterministic merge: duplicate topics merged at the data level,
     9→     9→//            page references verified against the extracted page text,
    10→    10→//            coverage accounted honestly (extracted / OCR-needed / failed)
    11→    11→// The uploaded material is untrusted data — every model call fences it.

    43→    43→export function collectTopicInputs(
    44→    44→  parts: (SectionAnalysis | null)[],
    45→    45→  units: AnalysisUnit[],
    46→    46→  anchoredPages: Map<number, number | null>,
    47→    47→  tocPages: number[] = [],
    48→    48→): TopicInput[] {
    49→    49→  const inputs: TopicInput[] = []
    50→    50→  // Heading-anchored topics: each headed section contributes its own heading
    51→    51→  // as a topic — that is the document's real organisation, not model opinion.
    52→    52→  for (const u of units) {
    53→    53→    if (u.kind === 'SECTION' && u.title) {
    54→    54→      inputs.push({
    55→    55→        name: u.title,
    56→    56→        subtopics: [],
    57→    57→        pages: pagesBetween(u.fromPage, u.toPage),
    58→    58→        unitIndex: u.index,
    59→    59→        unitTitle: u.title,
    60→    60→        unitFromPage: u.fromPage,
    61→    61→        unitToPage: u.toPage,
    62→    62→        anchored: true,
    63→    63→      })
    64→    64→    }
    65→    65→  }
    66→    66→  for (const u of units) {
    67→    67→    const part = parts[u.index]
    68→    68→    if (!part) continue
    69→    69→    // A front-matter unit that contains the document's contents pages yields
    70→    70→    // only INDEX ECHOES ("Eligibility Criteria" from the contents line) —
    71→    71→    // those are a map of the document, not its content. Skip them; the real
    72→    72→    // chapters contribute their own anchored topics.
    73→    73→    const isIndexEcho = u.kind === 'FRONT' && tocPages.some(tp => tp >= u.fromPage && tp <= u.toPage)
    74→    74→    if (isIndexEcho) continue
    75→    75→    for (const t of part.topics ?? []) {
    76→    76→      const name = String(t?.name ?? '').trim()
    77→    77→      if (!name) continue
    78→    78→      inputs.push({
    79→    79→        name,
    80→    80→        subtopics: (t.subtopics ?? []).map(s => String(s)).filter(Boolean),
    81→    81→        pages: (t.pages ?? []).filter(Number.isFinite),
    82→    82→        unitIndex: u.index,
    83→    83→        unitTitle: u.title,
    84→    84→        unitFromPage: u.fromPage,
    85→    85→        unitToPage: u.toPage,
    86→    86→      })
    87→    87→    }
    88→    88→  }
    89→    89→  void anchoredPages
    90→    90→  return inputs
    91→    91→}

    93→    93→function pagesBetween(from: number, to: number): number[] {
    94→    94→  const out: number[] = []
    95→    95→  for (let p = from; p <= to; p++) out.push(p)
    96→    96→  return out

    99→    99→function mergeFacts(parts: (SectionAnalysis | null)[], cap: number, pick: (p: SectionAnalysis) => string[]): string[] {
   100→   100→  const seen = new Set<string>()
   101→   101→  const out: string[] = []
   102→   102→  for (const part of parts) {
   103→   103→    if (!part) continue
   104→   104→    for (const raw of pick(part)) {
   105→   105→      const f = String(raw ?? '').trim()
   106→   106→      if (!f) continue
   107→   107→      const key = normalizeTitle(f).slice(0, 100)
   108→   108→      if (key && seen.has(key)) continue
   109→   109→      seen.add(key)
   110→   110→      out.push(f.slice(0, 400))
   111→   111→      if (out.length >= cap) return out
   112→   112→    }
   113→   113→  }
   114→   114→  return out
   115→   115→}

   173→   173→function buildCoverage(
   174→   174→  extraction: { numPages: number; scannedPages: number },
   175→   175→  readablePages: ExtractedPage[],
   176→   176→  units: AnalysisUnit[],
   177→   177→  outcomes: SectionOutcome[],
   178→   178→): CoverageMetrics {
   179→   179→  const unitPages = (u: AnalysisUnit) => pagesBetween(u.fromPage, u.toPage)
   180→   180→  const okUnits = units.filter((u, i) => outcomes[i]?.status === 'OK')
   181→   181→  const failedUnits = units.filter((u, i) => outcomes[i]?.status === 'FAILED')
   182→   182→  const skippedUnits = units.filter((u, i) => outcomes[i]?.status === 'SKIPPED')
   183→   183→  const accountedPageSet = new Set<number>()
   184→   184→  for (const u of okUnits) for (const p of unitPages(u)) accountedPageSet.add(p)
   185→   185→  const skippedPageSet = new Set<number>()
   186→   186→  for (const u of skippedUnits) for (const p of unitPages(u)) skippedPageSet.add(p)
   187→   187→  const readableSet = new Set(readablePages.map(p => p.page))
   188→   188→  const unanalysed = [...readableSet].filter(p => !accountedPageSet.has(p) && !skippedPageSet.has(p)).length
   189→   189→  return {
   190→   190→    totalPages: extraction.numPages,
   191→   191→    readablePages: readablePages.length,
   192→   192→    scannedPages: extraction.scannedPages,
   193→   193→    analysedPages: accountedPageSet.size,
   194→   194→    unanalysedPages: unanalysed,
   195→   195→    skippedPages: [...readableSet].filter(p => skippedPageSet.has(p)).length,
   196→   196→    unitsTotal: units.length,
   197→   197→    unitsOk: okUnits.length,
   198→   198→    unitsFailed: failedUnits.length,
   199→   199→    unitsSkipped: skippedUnits.length,
   200→   200→    complete: unanalysed === 0 && failedUnits.length === 0 && units.length > 0,
   201→   201→  }
   202→   202→}

   204→   204→function buildCoverageNote(cov: CoverageMetrics): string {
   205→   205→  const bits: string[] = []
   206→   206→  if (cov.unitsFailed > 0) {
   207→   207→    bits.push(`Analysed ${cov.analysedPages} of ${cov.readablePages} readable page${cov.readablePages === 1 ? '' : 's'} — ${cov.unitsFailed} section${cov.unitsFailed === 1 ? '' : 's'} failed and can be retried without losing the sections that succeeded.`)
   208→   208→  } else if (cov.unitsSkipped > 0) {
   209→   209→    bits.push(`All readable pages accounted for: ${cov.analysedPages} analysed across ${cov.unitsOk} section${cov.unitsOk === 1 ? '' : 's'}, ${cov.skippedPages} page${cov.skippedPages === 1 ? '' : 's'} in ${cov.unitsSkipped} section${cov.unitsSkipped === 1 ? '' : 's'} skipped with a recorded reason (see below).`)
   210→   210→  } else {
   211→   211→    bits.push(`All ${cov.readablePages} readable page${cov.readablePages === 1 ? '' : 's'} analysed across ${cov.unitsTotal} section${cov.unitsTotal === 1 ? '' : 's'}.`)
   212→   212→  }
   213→   213→  if (cov.scannedPages > 0) {
   214→   214→    bits.push(`${cov.scannedPages} image-only page${cov.scannedPages === 1 ? '' : 's'} would need OCR, which is not available yet — they are excluded from the analysis.`)
   215→   215→  }
   216→   216→  return bits.join(' ')
   217→   217→}

   219→   219→// ─── Main job ─────────────────────────────────────────────────────────────────
   220→   220→
   221→   221→/**
   222→   222→ * Background job: extract text, detect the document's real structure, analyse
   223→   223→ * every section, merge deterministically.
   224→   224→ *  opts.force      — full re-analysis of a READY document (fresh extraction)
   225→   225→ *  partial retry   — a READY v2 analysis with failed sections re-runs ONLY the
   226→   226→ *                    failed sections and keeps every successful result
   227→   227→ */
   228→   228→export async function processDocument(documentId: string, opts?: { force?: boolean }): Promise<void> {
   229→   229→  const force = opts?.force === true
   230→   230→  const doc = await db.studyDocument.findUnique({ where: { id: documentId } })
   231→   231→  if (!doc) return
   232→   232→
   233→   233→  const stored = safeParse<StoredAnalysis | null>(doc.analysisJson, null)
   234→   234→  const canPartialRetry =
   235→   235→    !force &&
   236→   236→    doc.status === 'READY' &&
   237→   237→    stored?.pipelineVersion === PIPELINE_VERSION &&
   238→   238→    Array.isArray(stored?._parts) &&
   239→   239→    Array.isArray(stored?.sectionOutcomes) &&
   240→   240→    stored?.coverage?.complete === false
   241→   241→
   242→   242→  const allowed = force ? ['QUEUED', 'FAILED', 'NEEDS_OCR', 'READY'] : ['QUEUED', 'FAILED', 'NEEDS_OCR']
   243→   243→  if (!allowed.includes(doc.status) && !canPartialRetry) return
   244→   244→
   245→   245→  // Identical-file reuse — only from the SAME pipeline version, so an upgrade
   246→   246→  // always re-analyses instead of copying a stale analysis.
   247→   247→  let twinAnalysisJson: string | null = null
   248→   248→  if (!force && (doc.status === 'FAILED' || doc.status === 'QUEUED')) {
   249→   249→    const twin = await db.studyDocument.findFirst({
   250→   250→      where: { contentHash: doc.contentHash, status: 'READY', id: { not: doc.id } },
   251→   251→      select: { analysisJson: true },
   252→   252→    })
   253→   253→    const twinJson = twin?.analysisJson ?? null
   254→   254→    const twinParsed = safeParse<StoredAnalysis | null>(twinJson, null)
   255→   255→    if (twinParsed?.pipelineVersion === PIPELINE_VERSION && twinParsed.topics) twinAnalysisJson = twinJson
   256→   256→  }
   257→   257→
   258→   258→  await db.studyDocument.update({
   259→   259→    where: { id: documentId },
   260→   260→    data: {
   261→   261→      status: 'EXTRACTING',
   262→   262→      statusDetail: canPartialRetry ? 'Preparing to re-analyse the failed sections…' : 'Reading the uploaded file…',
   263→   263→      updatedAt: new Date(),
   264→   264→    },
   265→   265→  })
   266→   266→
   267→   267→  let buffer: Buffer
   268→   268→  try {
   269→   269→    buffer = await readFile(doc.storedPath)
   270→   270→  } catch {
   271→   271→    await db.studyDocument.update({
   272→   272→      where: { id: documentId },
   273→   273→      data: { status: 'FAILED', statusDetail: 'The stored file could not be read. Please upload it again.' },
   274→   274→    })
   275→   275→    return
   276→   276→  }
   277→   277→
   278→   278→  // Verify the stored content still matches its hash (integrity)
   279→   279→  const hash = createHash('sha256').update(buffer).digest('hex')
   280→   280→  if (hash !== doc.contentHash) {
   281→   281→    await db.studyDocument.update({
   282→   282→      where: { id: documentId },
   283→   283→      data: { status: 'FAILED', statusDetail: 'Stored file failed its integrity check. Please upload it again.' },
   284→   284→    })
   285→   285→    return
   286→   286→  }
   287→   287→
   288→   288→  // ── Stage 1: extraction (re-extract for force / first runs; reuse for partial retry)
   289→   289→  let extraction: Awaited<ReturnType<typeof extractPdfPages>>
   290→   290→  let storedPages: ExtractedPage[] | null = null
   291→   291→  if (canPartialRetry && doc.extractionJson) {
   292→   292→    const parsed = safeParse<ExtractedPage[] | null>(doc.extractionJson, null)
   293→   293→    if (parsed && parsed.length > 0) storedPages = parsed
   294→   294→  }
   295→   295→  if (storedPages) {
   296→   296→    const readable = storedPages.filter(p => !p.scanned && p.text.length > 0)
   297→   297→    extraction = {
   298→   298→      pages: storedPages,
   299→   299→      numPages: doc.pageCount ?? storedPages.length,
   300→   300→      textChars: readable.reduce((s, p) => s + p.text.length, 0),
   301→   301→      scannedPages: storedPages.filter(p => p.scanned).length,
   302→   302→      titleGuess: null,
   303→   303→    }
   304→   304→  } else {
   305→   305→    try {
   306→   306→      extraction = await extractPdfPages(new Uint8Array(buffer))
   307→   307→    } catch (err) {
   308→   308→      console.error('[riishe extract]', err)
   309→   309→      await db.studyDocument.update({
   310→   310→        where: { id: documentId },
   311→   311→        data: {
   312→   312→          status: 'FAILED',
   313→   313→          statusDetail: `This file could not be parsed as a PDF (${err instanceof Error ? err.name : 'unknown error'}). It may be corrupted or not a real PDF.`,
   314→   314→        },
   315→   315→      })
   316→   316→      return
   317→   317→    }
   318→   318→  }
   319→   319→
   320→   320→  const readablePages = extraction.pages.filter(p => !p.scanned && p.text.length > 0)
   321→   321→  await db.studyDocument.update({
   322→   322→    where: { id: documentId },
   323→   323→    data: {
   324→   324→      pageCount: extraction.numPages,
   325→   325→      extractionJson: JSON.stringify(extraction.pages),
   326→   326→      status: readablePages.length === 0 ? 'NEEDS_OCR' : 'ANALYSING',
   327→   327→      statusDetail: readablePages.length === 0
   328→   328→        ? 'No extractable text found — every page looks like a scanned image. OCR is not available in BEAN EXAMS yet, so Riishe cannot read this document. Questions cannot be generated from it.'
   329→   329→        : 'Riishe is mapping the document structure…',
   330→   330→    },
   331→   331→  })
   332→   332→
   333→   333→  if (readablePages.length === 0) return
   334→   334→
   335→   335→  // ── Identical-file reuse (same pipeline version only)
   336→   336→  if (!canPartialRetry && twinAnalysisJson) {
   337→   337→    const analysis = safeParse<StoredAnalysis | null>(twinAnalysisJson, null)
   338→   338→    if (analysis) {
   339→   339→      analysis.title = analysis.title || doc.title
   340→   340→      await db.studyDocument.update({
   341→   341→        where: { id: documentId },
   342→   342→        data: {
   343→   343→          status: 'READY',
   344→   344→          statusDetail: `Ready — analysis reused from an identical upload (${extraction.numPages} pages).`,
   345→   345→          analysisJson: JSON.stringify(analysis),
   346→   346→        },
   347→   347→      })
   348→   348→      return
   349→   349→    }
   350→   350→  }
   351→   351→
   352→   352→  // ── Stage 2: deterministic structure detection
   353→   353→  const structure = detectStructure(extraction.pages, extraction.numPages, { maxUnitChars: UNIT_CHARS })
   354→   354→  const units = structure.units
   355→   355→
   356→   356→  if (units.length === 0) {
   357→   357→    await db.studyDocument.update({
   358→   358→      where: { id: documentId },
   359→   359→      data: {
   360→   360→        status: 'NEEDS_OCR',
   361→   361→        statusDetail: 'The PDF has pages but no readable text that Riishe can analyse. OCR is not available yet.',
   362→   362→      },
   363→   363→    })
   364→   364→    return
   365→   365→  }
   366→   366→
   367→   367→  // ── Partial retry: reuse stored successful parts, re-run only failures.
   368→   368→  // Previously SKIPPED units are terminal (their refusal reason is recorded);
   369→   369→  // carrying the decision over keeps retries convergent instead of re-asking
   370→   370→  // the provider to refuse the same text again.
   371→   371→  const skipReasons = new Map<number, string>()
   372→   372→  let storedParts: (SectionAnalysis | null)[] | null = null
   373→   373→  if (canPartialRetry && stored) {
   374→   374→    const prev = stored._parts ?? []
   375→   375→    const prevOutcomes = stored.sectionOutcomes ?? []
   376→   376→    if (prev.length === units.length && prevOutcomes.length === units.length) {
   377→   377→      storedParts = prev
   378→   378→      for (let i = 0; i < prevOutcomes.length; i++) {
   379→   379→        if (prevOutcomes[i]?.status === 'SKIPPED' && prevOutcomes[i]?.error) {
   380→   380→          skipReasons.set(i, prevOutcomes[i].error as string)
   381→   381→        }
   382→   382→      }
   383→   383→    }
   384→   384→  }
   385→   385→
   386→   386→  // ── Stage 3: analyse every unit (failures recorded, never fatal)
   387→   387→  const parts: (SectionAnalysis | null)[] = new Array(units.length).fill(null)
   388→   388→  if (storedParts) for (let i = 0; i < units.length; i++) parts[i] = storedParts[i]
   389→   389→
   390→   390→  let done = 0
   391→   391→  let failedCount = 0
   392→   392→  let skippedCount = 0
   393→   393→  for (const unit of units) {
   394→   394→    if (parts[unit.index]) continue // preserved from the previous run
   395→   395→    if (skipReasons.has(unit.index)) continue // terminal skip carried over
   396→   396→    const res = await analyseSection(doc.title, unit, units.length)
   397→   397→    if (res.part) {
   398→   398→      parts[unit.index] = res.part
   399→   399→      done++
   400→   400→    } else if (res.skippedReason) {
   401→   401→      skippedCount++
   402→   402→      skipReasons.set(unit.index, res.skippedReason)
   403→   403→    } else {
   404→   404→      failedCount++
   405→   405→    }
   406→   406→    const settled = done + failedCount + skippedCount
   407→   407→    if (settled % 3 === 0 || settled === units.length) {
   408→   408→      const extras = [
   409→   409→        failedCount > 0 ? `${failedCount} failed` : null,
   410→   410→        skippedCount > 0 ? `${skippedCount} skipped` : null,
   411→   411→      ].filter(Boolean).join(', ')
   412→   412→      await db.studyDocument.update({
   413→   413→        where: { id: documentId },
   414→   414→        data: {
   415→   415→          statusDetail: extras
   416→   416→            ? `Riishe is analysing the content — section ${settled} of ${units.length} (${extras})…`
   417→   417→            : `Riishe is analysing the content — section ${settled} of ${units.length}…`,
   418→   418→        },
   419→   419→      }).catch(() => {})
   420→   420→    }
   421→   421→  }
   422→   422→
   423→   423→  const outcomes: SectionOutcome[] = units.map((u, i) => {
   424→   424→    const range = `${u.fromPage}${u.toPage !== u.fromPage ? `–${u.toPage}` : ''}`
   425→   425→    const label = u.title ? `${u.title} (${range})` : `pages ${range}`
   426→   426→    if (parts[i]) return { index: i, title: u.title, label, fromPage: u.fromPage, toPage: u.toPage, status: 'OK' as const }
   427→   427→    if (skipReasons.has(i)) return { index: i, title: u.title, label, fromPage: u.fromPage, toPage: u.toPage, status: 'SKIPPED' as const, error: skipReasons.get(i) }
   428→   428→    return { index: i, title: u.title, label, fromPage: u.fromPage, toPage: u.toPage, status: 'FAILED' as const, error: 'AI analysis failed for this section' }
   429→   429→  })
   430→   430→
   431→   431→  if (parts.every(p => p === null)) {
   432→   432→    // Nothing succeeded — honest, retryable failure (extraction is preserved).
   433→   433→    // A skip-only outcome (every unit skipped with reason) is still honest
   434→   434→    // READY-partial, not FAILED: the extraction worked, the provider refused.
   435→   435→    const allSkipped = skipReasons.size === units.length
   436→   436→    if (!allSkipped) {
   437→   437→      await db.studyDocument.update({
   438→   438→        where: { id: documentId },
   439→   439→        data: {
   440→   440→          status: 'FAILED',
   441→   441→          statusDetail: `Extraction succeeded (${readablePages.length} readable pages) but AI analysis failed for all ${units.length} section${units.length === 1 ? '' : 's'}: ${AI_UNAVAILABLE_MESSAGE}`,
   442→   442→          extractionJson: JSON.stringify(extraction.pages),
   443→   443→        },
   444→   444→      })
   445→   445→      return
   446→   446→    }
   447→   447→  }
   448→   448→
   449→   449→  // ── Stage 4: deterministic merge + verification + coverage
   450→   450→  const pageTextIndex = new Map<number, string>()
   451→   451→  for (const p of extraction.pages) {
   452→   452→    if (!p.scanned && p.text) pageTextIndex.set(p.page, p.text.toLowerCase().replace(/[^a-z0-9]+/g, ' '))
   453→   453→  }
   454→   454→
   455→   455→  const topicInputs = collectTopicInputs(parts, units, new Map(), structure.tocPages)
   456→   456→  const topics: AnalyzedTopic[] = canonicalizeTopics(topicInputs, { pageTextIndex, maxTopics: 200, maxSubtopicsPerTopic: 12 })
   457→   457→
   458→   458→  // Printed page labels: only when a consistent printed numbering was detected
   459→   459→  // (physical = printed + offset). Never invented — null otherwise.
   460→   460→  if (structure.printed.offset !== null) {
   461→   461→    const off = structure.printed.offset
   462→   462→    for (const t of topics) {
   463→   463→      const labels = t.pages.map(p => p - off).filter(n => n >= 1).map(String)
   464→   464→      if (labels.length === t.pages.length && labels.length > 0) t.printedPages = labels
   465→   465→    }
   466→   466→  }
   467→   467→
   468→   468→  const cov = buildCoverage(
   469→   469→    { numPages: extraction.numPages, scannedPages: extraction.scannedPages },
   470→   470→    readablePages,
   471→   471→    units,
   472→   472→    outcomes,
   473→   473→  )
   474→   474→
   475→   475→  const merged: StoredAnalysis = {
   476→   476→    title: doc.title,
   477→   477→    summary: '',
   478→   478→    topics,
   479→   479→    definitions: mergeDefinitions(parts, 60),
   480→   480→    keyFacts: mergeFacts(parts, 80, p => p.keyFacts ?? []),
   481→   481→    formulas: mergeFacts(parts, 50, p => p.formulas ?? []),
   482→   482→    misconceptions: mergeMisconceptions(parts, 40),
   483→   483→    keyPages: [...new Set(topics.flatMap(t => t.pages.slice(0, 3)))].filter(p => p >= 1).sort((a, b) => a - b).slice(0, 60),
   484→   484→    unreadableSections: mergeUnreadable(parts, 16),
   485→   485→    studyGaps: [],
   486→   486→    coverageNote: buildCoverageNote(cov),
   487→   487→    pipelineVersion: PIPELINE_VERSION,
   488→   488→    coverage: cov,
   489→   489→    sectionOutcomes: outcomes,
   490→   490→    structure: {
   491→   491→      kind: structure.kind,
   492→   492→      tocValidated: structure.tocValidated,
   493→   493→      tocEntries: structure.toc.length,
   494→   494→      headingCount: structure.headings.length,
   495→   495→      printedPageOffset: structure.printed.offset,
   496→   496→      printedRomanFrontMatter: structure.printed.romanFrontMatter,
   497→   497→    },
   498→   498→    _parts: parts, // retained so a partial retry keeps every successful section
   499→   499→  }
   500→   500→  merged.summary = await summarise(doc.title, extraction.pages, merged)
   501→   501→  merged.studyGaps = deriveStudyGaps(merged, cov)
   502→   502→
   503→   503→  const partial = !cov.complete
   504→   504→  await db.studyDocument.update({
   505→   505→    where: { id: documentId },
   506→   506→    data: {
   507→   507→      status: 'READY',
   508→   508→      statusDetail: partial
   509→   509→        ? `Ready (partial) — analysed ${cov.unitsOk}/${cov.unitsTotal} sections covering ${cov.analysedPages} of ${cov.readablePages} readable pages${cov.unitsSkipped > 0 ? `; ${cov.unitsSkipped} section${cov.unitsSkipped === 1 ? '' : 's'} skipped with reason` : ''}. ${cov.unitsFailed > 0 ? 'Retry analysis to cover the rest.' : ''}`.trim()
   510→   510→        : `Ready — ${merged.topics.length} topic${merged.topics.length === 1 ? '' : 's'} across ${cov.unitsTotal} section${cov.unitsTotal === 1 ? '' : 's'}; all ${cov.readablePages} readable page${cov.readablePages === 1 ? '' : 's'} accounted for${cov.unitsSkipped > 0 ? ` (${cov.unitsSkipped} skipped with reason)` : ''}.`,
   511→   511→      analysisJson: JSON.stringify(merged),
   512→   512→    },
   513→   513→  })
   514→   514→}

   518→   518→type SectionResult =
   519→   519→  | { part: SectionAnalysis; skippedReason?: undefined }
   520→   520→  | { part: null; skippedReason: string }
   521→   521→  | { part: null; skippedReason?: undefined }
   522→   522→
   523→   523→const RATE_LIMIT_BACKOFF_MS = 15_000
   524→   524→
   525→   525→/**
   526→   526→ * Analyse one unit with provider-failure strategies that keep the pipeline
   527→   527→ * honest AND convergent:
   528→   528→ *   • 429 rate limiting → ONE bounded 15 s backoff, then give up (the section
   529→   529→ *     stays FAILED and the existing partial-retry path covers it);
   530→   530→ *   • provider content filter (some legitimate dense name/term lists trip it
   531→   531→ *     and an identical retry can NEVER succeed) → split the unit into smaller
   532→   532→ *     page-range sub-chunks and analyse each independently, merging what the
   533→   533→ *     provider accepts; if every sub-chunk is refused, the unit is SKIPPED
   534→   534→ *     with a recorded reason instead of looping as FAILED forever.
   535→   535→ */
   536→   536→async function analyseSection(docTitle: string, unit: AnalysisUnit, totalUnits: number): Promise<SectionResult> {
   537→   537→  try {
   538→   538→    return { part: await analyseUnit(docTitle, unit, totalUnits) }
   539→   539→  } catch (err) {
   540→   540→    if (isRateLimitError(err)) {
   541→   541→      console.warn(`[riishe analyse unit ${unit.index + 1}/${totalUnits}] rate limited — backing off ${RATE_LIMIT_BACKOFF_MS / 1000}s once`)
   542→   542→      await new Promise(r => setTimeout(r, RATE_LIMIT_BACKOFF_MS))
   543→   543→      try {
   544→   544→        return { part: await analyseUnit(docTitle, unit, totalUnits) }
   545→   545→      } catch (err2) {
   546→   546→        if (!isContentFilterError(err2)) {
   547→   547→          console.error(`[riishe analyse unit ${unit.index + 1}/${totalUnits} ${unit.title ?? `pages ${unit.fromPage}-${unit.toPage}`}]`, err2)
   548→   548→          return { part: null }
   549→   549→        }
   550→   550→        err = err2
   551→   551→      }
   552→   552→    }
   553→   553→    if (!isContentFilterError(err)) {
   554→   554→      console.error(`[riishe analyse unit ${unit.index + 1}/${totalUnits} ${unit.title ?? `pages ${unit.fromPage}-${unit.toPage}`}]`, err)
   555→   555→      return { part: null }
   556→   556→    }
   557→   557→  }
   558→   558→
   559→   559→  // Content-filter refusal → try smaller page-range sub-chunks of this unit.
   560→   560→  const subChunks = splitUnitPages(unit)
   561→   561→  if (subChunks.length <= 1) {
   562→   562→    return {
   563→   563→      part: null,
   564→   564→      skippedReason: 'The AI provider\'s safety filter refused to process this section\'s text (dense lists of names or categories are a known trigger). Its pages are accounted as skipped, not analysed — no content was invented for them.',
   565→   565→    }
   566→   566→  }
   567→   567→  console.warn(`[riishe analyse unit ${unit.index + 1}/${totalUnits}] content-filtered — retrying as ${subChunks.length} sub-chunks`)
   568→   568→  const mergedParts: SectionAnalysis[] = []
   569→   569→  for (const [i, chunk] of subChunks.entries()) {
   570→   570→    try {
   571→   571→      mergedParts.push(await analyseUnit(docTitle, chunk, totalUnits))
   572→   572→    } catch (chunkErr) {
   573→   573→      if (!isContentFilterError(chunkErr)) {
   574→   574→        console.error(`[riishe analyse unit ${unit.index + 1}/${totalUnits} sub-chunk ${i + 1}/${subChunks.length}]`, chunkErr)
   575→   575→      }
   576→   576→      // filter-refused sub-chunks are dropped; remaining chunks still count
   577→   577→    }
   578→   578→  }
   579→   579→  if (mergedParts.length === 0) {
   580→   580→    return {
   581→   581→      part: null,
   582→   582→      skippedReason: `The AI provider's safety filter refused this section's text in full and in ${subChunks.length} smaller parts (dense lists of names or categories are a known trigger). Its pages are accounted as skipped, not analysed — no content was invented for them.`,
   583→   583→    }
   584→   584→  }
   585→   585→  return { part: mergeSectionAnalyses(mergedParts) }
   586→   586→}
   587→   587→
   588→   588→/** Split a unit into 2–4 page-range sub-chunks (halves, quarters for 8+ pages). */
   589→   589→export function splitUnitPages(unit: AnalysisUnit): AnalysisUnit[] {
   590→   590→  const pageCount = unit.toPage - unit.fromPage + 1
   591→   591→  if (pageCount < 2) return [unit]
   592→   592→  const parts = pageCount >= 8 ? 4 : 2
   593→   593→  const size = Math.ceil(pageCount / parts)
   594→   594→  const out: AnalysisUnit[] = []
   595→   595→  for (let from = unit.fromPage; from <= unit.toPage; from += size) {
   596→   596→    const to = Math.min(from + size - 1, unit.toPage)
   597→   597→    const subText = unit.text
   598→   598→      .split(/\n\n+/)
   599→   599→      .filter(block => {
   600→   600→        const m = /^\[page (\d+)\]/.exec(block)
   601→   601→        return m ? Number(m[1]) >= from && Number(m[1]) <= to : true
   602→   602→      })
   603→   603→      .join('\n\n')
   604→   604→    out.push({ ...unit, fromPage: from, toPage: to, text: subText, chars: subText.length })
   605→   605→  }
   606→   606→  return out.length > 0 ? out : [unit]
   607→   607→}

   609→   609→export function mergeSectionAnalyses(parts: SectionAnalysis[]): SectionAnalysis {

   611→   611→    topics: parts.flatMap(p => p.topics ?? []).slice(0, 12),
   612→   612→    definitions: parts.flatMap(p => p.definitions ?? []).slice(0, 20),
   613→   613→    keyFacts: parts.flatMap(p => p.keyFacts ?? []).slice(0, 20),
   614→   614→    formulas: parts.flatMap(p => p.formulas ?? []).slice(0, 10),
   615→   615→    misconceptions: parts.flatMap(p => p.misconceptions ?? []).slice(0, 10),
   616→   616→    unreadable: parts.flatMap(p => p.unreadable ?? []).slice(0, 10),
   617→   617→  }

   620→   620→function unitDescriptor(unit: AnalysisUnit): string {
   621→   621→  const range = unit.fromPage === unit.toPage ? `page ${unit.fromPage}` : `pages ${unit.fromPage}–${unit.toPage}`
   622→   622→  if (unit.kind === 'SECTION' && unit.title) return `the section "${unit.title}" (${range} of the uploaded PDF)`
   623→   623→  if (unit.kind === 'FRONT') return `the front matter (${range} of the uploaded PDF)`
   624→   624→  if (unit.kind === 'OVERFLOW') return `an excerpt of ${range} of the uploaded PDF`
   625→   625→  return `${range} of the uploaded PDF`
   626→   626→}

   628→   628→async function analyseUnit(docTitle: string, unit: AnalysisUnit, totalUnits: number): Promise<SectionAnalysis> {
   629→   629→  return chatJson<SectionAnalysis>({
   630→   630→    system: SYSTEM,
   631→   631→    user: `Analyse ${unitDescriptor(unit)} — part ${unit.index + 1} of ${totalUnits} covering the whole document ("${docTitle}"). Extract ONLY what is present in the text below.
   632→   632→
   633→   633→Return JSON with this exact shape:
   634→   634→{
   635→   635→  "topics": [{"name": string, "subtopics": string[], "pages": number[]}],
   636→   636→  "definitions": [{"term": string, "meaning": string, "page": number}],
   637→   637→  "keyFacts": string[],
   638→   638→  "formulas": string[],
   639→   639→  "misconceptions": [{"misconception": string, "correction": string, "page": number}],
   640→   640→  "unreadable": string[]
   641→   641→}
   642→   642→
   643→   643→Rules:
   644→   644→- "topics": the concrete topics/concepts this text actually covers, using the document's own headings and terminology (3-8 at most; fewer is fine). Topic names must be SHORT noun phrases (≤ 8 words) — never full sentences. Do NOT invent topics, do NOT echo the section title unless the text genuinely develops it, and do NOT promote every heading you see.
   645→   645→- "subtopics": concepts or key terms under each topic found in this text.
   646→   646→- "pages" must cite page numbers from the [page N] markers in the text — never other numbers.
   647→   647→- If this text is from a question paper, answer key or marks scheme, extract the subject areas the questions test (not the questions themselves) and note it in keyFacts.
   648→   648→- "unreadable": list page labels whose text is garbled, truncated or incomprehensible ([] if none).
   649→   649→- "misconceptions": only if the text supports a correction (common wrong belief vs what the text says).
   650→   650→- "formulas": mathematical/scientific formulas or rules stated in the text ([] if none).
   651→   651→
   652→   652→${fenceDocumentText(docTitle, unit.text)}`,
   653→   653→  })
   654→   654→}

   656→   656→async function summarise(title: string, pages: ExtractedPage[], merged: StoredAnalysis): Promise<string> {
   657→   657→  // Grounded summary from the first readable pages + identified topic names
   658→   658→  const head = pages
   659→   659→    .filter(p => !p.scanned && p.text)
   660→   660→    .slice(0, 3)
   661→   661→    .map(p => `[page ${p.page}]\n${p.text}`)
   662→   662→    .join('\n\n')
   663→   663→    .slice(0, 3000)
   664→   664→  try {
   665→   665→    const res = await chatJson<{ summary: string }>({
   666→   666→      system: SYSTEM,
   667→   667→      user: `Write a 2–3 sentence summary of this study document ("${title}") for a student, strictly from the text. Return JSON: {"summary": string}
   668→   668→
   669→   669→Known topics identified: ${merged.topics.slice(0, 12).map(t => t.name).join(', ') || '(none)'}.
   670→   670→
   671→   671→${fenceDocumentText(title, head)}`,
   672→   672→    })
   673→   673→    return String(res.summary ?? '').slice(0, 800)
   674→   674→  } catch {
   675→   675→    // Summary is cosmetic — fall back to a factual non-AI line rather than failing the job
   676→   676→    return `Study document with ${merged.topics.length} identified topic${merged.topics.length === 1 ? '' : 's'}.`
   677→   677→  }
   678→   678→}

   680→   680→function deriveStudyGaps(a: DocumentAnalysis, cov: CoverageMetrics): string[] {
   681→   681→  const gaps: string[] = []
   682→   682→  if (a.topics.length === 0) gaps.push('No clear topics were identified — the text may be too sparse for question generation.')
   683→   683→  if (a.definitions.length === 0) gaps.push('No key definitions were found — you may need supporting material for terminology-heavy questions.')
   684→   684→  if (a.formulas.length === 0) gaps.push('No formulas or rules were detected — calculation-based questions will not be generated.')
   685→   685→  if (a.unreadableSections.length > 0) gaps.push(`${a.unreadableSections.length} section${a.unreadableSections.length === 1 ? '' : 's'} were unreadable and excluded from the analysis.`)
   686→   686→  if (cov.scannedPages > 0) gaps.push(`${cov.scannedPages} image-only page${cov.scannedPages === 1 ? ' needs' : 's need'} OCR to be analysed — OCR is not available yet.`)
   687→   687→  if (cov.unitsFailed > 0) gaps.push(`${cov.unanalysedPages} readable page${cov.unanalysedPages === 1 ? ' is' : 's are'} not yet covered (failed sections) — retry the analysis to include them.`)
   688→   688→  if (cov.unitsSkipped > 0) gaps.push(`${cov.unitsSkipped} section${cov.unitsSkipped === 1 ? '' : 's'} (${cov.skippedPages} page${cov.skippedPages === 1 ? '' : 's'}) were skipped because the AI provider's safety filter refused their text — see the section breakdown for the exact reason.`)
   689→   689→  return gaps
   690→   690→}
