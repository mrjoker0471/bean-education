    82→function formatBytes(n: number): string {
    83→  if (!Number.isFinite(n) || n <= 0) return '0 B'
    84→  if (n < 1024) return `${n} B`
    85→  if (n < 1024 * 1024) return `${(n / 1024).toFixed(n < 10 * 1024 ? 1 : 0)} KB`
    86→  return `${(n / (1024 * 1024)).toFixed(1)} MB`
    87→}
    88→
    89→function formatDate(iso: string): string {
    90→  const d = new Date(iso)
    91→  if (Number.isNaN(d.getTime())) return iso
    92→  return d.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })
    93→}

    95→/** Same scaling the backend uses for the distribution preview. */
    96→function distributionLine(count: number): string {
    97→  const scale = count / 50
    98→  const d = {
    99→    f: Math.max(1, Math.round(10 * scale)),
   100→    a: Math.max(1, Math.round(15 * scale)),
   101→    s: Math.max(1, Math.round(10 * scale)),
   102→    m: Math.max(1, Math.round(10 * scale)),
   103→    x: Math.max(1, Math.round(5 * scale)),
   104→  }
   105→  return `${d.f} foundational · ${d.a} application · ${d.s} statement · ${d.m} multi-step · ${d.x} misconception`
   106→}

   368→type WorkspaceTab = 'material' | 'analysis' | 'tests'
   369→
   370→function RiisheWorkspace({ docId }: { docId: string }) {
   371→  const guestId = useAppStore(s => s.guestId)
   372→  const navigate = useAppStore(s => s.navigate)
   373→
   374→  const [reloadTick, setReloadTick] = useState(0)
   375→  const [loaded, setLoaded] = useState<{ key: string; data: MaterialDetailDTO } | null>(null)
   376→  const [failure, setFailure] = useState<{ key: string; message: string } | null>(null)
   377→
   378→  const loadKey = guestId ? `${guestId}|${docId}|${reloadTick}` : 'no-session'
   379→
   380→  useEffect(() => {
   381→    if (!guestId) return
   382→    let alive = true
   383→    api
   384→      .material(docId, guestId)
   385→      .then(d => {
   386→        if (!alive) return
   387→        setLoaded({ key: loadKey, data: d })
   388→        setFailure(null)
   389→      })
   390→      .catch((e: unknown) => {
   391→        if (!alive) return
   392→        setFailure({ key: loadKey, message: e instanceof Error ? e.message : 'Could not load this material.' })
   393→      })
   394→    return () => {
   395→      alive = false
   396→    }
   397→  }, [guestId, docId, loadKey])
   398→
   399→  const doc = loaded && loaded.key === loadKey ? loaded.data : null
   400→  const error = failure && failure.key === loadKey ? failure.message : null
   401→  const loading = !doc && !error
   402→  const notFound = !!error && /not found/i.test(error)
   403→
   404→  // Poll the material every 5s ONLY while it (or one of its tests) is in flight.
   405→  const shouldPoll =
   406→    !!doc && (isProcessing(doc.status) || doc.tests.some(t => t.status === 'QUEUED' || t.status === 'GENERATING'))
   407→  useEffect(() => {
   408→    if (!shouldPoll || !guestId || !docId) return
   409→    const timer = setInterval(() => {
   410→      api
   411→        .material(docId, guestId)
   412→        .then(d => setLoaded({ key: loadKey, data: d }))
   413→        .catch(() => {
   414→          /* keep the last honest state; the next tick retries */
   415→        })
   416→    }, 5000)
   417→    return () => clearInterval(timer)
   418→  }, [shouldPoll, guestId, docId, loadKey])
   419→
   420→  // Default tab: land on Tests when the document already has a READY test to take.
   421→  const [tab, setTab] = useState<WorkspaceTab>('material')
   422→  const docKey = loaded?.key ?? null
   423→  const [appliedKey, setAppliedKey] = useState<string | null>(null)
   424→  if (doc && docKey && docKey !== appliedKey) {
   425→    setAppliedKey(docKey)
   426→    setTab(doc.status === 'READY' && doc.tests.some(t => t.status === 'READY') ? 'tests' : 'material')
   427→  }
   428→
   429→  const [renameOpen, setRenameOpen] = useState(false)
   430→  const [renameValue, setRenameValue] = useState('')
   431→  const [renaming, setRenaming] = useState(false)
   432→
   433→  // Integrated reader overlay — lifted to the workspace so page chips in the
   434→  // Analysis tab can deep-link to the exact page, and "Open in reader" in the
   435→  // Material tab opens without one.
   436→  const [reader, setReader] = useState<{ startPage?: number } | null>(null)
   437→  const openReaderAt = (page?: number) => setReader({ startPage: page })
   438→
   439→  const refresh = () => setReloadTick(t => t + 1)
   440→
   441→  // ── Not found ──
   442→  if (notFound) {
   443→    return (
   444→      <div className="mx-auto w-full max-w-3xl px-4 py-16 sm:px-6 lg:px-8">
   445→        <EmptyState
   446→          icon={FileWarning}
   447→          title="Material not found"
   448→          description="This material does not exist in your library — it may have been removed from this device, or the link is stale."
   449→          actionLabel="Back to My Materials"
   450→          onAction={() => navigate({ view: 'materials' })}
   451→        />
   452→      </div>
   453→    )
   454→  }
   455→
   456→  // ── Load error ──
   457→  if (!loading && error && doc === null) {
   458→    return (
   459→      <div className="mx-auto w-full max-w-3xl px-4 py-16 sm:px-6 lg:px-8">
   460→        <div role="alert" className="rounded-xl border border-destructive/40 bg-destructive/10 p-6">
   461→          <p className="font-display text-lg font-semibold text-destructive">Could not load this material</p>
   462→          <p className="mt-2 text-sm text-destructive/90">{error}</p>
   463→          <div className="mt-5 flex flex-wrap gap-3">
   464→            <Button variant="outline" className="border-gold/40 text-gold hover:bg-gold/10" onClick={refresh}>
   465→              <RotateCcw className="mr-1.5 h-4 w-4" /> Try again
   466→            </Button>
   467→            <Button
   468→              variant="outline"
   469→              className="border-border text-muted-foreground hover:bg-elevated"
   470→              onClick={() => navigate({ view: 'materials' })}
   471→            >
   472→              Back to My Materials
   473→            </Button>
   474→          </div>
   475→        </div>
   476→      </div>
   477→    )
   478→  }
   479→
   480→  // ── Loading ──
   481→  if (loading || !doc) {
   482→    return (
   483→      <div className="mx-auto w-full max-w-5xl px-4 py-10 sm:px-6 lg:px-8" aria-hidden>
   484→        <Skeleton className="h-8 w-40 bg-elevated" />
   485→        <Skeleton className="mt-4 h-10 w-2/3 bg-elevated" />
   486→        <Skeleton className="mt-3 h-6 w-1/2 bg-elevated" />
   487→        <Skeleton className="mt-8 h-10 bg-elevated" />
   488→        <Skeleton className="mt-4 h-96 bg-elevated" />
   489→      </div>
   490→    )
   491→  }
   492→
   493→  const hasAnalysis = doc.status === 'READY' && !!doc.analysis
   494→  const metaLabels = [doc.examLabel, doc.subjectLabel, doc.topicLabel].filter((v): v is string => !!v)
   495→
   496→  const doRename = () => {
   497→    const next = renameValue.trim()
   498→    if (!next || renaming) return
   499→    setRenaming(true)
   500→    api
   501→      .renameMaterial(doc.id, guestId, { title: next })

   826→function AnalysisPanel({ analysis, onOpenPage, onReanalyse }: {
   827→  analysis: DocumentAnalysisDTO
   828→  onOpenPage: (page: number) => void
   829→  onReanalyse: (force: boolean) => void
   830→}) {

   834→
   835→  // Hierarchy: level-1 sections carry their level-2 children; children whose
   836→  // parent vanished (or old flat analyses) render at top level.
   837→  const tops = analysis.topics.filter(t => t.level !== 2 || !t.parentId || !analysis.topics.some(p => p.id === t.parentId))
   838→  const childrenOf = (id: string) => analysis.topics.filter(t => t.level === 2 && t.parentId === id)
   839→
   840→  return (
   841→    <StructureContext.Provider value={structure}>
   842→      <div className="space-y-6">
   843→      {/* Golden rule */}
   844→      <div className="flex items-start gap-3 rounded-lg border border-gold/40 bg-gold/10 px-4 py-3">

   846→        <p className="text-sm leading-relaxed text-foreground">
   847→          <span className="font-semibold text-gold">The golden rule:</span> every question is generated FROM this
   848→          material — answers and page references point back to the document itself, never to outside knowledge.
   849→        </p>
   850→      </div>
   851→
   852→      {/* Summary */}
   853→      <section aria-label="Summary" className="rounded-xl border border-border bg-card p-5 sm:p-6">
   854→        <h2 className="font-display text-xl font-semibold text-foreground">Summary</h2>

   945→      {/* Misconceptions */}

  1243→  const [timed, setTimed] = useState(true)
  1244→  const [count, setCount] = useState<50 | 30 | 20>(50)
  1245→  const [generating, setGenerating] = useState(false)
  1246→
  1247→  // Active test detail (fetched + polled separately from the material)
  1248→  const [activeTestId, setActiveTestId] = useState<string | null>(null)
  1249→  const [testLoaded, setTestLoaded] = useState<{ key: string; data: RiisheTestDetailDTO } | null>(null)
  1250→  const [testError, setTestError] = useState<string | null>(null)
  1251→  const [startingId, setStartingId] = useState<string | null>(null)
  1252→
  1253→  const sortedTests = [...doc.tests].sort((a, b) => b.createdAt.localeCompare(a.createdAt))
  1254→
  1255→  // Auto-pick: an in-flight test first, otherwise the newest READY test.
  1256→  const inflight = doc.tests.find(t => t.status === 'QUEUED' || t.status === 'GENERATING')
  1257→  const latestReady = sortedTests.find(t => t.status === 'READY')
  1258→  const activeId = activeTestId ?? inflight?.id ?? latestReady?.id ?? null
  1259→  const activeTest = testLoaded && testLoaded.key === activeId ? testLoaded.data : null

  1261→  // Fetch + poll the active test every 4s until it settles (READY / FAILED).

  1263→    if (!guestId || !activeId) return

  1267→      api

  1270→          if (!alive) return
  1271→          setTestLoaded({ key: activeId, data: d })
  1272→          setTestError(null)

  1277→        })

  1280→          setTestError(e instanceof Error ? e.message : 'Could not load the test status.')
  1281→        })
  1282→    }
  1283→    fetchOnce()
  1284→    timer = setInterval(fetchOnce, 4000)
  1285→    return () => {
  1286→      alive = false
  1287→      if (timer) clearInterval(timer)

  1289→  }, [guestId, activeId])
  1290→
  1291→  const toggleFocus = (name: string) => {

  1293→      prev.includes(name) ? prev.filter(n => n !== name) : prev.length >= 3 ? prev : [...prev, name],
  1294→    )
  1295→  }
  1296→

  1298→    if (generating || doc.status !== 'READY') return

  1300→    setTestError(null)

  1302→      .generateRiisheTest({
  1303→        guestId,
  1304→        documentId: doc.id,

  1358→          <Sparkles className="h-4 w-4 text-gold" aria-hidden /> Generate a test
