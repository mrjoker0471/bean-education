    61→/** New tabs with graceful handling of legacy deep links. */
    62→function normalizeTab(tab: string | undefined): TabId {
    63→  switch (tab) {
    64→    case 'study': case 'intelligence': case 'practice':
    65→      return tab
    66→    case 'overview': case 'syllabus': case 'materials':
    67→      return 'study'
    68→    case 'tests': case 'practice-exam':
    69→      return 'practice'
    70→    case 'bean':
    71→      return 'intelligence'
    72→    default:
    73→      return 'study'
    74→  }
    75→}

    77→export function ExamView({ slug, initialTab }: { slug: string; initialTab?: string }) {
    78→  const navigate = useAppStore(s => s.navigate)
    79→  const guestId = useAppStore(s => s.guestId)
    80→
    81→  const [reloadToken, setReloadToken] = useState(0)
    82→  const [tab, setTab] = useState<TabId>(normalizeTab(initialTab))
    83→  const [pendingTopicId, setPendingTopicId] = useState<string | null>(null)
    84→  // The selected topic shared across the three layers (Study ⇄ Intelligence ⇄ Practice)
    85→  const [focusTopic, setFocusTopic] = useState<FocusTopic | null>(null)
    86→
    87→  // Keyed load state — each slug+retry combination has its own slot, so loading/error are
    88→  // derived during render (no cascading setState inside effects).
    89→  const loadKey = `${slug}|${reloadToken}`
    90→  const [loaded, setLoaded] = useState<{ key: string; data: ExamDetailDTO } | null>(null)
    91→  const [failure, setFailure] = useState<{ key: string; message: string } | null>(null)
    92→
    93→  // Load the exam
    94→  useEffect(() => {
    95→    let alive = true
    96→    api.exam(slug)
    97→      .then(d => {
    98→        if (!alive) return
    99→        setLoaded({ key: loadKey, data: d })
   100→        setFailure(null)
   101→        // Carry exam context into BEAN Intelligence (#/bean inherits it)
   102→        useAppStore.getState().setBeanContext({ examSlug: slug, examName: d.exam.name })
   103→      })
   104→      .catch(e => {
   105→        if (alive) setFailure({ key: loadKey, message: e.message })
   106→      })
   107→    return () => {
   108→      alive = false
   109→    }
   110→  }, [slug, loadKey])
   111→
   112→  // Keep the active tab in sync with deep links (browser back/forward).
   113→  // Guarded render-time adjustment — the pattern documented at react.dev/learn/you-might-not-need-an-effect.
   114→  const [lastInitialTab, setLastInitialTab] = useState(initialTab)
   115→  if (lastInitialTab !== initialTab) {
   116→    setLastInitialTab(initialTab)
   117→    setTab(normalizeTab(initialTab))
   118→  }
   119→
   120→  const data = loaded && loaded.key === loadKey ? loaded.data : null
   121→  const error = failure && failure.key === loadKey ? failure.message : null
   122→  const loading = !data && !error
   123→
   124→  const changeTab = (value: string) => {
   125→    const next = normalizeTab(value)
   126→    setTab(next)
   127→    navigate({ view: 'exam', slug, tab: next === 'study' ? undefined : next })
   128→  }
   129→
   130→  /** Layer-crossing: focus a topic and hand its full context to BEAN Intelligence. */
   131→  const focusOnTopic = (t: FocusTopic, noteTitle?: string) => {
   132→    setFocusTopic(t)
   133→    useAppStore.getState().setBeanContext({
   134→      examSlug: slug,
   135→      examName: data?.exam.name,
   136→      stageName: t.stageName,
   137→      subjectName: t.subjectName,
   138→      topicId: t.id,
   139→      topicName: t.name,
   140→      noteTitle,
   141→    })
   142→    changeTab('intelligence')
   143→  }
   144→
   145→  const startTopicTest = (topic: TopicNode) => {
   146→    if (!guestId) {
   147→      toast({
   148→        title: 'Preparing your session',
   149→        description: 'Your guest profile is still initialising — try again in a moment.',
   150→        variant: 'destructive',
   151→      })
   152→      return
   153→    }
   154→    if (pendingTopicId) return
   155→    setPendingTopicId(topic.id)
   156→    api.startTest({ guestId, topicId: topic.id })
   157→      .then(r => navigate({ view: 'test', attemptId: r.attemptId }))
   158→      .catch(e => {
   159→        toast({ title: 'Could not start the practice test', description: e.message, variant: 'destructive' })
   160→      })
   161→      .finally(() => setPendingTopicId(null))
   162→  }
   163→
   164→  // ─── Loading ───
   165→  if (loading) return <ExamSkeleton />
   166→
   167→  // ─── Not found / errors ───
   168→  if (error) {
   169→    if (/not found/i.test(error)) {
   170→      return (
   171→        <div className="mx-auto w-full max-w-3xl px-4 py-16 sm:px-6 lg:px-8">
   172→          <EmptyState
   173→            icon={Compass}
   174→            title="Examination not found"
   175→            description={`No examination lives at “${slug}” in the BEAN EXAMS catalogue. It may have been renamed — browse the catalogue to find what you need.`}
   176→            actionLabel="Back to catalogue"
   177→            onAction={() => navigate({ view: 'catalog' })}
   178→          />
   179→        </div>
   180→      )
   181→    }
   182→    return (
   183→      <div className="mx-auto w-full max-w-3xl px-4 py-16 sm:px-6 lg:px-8">
   184→        <div role="alert" className="rounded-xl border border-destructive/40 bg-destructive/10 p-6">
   185→          <p className="font-display text-lg font-semibold text-destructive">Could not load this examination</p>
   186→          <p className="mt-2 text-sm text-destructive/90">{error}</p>
   187→          <div className="mt-5 flex flex-wrap gap-3">
   188→            <Button variant="outline" className="border-gold/40 text-gold hover:bg-gold/10" onClick={() => setReloadToken(t => t + 1)}>
   189→              <RotateCcw className="mr-1.5 h-4 w-4" /> Try again
   190→            </Button>
   191→            <Button variant="outline" className="border-border text-muted-foreground hover:bg-elevated" onClick={() => navigate({ view: 'catalog' })}>
   192→              Back to catalogue
   193→            </Button>
   194→          </div>
   195→        </div>
   196→      </div>
   197→    )
   198→  }
   199→
   200→  if (!data) return null
   201→  const { exam } = data
   202→  const officialDocs = data.examResources.filter(isOfficialDoc)
   203→  // Subject-wise syllabus PDFs (37 for CUET UG) get their own compact section
   204→  const primaryDocs = officialDocs.filter(r => r.type !== 'OFFICIAL_SUBJECT_SYLLABUS')
   205→  const subjectSyllabusDocs = officialDocs.filter(r => r.type === 'OFFICIAL_SUBJECT_SYLLABUS')
   206→  const otherExamResources = data.examResources.filter(r => !isOfficialDoc(r))
   207→
   208→  return (
   209→    <div className="mx-auto w-full max-w-6xl px-4 py-8 sm:px-6 sm:py-10 lg:px-8">
   210→      {/* ─── Header ─── */}
   211→      <header>
   212→        {/* Breadcrumbs */}
   213→        <nav aria-label="Breadcrumb" className="flex flex-wrap items-center gap-x-1 gap-y-1 text-xs text-muted-foreground sm:text-sm">
   214→          <button type="button" onClick={() => navigate({ view: 'home' })} className="transition-colors hover:text-gold">
   215→            Home
   216→          </button>
   217→          <ChevronRight className="h-3.5 w-3.5 text-muted-foreground/60" aria-hidden />
   218→          <button type="button" onClick={() => navigate({ view: 'catalog' })} className="transition-colors hover:text-gold">
   219→            All examinations
   220→          </button>
   221→          {exam.categoryName && (
   222→            <>
   223→              <ChevronRight className="h-3.5 w-3.5 text-muted-foreground/60" aria-hidden />
   224→              <button
   225→                type="button"
   226→                onClick={() => navigate({ view: 'catalog', category: exam.categorySlug })}
   227→                className="transition-colors hover:text-gold"
   228→              >
   229→                {exam.categoryName}
   230→              </button>
   231→            </>
   232→          )}
   233→          <ChevronRight className="h-3.5 w-3.5 text-muted-foreground/60" aria-hidden />
   234→          <span aria-current="page" className="font-medium text-foreground">{exam.shortName ?? exam.name}</span>
   235→        </nav>
   236→
   237→        <button
   238→          type="button"
   239→          onClick={() => navigate({ view: 'catalog' })}
   240→          className="mt-4 inline-flex items-center gap-1.5 text-sm text-muted-foreground transition-colors hover:text-gold"
   241→        >
   242→          <ArrowLeft className="h-4 w-4" aria-hidden /> All examinations
   243→        </button>
   244→
   245→        <div className="mt-5 flex flex-wrap items-center gap-2">
   246→          <span className="rounded-full border border-gold/30 bg-gold/5 px-3 py-1 text-xs font-medium text-gold">
   247→            {exam.shortName ?? 'Examination'}
   248→          </span>
   249→          <SyllabusStatusBadge status={exam.syllabusStatus} />
   250→        </div>
   251→
   252→        <h1 className="mt-3 font-display text-2xl font-bold leading-tight text-foreground sm:text-3xl lg:text-4xl">
   253→          {exam.name}
   254→        </h1>
   255→
   256→        <p className="mt-2 flex flex-wrap items-center gap-x-4 gap-y-1 text-sm text-muted-foreground">
   257→          <span>Conducted by {exam.authority}</span>
   258→          {exam.state && <span>{exam.state}</span>}
   259→        </p>
   260→
   261→        <p className="mt-4 max-w-3xl text-sm leading-relaxed text-muted-foreground sm:text-base">{exam.description}</p>
   262→
   263→        {/* Exam pattern summary — always visible above the three learning tabs */}
   264→        <PatternSummaryStrip stages={data.stages} />
   265→
   266→        {/* Honesty feature: preparation coverage */}
   267→        {exam.contentNote && (
   268→          <div className="mt-5 max-w-3xl rounded-xl border border-gold/30 bg-gold/5 p-4 sm:p-5" role="note">
   269→            <p className="flex items-center gap-2 text-sm font-semibold text-gold">
   270→              <Info className="h-4 w-4 shrink-0" aria-hidden /> Honest coverage status
   271→            </p>
   272→            <p className="mt-2 text-sm leading-relaxed text-foreground/90">{exam.contentNote}</p>
   273→          </div>
   274→        )}
   275→
   276→        {/* Official source links */}
   277→        {exam.sourceRefs && exam.sourceRefs.length > 0 && (
   278→          <div className="mt-4 flex flex-wrap items-center gap-2">
   279→            <span className="text-xs uppercase tracking-[0.14em] text-muted-foreground">Official sources</span>
   280→            {exam.sourceRefs.map(ref => (
   281→              <a
   282→                key={ref.url}
   283→                href={ref.url}
   284→                target="_blank"
   285→                rel="noopener noreferrer"
   286→                className="inline-flex items-center gap-1 rounded-full border border-border bg-card px-3 py-1 text-xs text-muted-foreground transition-colors hover:border-gold/40 hover:text-gold"
   287→              >
   288→                {ref.label} <ExternalLink className="h-3 w-3" aria-hidden />
   289→              </a>
   290→            ))}
   291→          </div>
   292→        )}
   293→
   294→        <div className="mt-5 flex flex-wrap gap-2">
   295→          <CountPill label="topics" value={exam.counts.topics} />
   296→          <CountPill label="notes" value={exam.counts.notes} />
   297→          <CountPill label="questions" value={exam.counts.questions} />
   298→          <CountPill label="tests" value={exam.counts.tests} />
   299→        </div>
   300→      </header>
   301→
   302→      {/* ─── THE THREE LEARNING LAYERS ─── */}
   303→      <div
   304→        role="tablist"
   305→        aria-label="Learning layers"
   306→        className="mt-8 grid grid-cols-1 gap-2 sm:grid-cols-3 sm:gap-3"
   307→      >
   308→        {TABS.map(t => {
   309→          const active = tab === t.id
   310→          return (
   311→            <button
   312→              key={t.id}
   313→              type="button"
   314→              role="tab"
   315→              id={`exam-tab-${t.id}`}
   316→              aria-selected={active}
   317→              aria-controls={`exam-panel-${t.id}`}
   318→              onClick={() => changeTab(t.id)}
   319→              className={cn(
   320→                'group flex items-center gap-3 rounded-xl border px-4 py-3 text-left transition-colors',
   321→                active
   322→                  ? 'border-gold bg-gold/10 shadow-[0_0_0_1px_rgba(214,165,20,0.35)]'
   323→                  : 'border-border bg-card hover:border-gold/40 hover:bg-elevated'
   324→              )}
   325→            >
   326→              <span
   327→                className={cn(
   328→                  'flex h-9 w-9 shrink-0 items-center justify-center rounded-lg border',
   329→                  active ? 'border-gold/50 bg-gold/15' : 'border-border bg-elevated'
   330→                )}
   331→              >
   332→                <t.icon className={cn('h-4 w-4', active ? 'text-gold' : 'text-muted-foreground')} aria-hidden />
   333→              </span>
   334→              <span className="min-w-0">
   335→                <span className={cn('block text-sm font-semibold sm:text-base', active ? 'text-gold' : 'text-foreground')}>
   336→                  {t.label}
   337→                </span>
   338→                <span className="block truncate text-[11px] text-muted-foreground sm:text-xs">{t.sub}</span>
   339→              </span>
   340→            </button>
   341→          )
   342→        })}
   343→      </div>
   344→
   345→      <div
   346→        role="tabpanel"
   347→        id={`exam-panel-${tab}`}
   348→        aria-labelledby={`exam-tab-${tab}`}
   349→        className="mt-8 focus-visible:outline-none"
   350→      >
   351→        {tab === 'study' && (
   352→          <StudyTab
   353→            data={data}
   354→            officialDocs={primaryDocs}
   355→            subjectSyllabusDocs={subjectSyllabusDocs}
   356→            otherExamResources={otherExamResources}
   357→            pendingTopicId={pendingTopicId}
   358→            onStartTopicTest={startTopicTest}
   359→            onFocusTopic={focusOnTopic}
   360→          />
   361→        )}
   362→        {tab === 'intelligence' && (
   363→          <IntelligenceTab
   364→            data={data}
   365→            slug={slug}
   366→            focusTopic={focusTopic}
   367→            onFocusTopic={focusOnTopic}
   368→            pendingTopicId={pendingTopicId}
   369→            onStartTopicTest={startTopicTest}
   370→            onGoPractice={() => changeTab('practice')}
   371→          />
   372→        )}
   373→        {tab === 'practice' && (
   374→          <PracticeTab
   375→            data={data}
   376→            guestId={guestId}
   377→            slug={slug}
   378→            focusTopic={focusTopic}
   379→          />
   380→        )}
   381→      </div>
   382→    </div>
   383→  )
   384→}

   387→function StudyTab({
   388→  data,
   389→  officialDocs,
   390→  subjectSyllabusDocs,
   391→  otherExamResources,
   392→  pendingTopicId,
   393→  onStartTopicTest,

   397→  officialDocs: ResourceDTO[]
   398→  subjectSyllabusDocs: ResourceDTO[]
   399→  otherExamResources: ResourceDTO[]

   402→  onFocusTopic: (t: FocusTopic, noteTitle?: string) => void
   403→}) {

   439→            />
   440→          </div>
   441→        )}

   450→                {subjectSyllabusDocs.length} official document{subjectSyllabusDocs.length === 1 ? '' : 's'} — direct NTA downloads
   451→              </span>
   452→            </div>
   453→            <p className="mt-1 max-w-3xl text-sm text-muted-foreground">
   454→              One official syllabus PDF per subject, exactly as published by {data.exam.authority} for the current cycle.
   455→              Each card offers a direct download — verified as a real PDF file, never a disguised webpage.
   456→            </p>

   953→// ─── Revision section (lazy check against this guest's wrong answers) ────────
   954→function RevisionSection({
   955→  examId,
   956→  examName,
   957→  guestId,
   958→  pending,
   959→  onStart,
   960→}: {
   961→  examId: string
   962→  examName: string
   963→  guestId: string
   964→  pending: boolean
   965→  onStart: () => void
   966→}) {
   967→  const loadKey = `${examId}|${guestId}`
   968→  const [loaded, setLoaded] = useState<{ key: string; count: number; ready: boolean } | null>(null)
   969→  const [failed, setFailed] = useState<{ key: string; message: string } | null>(null)
   970→  const [retry, setRetry] = useState(0)
   971→
   972→  const data = loaded && loaded.key === `${loadKey}|${retry}` ? loaded : null
   973→  const failure = failed && failed.key === `${loadKey}|${retry}` ? failed.message : null
   974→
   975→  useEffect(() => {
   976→    if (!guestId) return
   977→    let alive = true
   978→    api.revisionCheck(examId, guestId)
   979→      .then(r => { if (alive) setLoaded({ key: `${loadKey}|${retry}`, count: r.count, ready: r.ready }) })
   980→      .catch(e => { if (alive) setFailed({ key: `${loadKey}|${retry}`, message: e.message }) })
   981→    return () => { alive = false }
   982→  }, [loadKey, retry, guestId, examId])
   983→
   984→  return (
   985→    <section aria-labelledby="revision-heading">
   986→      <h2 id="revision-heading" className="font-display text-lg font-semibold text-foreground sm:text-xl">
   987→        Revision test — built from your mistakes
   988→      </h2>
   989→      <Card className="mt-4 border-gold/25 bg-gradient-to-b from-gold/10 to-transparent">
   990→        <CardContent className="p-4 sm:p-5">
   991→          {!guestId || (!data && !failure) ? (
   992→            <div className="space-y-2" aria-busy="true" aria-label="Checking revision availability">
   993→              <Skeleton className="h-4 w-64 bg-elevated" />
   994→              <Skeleton className="h-10 w-48 bg-elevated" />
   995→            </div>
   996→          ) : failure ? (
   997→            <div>
   998→              <p role="alert" className="text-sm text-destructive">{failure}</p>
   999→              <Button size="sm" variant="outline" className="mt-3 border-gold/40 text-gold hover:bg-gold/10" onClick={() => setRetry(r => r + 1)}>
  1000→                <RotateCcw className="mr-1.5 h-3.5 w-3.5" /> Try again
  1001→              </Button>
  1002→            </div>
  1003→          ) : data && data.ready ? (
  1004→            <>
  1005→              <p className="text-sm leading-relaxed text-foreground/90">
  1006→                You have <span className="font-semibold text-gold">{data.count} wrong answer{data.count === 1 ? '' : 's'}</span> saved
  1007→                from real attempts in {examName}. This test rebuilds from exactly those questions — the honest way to
  1008→                close gaps.
  1009→              </p>
  1010→              <Button
  1011→                size="lg"
  1012→                className="mt-4 w-full bg-gold text-primary-foreground hover:bg-gold-hover sm:w-auto"
  1013→                onClick={onStart}
  1014→                disabled={pending}
  1015→              >
  1016→                {pending
  1017→                  ? <><Loader2 className="mr-2 h-4 w-4 animate-spin" aria-hidden /> Building your revision test…</>
  1018→                  : <><RotateCcw className="mr-2 h-4 w-4" /> Start revision test ({Math.min(15, data.count)} questions)</>}
  1019→              </Button>
  1020→            </>
  1021→          ) : data ? (
  1022→            <p className="text-sm leading-relaxed text-muted-foreground">
  1023→              {data.count === 0
  1024→                ? `Revision tests are built from your own wrong answers in ${examName}. Take a practice test first — this card activates automatically once you've made (and we've saved) a few mistakes.`
  1025→                : `${data.count} wrong answer${data.count === 1 ? '' : 's'} saved so far — a revision test unlocks at 4. Keep practising.`}
  1026→            </p>
  1027→          ) : null}
  1028→        </CardContent>
  1029→      </Card>
  1030→    </section>
  1031→  )
  1032→}
