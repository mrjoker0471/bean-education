    61→/** New tabs with graceful handling of legacy deep links. */
    62→function normalizeTab(tab: string | undefined): TabId {
    63→  switch (tab) {
    64→    case 'study': case 'intelligence': case 'practice':
    65→      return tab
    66→    case 'overview': case 'syllabus': case 'materials':
    67→      return 'study'
    68→    case 'tests': case 'practice-exam':
    69→      return 'practice'
    70→    case 'bean':
    71→      return 'intelligence'
    72→    default:
    73→      return 'study'
    74→  }
    75→}

    77→export function ExamView({ slug, initialTab }: { slug: string; initialTab?: string }) {
    78→  const navigate = useAppStore(s => s.navigate)
    79→  const guestId = useAppStore(s => s.guestId)
    80→
    81→  const [reloadToken, setReloadToken] = useState(0)
    82→  const [tab, setTab] = useState<TabId>(normalizeTab(initialTab))
    83→  const [pendingTopicId, setPendingTopicId] = useState<string | null>(null)
    84→  // The selected topic shared across the three layers (Study ⇄ Intelligence ⇄ Practice)
    85→  const [focusTopic, setFocusTopic] = useState<FocusTopic | null>(null)
    86→
    87→  // Keyed load state — each slug+retry combination has its own slot, so loading/error are
    88→  // derived during render (no cascading setState inside effects).
    89→  const loadKey = `${slug}|${reloadToken}`
    90→  const [loaded, setLoaded] = useState<{ key: string; data: ExamDetailDTO } | null>(null)
    91→  const [failure, setFailure] = useState<{ key: string; message: string } | null>(null)
    92→
    93→  // Load the exam
    94→  useEffect(() => {
    95→    let alive = true
    96→    api.exam(slug)
    97→      .then(d => {
    98→        if (!alive) return
    99→        setLoaded({ key: loadKey, data: d })
   100→        setFailure(null)
   101→        // Carry exam context into BEAN Intelligence (#/bean inherits it)
   102→        useAppStore.getState().setBeanContext({ examSlug: slug, examName: d.exam.name })
   103→      })
   104→      .catch(e => {
   105→        if (alive) setFailure({ key: loadKey, message: e.message })
   106→      })
   107→    return () => {
   108→      alive = false
   109→    }
   110→  }, [slug, loadKey])
   111→
   112→  // Keep the active tab in sync with deep links (browser back/forward).
   113→  // Guarded render-time adjustment — the pattern documented at react.dev/learn/you-might-not-need-an-effect.
   114→  const [lastInitialTab, setLastInitialTab] = useState(initialTab)
   115→  if (lastInitialTab !== initialTab) {
   116→    setLastInitialTab(initialTab)
   117→    setTab(normalizeTab(initialTab))
   118→  }
   119→
   120→  const data = loaded && loaded.key === loadKey ? loaded.data : null
   121→  const error = failure && failure.key === loadKey ? failure.message : null
   122→  const loading = !data && !error
   123→
   124→  const changeTab = (value: string) => {
   125→    const next = normalizeTab(value)
   126→    setTab(next)
   127→    navigate({ view: 'exam', slug, tab: next === 'study' ? undefined : next })
   128→  }
   129→
   130→  /** Layer-crossing: focus a topic and hand its full context to BEAN Intelligence. */
   131→  const focusOnTopic = (t: FocusTopic, noteTitle?: string) => {
   132→    setFocusTopic(t)
   133→    useAppStore.getState().setBeanContext({
   134→      examSlug: slug,
   135→      examName: data?.exam.name,
   136→      stageName: t.stageName,
   137→      subjectName: t.subjectName,
   138→      topicId: t.id,
   139→      topicName: t.name,
   140→      noteTitle,
   141→    })
   142→    changeTab('intelligence')
   143→  }
   144→
   145→  const startTopicTest = (topic: TopicNode) => {
   146→    if (!guestId) {
   147→      toast({
   148→        title: 'Preparing your session',
   149→        description: 'Your guest profile is still initialising — try again in a moment.',
   150→        variant: 'destructive',
   151→      })
   152→      return
   153→    }
   154→    if (pendingTopicId) return
   155→    setPendingTopicId(topic.id)
   156→    api.startTest({ guestId, topicId: topic.id })
   157→      .then(r => navigate({ view: 'test', attemptId: r.attemptId }))
   158→      .catch(e => {
   159→        toast({ title: 'Could not start the practice test', description: e.message, variant: 'destructive' })
   160→      })
   161→      .finally(() => setPendingTopicId(null))
   162→  }
   163→
   164→  // ─── Loading ───
   165→  if (loading) return <ExamSkeleton />
   166→
   167→  // ─── Not found / errors ───
   168→  if (error) {
   169→    if (/not found/i.test(error)) {
   170→      return (
   171→        <div className="mx-auto w-full max-w-3xl px-4 py-16 sm:px-6 lg:px-8">
   172→          <EmptyState
   173→            icon={Compass}
   174→            title="Examination not found"
   175→            description={`No examination lives at “${slug}” in the BEAN EXAMS catalogue. It may have been renamed — browse the catalogue to find what you need.`}
   176→            actionLabel="Back to catalogue"
   177→            onAction={() => navigate({ view: 'catalog' })}
   178→          />
   179→        </div>
   180→      )
   181→    }
   182→    return (
   183→      <div className="mx-auto w-full max-w-3xl px-4 py-16 sm:px-6 lg:px-8">
   184→        <div role="alert" className="rounded-xl border border-destructive/40 bg-destructive/10 p-6">
   185→          <p className="font-display text-lg font-semibold text-destructive">Could not load this examination</p>
   186→          <p className="mt-2 text-sm text-destructive/90">{error}</p>
   187→          <div className="mt-5 flex flex-wrap gap-3">
   188→            <Button variant="outline" className="border-gold/40 text-gold hover:bg-gold/10" onClick={() => setReloadToken(t => t + 1)}>
   189→              <RotateCcw className="mr-1.5 h-4 w-4" /> Try again
   190→            </Button>
   191→            <Button variant="outline" className="border-border text-muted-foreground hover:bg-elevated" onClick={() => navigate({ view: 'catalog' })}>
   192→              Back to catalogue
   193→            </Button>
   194→          </div>
   195→        </div>
   196→      </div>
   197→    )
   198→  }
   199→
   200→  if (!data) return null
   201→  const { exam } = data
   202→  const officialDocs = data.examResources.filter(isOfficialDoc)
   203→  const otherExamResources = data.examResources.filter(r => !isOfficialDoc(r))
   204→
   205→  return (
   206→    <div className="mx-auto w-full max-w-6xl px-4 py-8 sm:px-6 sm:py-10 lg:px-8">
   207→      {/* ─── Header ─── */}
   208→      <header>
   209→        {/* Breadcrumbs */}
   210→        <nav aria-label="Breadcrumb" className="flex flex-wrap items-center gap-x-1 gap-y-1 text-xs text-muted-foreground sm:text-sm">
   211→          <button type="button" onClick={() => navigate({ view: 'home' })} className="transition-colors hover:text-gold">
   212→            Home
   213→          </button>
   214→          <ChevronRight className="h-3.5 w-3.5 text-muted-foreground/60" aria-hidden />
   215→          <button type="button" onClick={() => navigate({ view: 'catalog' })} className="transition-colors hover:text-gold">
   216→            All examinations
   217→          </button>
   218→          {exam.categoryName && (
   219→            <>
   220→              <ChevronRight className="h-3.5 w-3.5 text-muted-foreground/60" aria-hidden />
   221→              <button
   222→                type="button"
   223→                onClick={() => navigate({ view: 'catalog', category: exam.categorySlug })}
   224→                className="transition-colors hover:text-gold"
   225→              >
   226→                {exam.categoryName}
   227→              </button>
   228→            </>
   229→          )}
   230→          <ChevronRight className="h-3.5 w-3.5 text-muted-foreground/60" aria-hidden />
   231→          <span aria-current="page" className="font-medium text-foreground">{exam.shortName ?? exam.name}</span>
   232→        </nav>
   233→
   234→        <button
   235→          type="button"
   236→          onClick={() => navigate({ view: 'catalog' })}
   237→          className="mt-4 inline-flex items-center gap-1.5 text-sm text-muted-foreground transition-colors hover:text-gold"
   238→        >
   239→          <ArrowLeft className="h-4 w-4" aria-hidden /> All examinations
   240→        </button>
   241→
   242→        <div className="mt-5 flex flex-wrap items-center gap-2">
   243→          <span className="rounded-full border border-gold/30 bg-gold/5 px-3 py-1 text-xs font-medium text-gold">
   244→            {exam.shortName ?? 'Examination'}
   245→          </span>
   246→          <SyllabusStatusBadge status={exam.syllabusStatus} />
   247→        </div>
   248→
   249→        <h1 className="mt-3 font-display text-2xl font-bold leading-tight text-foreground sm:text-3xl lg:text-4xl">
   250→          {exam.name}
   251→        </h1>
   252→
   253→        <p className="mt-2 flex flex-wrap items-center gap-x-4 gap-y-1 text-sm text-muted-foreground">
   254→          <span>Conducted by {exam.authority}</span>
   255→          {exam.state && <span>{exam.state}</span>}
   256→        </p>
   257→
   258→        <p className="mt-4 max-w-3xl text-sm leading-relaxed text-muted-foreground sm:text-base">{exam.description}</p>
   259→
   260→        {/* Exam pattern summary — always visible above the three learning tabs */}
   261→        <PatternSummaryStrip stages={data.stages} />
   262→
   263→        {/* Honesty feature: preparation coverage */}
   264→        {exam.contentNote && (
   265→          <div className="mt-5 max-w-3xl rounded-xl border border-gold/30 bg-gold/5 p-4 sm:p-5" role="note">
   266→            <p className="flex items-center gap-2 text-sm font-semibold text-gold">
   267→              <Info className="h-4 w-4 shrink-0" aria-hidden /> Honest coverage status
   268→            </p>
   269→            <p className="mt-2 text-sm leading-relaxed text-foreground/90">{exam.contentNote}</p>
   270→          </div>
   271→        )}
   272→
   273→        {/* Official source links */}
   274→        {exam.sourceRefs && exam.sourceRefs.length > 0 && (
   275→          <div className="mt-4 flex flex-wrap items-center gap-2">
   276→            <span className="text-xs uppercase tracking-[0.14em] text-muted-foreground">Official sources</span>
   277→            {exam.sourceRefs.map(ref => (
   278→              <a
   279→                key={ref.url}
   280→                href={ref.url}
   281→                target="_blank"
   282→                rel="noopener noreferrer"
   283→                className="inline-flex items-center gap-1 rounded-full border border-border bg-card px-3 py-1 text-xs text-muted-foreground transition-colors hover:border-gold/40 hover:text-gold"
   284→              >
   285→                {ref.label} <ExternalLink className="h-3 w-3" aria-hidden />
   286→              </a>
   287→            ))}
   288→          </div>
   289→        )}
   290→
   291→        <div className="mt-5 flex flex-wrap gap-2">
   292→          <CountPill label="topics" value={exam.counts.topics} />
   293→          <CountPill label="notes" value={exam.counts.notes} />
   294→          <CountPill label="questions" value={exam.counts.questions} />
   295→          <CountPill label="tests" value={exam.counts.tests} />
   296→        </div>
   297→      </header>
   298→
   299→      {/* ─── THE THREE LEARNING LAYERS ─── */}
   300→      <div
   301→        role="tablist"
   302→        aria-label="Learning layers"
   303→        className="mt-8 grid grid-cols-1 gap-2 sm:grid-cols-3 sm:gap-3"
   304→      >
   305→        {TABS.map(t => {
   306→          const active = tab === t.id
   307→          return (
   308→            <button
   309→              key={t.id}
   310→              type="button"
   311→              role="tab"
   312→              id={`exam-tab-${t.id}`}
   313→              aria-selected={active}
   314→              aria-controls={`exam-panel-${t.id}`}
   315→              onClick={() => changeTab(t.id)}
   316→              className={cn(
   317→                'group flex items-center gap-3 rounded-xl border px-4 py-3 text-left transition-colors',
   318→                active
   319→                  ? 'border-gold bg-gold/10 shadow-[0_0_0_1px_rgba(214,165,20,0.35)]'
   320→                  : 'border-border bg-card hover:border-gold/40 hover:bg-elevated'
   321→              )}
   322→            >
   323→              <span
   324→                className={cn(
   325→                  'flex h-9 w-9 shrink-0 items-center justify-center rounded-lg border',
   326→                  active ? 'border-gold/50 bg-gold/15' : 'border-border bg-elevated'
   327→                )}
   328→              >
   329→                <t.icon className={cn('h-4 w-4', active ? 'text-gold' : 'text-muted-foreground')} aria-hidden />
   330→              </span>
   331→              <span className="min-w-0">
   332→                <span className={cn('block text-sm font-semibold sm:text-base', active ? 'text-gold' : 'text-foreground')}>
   333→                  {t.label}
   334→                </span>
   335→                <span className="block truncate text-[11px] text-muted-foreground sm:text-xs">{t.sub}</span>
   336→              </span>
   337→            </button>
   338→          )
   339→        })}
   340→      </div>
   341→
   342→      <div
   343→        role="tabpanel"
   344→        id={`exam-panel-${tab}`}
   345→        aria-labelledby={`exam-tab-${tab}`}
   346→        className="mt-8 focus-visible:outline-none"
   347→      >
   348→        {tab === 'study' && (
   349→          <StudyTab
   350→            data={data}
   351→            officialDocs={officialDocs}
   352→            otherExamResources={otherExamResources}
   353→            pendingTopicId={pendingTopicId}
   354→            onStartTopicTest={startTopicTest}
   355→            onFocusTopic={focusOnTopic}
   356→          />
   357→        )}
   358→        {tab === 'intelligence' && (
   359→          <IntelligenceTab
   360→            data={data}
   361→            slug={slug}
   362→            focusTopic={focusTopic}
   363→            onFocusTopic={focusOnTopic}
   364→            pendingTopicId={pendingTopicId}
   365→            onStartTopicTest={startTopicTest}
   366→            onGoPractice={() => changeTab('practice')}
   367→          />
   368→        )}
   369→        {tab === 'practice' && (
   370→          <PracticeTab
   371→            data={data}
   372→            guestId={guestId}
   373→            slug={slug}
   374→            focusTopic={focusTopic}
   375→          />
   376→        )}
   377→      </div>
   378→    </div>
   379→  )
   380→}

   382→// ═══ LAYER 1 — STUDY MATERIAL + OFFICIAL PDFS ═══════════════════════════════
   383→function StudyTab({
   384→  data,
   385→  officialDocs,
   386→  otherExamResources,
   387→  pendingTopicId,
   388→  onStartTopicTest,
   389→  onFocusTopic,
   390→}: {
   391→  data: ExamDetailDTO
   392→  officialDocs: ResourceDTO[]
   393→  otherExamResources: ResourceDTO[]
   394→  pendingTopicId: string | null
   395→  onStartTopicTest: (topic: TopicNode) => void
   396→  onFocusTopic: (t: FocusTopic, noteTitle?: string) => void
   397→}) {
   398→  const multiStage = data.stages.length > 1
   399→  const [stageIdx, setStageIdx] = useState(0)
   400→  const stage = data.stages[Math.min(stageIdx, data.stages.length - 1)]

   440→            Some official sites block automated checks from our servers, so their status may read &ldquo;unreachable&rdquo;
   441→            even though the link is the authority&apos;s real, current page. Statuses reflect our last actual check —
   442→            they are never assumed.
   443→          </p>
   444→        )}
   445→      </section>
   446→
   447→      {/* ── Exam pattern & selection stages ── */}
   448→      <section aria-labelledby="pattern-heading">
   449→        <h2 id="pattern-heading" className="font-display text-lg font-semibold text-foreground sm:text-xl">

   451→        </h2>

   453→          <p className="mt-4 rounded-lg border border-border bg-card px-4 py-3 text-sm text-muted-foreground">
   454→            Stage structure for this examination is being mapped.
   455→          </p>
   456→        ) : (
   457→          <div className="mt-4 space-y-4">
   458→            {data.stages.map(s => (
   459→              <Card key={s.id} className="border-border bg-card">

   465→                    </div>
   466→                    <CountPill label="subjects" value={s.subjects.length} />
   467→                  </div>
   468→                  {s.pattern ? (
   469→                    <div className="mt-4">
   470→                      <PatternTable pattern={s.pattern} />
   471→                    </div>

   487→          Syllabus &amp; topic-wise study notes
   488→        </h2>
   489→        <p className="mt-1 max-w-3xl text-sm text-muted-foreground">
   490→          Expand any topic to read its original BEAN EXAMS note right here, see its verified documents, and jump
   491→          into practice or BEAN Intelligence with the topic&apos;s context carried over.

   494→        {multiStage && (

   496→            {data.stages.map((s, i) => (
   497→              <button
   498→                key={s.id}
   499→                type="button"
   500→                role="tab"
   501→                aria-selected={i === stageIdx}
   502→                onClick={() => setStageIdx(i)}

   927→// ─── Revision section (lazy check against this guest's wrong answers) ────────
   928→function RevisionSection({
   929→  examId,
   930→  examName,
   931→  guestId,
   932→  pending,
   933→  onStart,
   934→}: {
   935→  examId: string
   936→  examName: string
   937→  guestId: string
   938→  pending: boolean
   939→  onStart: () => void
   940→}) {
   941→  const loadKey = `${examId}|${guestId}`
   942→  const [loaded, setLoaded] = useState<{ key: string; count: number; ready: boolean } | null>(null)
   943→  const [failed, setFailed] = useState<{ key: string; message: string } | null>(null)
   944→  const [retry, setRetry] = useState(0)
   945→
   946→  const data = loaded && loaded.key === `${loadKey}|${retry}` ? loaded : null
   947→  const failure = failed && failed.key === `${loadKey}|${retry}` ? failed.message : null
   948→
   949→  useEffect(() => {
   950→    if (!guestId) return
   951→    let alive = true
   952→    api.revisionCheck(examId, guestId)
   953→      .then(r => { if (alive) setLoaded({ key: `${loadKey}|${retry}`, count: r.count, ready: r.ready }) })
   954→      .catch(e => { if (alive) setFailed({ key: `${loadKey}|${retry}`, message: e.message }) })
   955→    return () => { alive = false }
   956→  }, [loadKey, retry, guestId, examId])
   957→
   958→  return (
   959→    <section aria-labelledby="revision-heading">
   960→      <h2 id="revision-heading" className="font-display text-lg font-semibold text-foreground sm:text-xl">
   961→        Revision test — built from your mistakes
   962→      </h2>
   963→      <Card className="mt-4 border-gold/25 bg-gradient-to-b from-gold/10 to-transparent">
   964→        <CardContent className="p-4 sm:p-5">
   965→          {!guestId || (!data && !failure) ? (
   966→            <div className="space-y-2" aria-busy="true" aria-label="Checking revision availability">
   967→              <Skeleton className="h-4 w-64 bg-elevated" />
   968→              <Skeleton className="h-10 w-48 bg-elevated" />
   969→            </div>
   970→          ) : failure ? (
   971→            <div>
   972→              <p role="alert" className="text-sm text-destructive">{failure}</p>
   973→              <Button size="sm" variant="outline" className="mt-3 border-gold/40 text-gold hover:bg-gold/10" onClick={() => setRetry(r => r + 1)}>
   974→                <RotateCcw className="mr-1.5 h-3.5 w-3.5" /> Try again
   975→              </Button>
   976→            </div>
   977→          ) : data && data.ready ? (
   978→            <>
   979→              <p className="text-sm leading-relaxed text-foreground/90">
