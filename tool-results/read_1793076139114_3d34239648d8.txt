     1→     1→// ─── Riishe document structure detection (deterministic, no AI) ──────────────
     2→     2→// Infers a document's real organisation from the extracted text itself:
     3→     3→//   • table of contents (detected + validated against the body)
     4→     4→//   • headings (numbered, chapter-style, ALL-CAPS, Title-Case) with levels

     7→     7→//     matter) so page references can show both honestly

     9→     9→// whole document — never an arbitrary "first N pages" slice.

    79→    79→function linesOf(page: ExtractedPage): string[] {

    83→    83→function letters(s: string): string {
    84→    84→  return s.replace(/[^A-Za-z]/g, '')

    87→    87→function upperRatio(s: string): number {
    88→    88→  const ls = letters(s)
    89→    89→  if (ls.length === 0) return 0
    90→    90→  let up = 0
    91→    91→  for (const c of ls) if (c === c.toUpperCase() && c !== c.toLowerCase()) up++
    92→    92→  return up / ls.length
    93→    93→}

    95→    95→function titleCaseRatio(s: string): number {
    96→    96→  const words = s.split(/\s+/).filter(w => WORD_RE.test(w))
    97→    97→  if (words.length === 0) return 0
    98→    98→  let cap = 0
    99→    99→  for (const w of words) {
   100→   100→    const first = letters(w)[0]
   101→   101→    if (first && first === first.toUpperCase()) cap++
   102→   102→  }
   103→   103→  return cap / words.length
   104→   104→}

   106→   106→const NON_HEADING_RE = /^(page|fig(?:ure)?|table|source|note|answer|marks?|time\s*[:–-]|total|name|roll|date|reg(istration)?\s*(no|number))/i

   110→   110→const TOC_LEADER_RE = /^(.{2,90}?)[\s.·⋅•‣⁃…]{2,}(\d{1,3})(?:\s*[-–—]\s*\d{1,3})?$/
   111→   111→const TOC_GAP_RE = /^(.{2,90}?)\s{3,}(\d{1,3})(?:\s*[-–—]\s*\d{1,3})?$/
   112→   112→// Real contents tables often use a single space: "Chapter-1 Introduction 13-14"
   113→   113→const TOC_TAIL_RE = /^(.{2,90}?)\s+(\d{1,3})(?:\s*[-–—]\s*\d{1,3})?$/
   114→   114→const TOC_PAGE_HINT_RE = /^(index|contents?|table of contents|content page no\.?|appendices?)\s*$/i

   121→   121→function detectTocOnPage(page: ExtractedPage): TocEntry[] {
   122→   122→  const lines = linesOf(page)
   123→   123→  if (lines.length < TOC_MIN_ENTRIES) return []
   124→   124→  const entries: TocEntry[] = []
   125→   125→  for (const line of lines) {
   126→   126→    const m = TOC_LEADER_RE.exec(line) ?? TOC_GAP_RE.exec(line) ?? TOC_TAIL_RE.exec(line)
   127→   127→    if (!m) continue
   128→   128→    const title = m[1].replace(/[\s.·⋅•‣⁃…]+$/g, '').trim()
   129→   129→    const num = Number(m[2])
   130→   130→    if (title.length < 2 || title.length > 90) continue
   131→   131→    if (!WORD_RE.test(title)) continue
   132→   132→    if (PURE_NUMBER_RE.test(title)) continue
   133→   133→    entries.push({ title, page: Number.isFinite(num) ? num : null })
   134→   134→  }
   135→   135→  // A page is a TOC page when either a clear majority of its lines are entries,
   136→   136→  // or it announces itself (Index / Contents / Appendices) and has several.
   137→   137→  const announces = lines.slice(0, 6).some(l => TOC_PAGE_HINT_RE.test(l))
   138→   138→  const threshold = announces
   139→   139→    ? Math.max(TOC_MIN_ENTRIES, Math.ceil(lines.length * 0.25))
   140→   140→    : Math.max(TOC_MIN_ENTRIES, Math.ceil(lines.length * 0.5))
   141→   141→  return entries.length >= threshold ? entries : []
   142→   142→}

   144→   144→/** Validate a TOC: entry pages must be plausible and broadly increasing. */
   145→   145→function validateToc(entries: TocEntry[], numPages: number): boolean {
   146→   146→  const withPages = entries.filter(e => e.page !== null) as (TocEntry & { page: number })[]
   147→   147→  if (withPages.length < TOC_MIN_ENTRIES) return false
   148→   148→  let inRange = 0
   149→   149→  for (const e of withPages) if (e.page >= 1 && e.page <= numPages) inRange++
   150→   150→  if (inRange / withPages.length < 0.8) return false
   151→   151→  // Broadly increasing: allow local noise (multi-column TOCs reorder), but the
   152→   152→  // overall trend must go up — otherwise the "TOC" is likely a name list.
   153→   153→  let up = 0
   154→   154→  let total = 0
   155→   155→  for (let i = 1; i < withPages.length; i++) {
   156→   156→    total++
   157→   157→    if (withPages[i].page >= withPages[i - 1].page - 2) up++
   158→   158→  }
   159→   159→  return total === 0 || up / total >= 0.75
   160→   160→}

   164→   164→const CHAPTER_WORD_RE =
   165→   165→  /^(chapter|unit|section|part|module|appendix|annexure|annex|experiment|activity|practical|lesson)\s*[-–—:]?\s*([0-9ivxlcdm]+|[a-z])\b[\s.:–—-]*(.*)$/i

   167→   167→const NUMBERED_L2_RE = /^(\d{1,2})\.(\d{1,2})[.)]?\s+(\S.*)$/

   169→   169→const QUESTION_LINE_RE = /^\d{1,3}[.)]\s+(.{10,})$/
   170→   170→const PURE_NUMBER_RE = /^\d{1,4}$/
   171→   171→const ROMAN_RE = /^(?:[ivxlcdm]{1,7})$/i
   172→   172→/** "CHAPTER 10", "Appendix VII", "UNIT 3" — a bare label with no title text */
   173→   173→const BARE_CHAPTER_RE = /^(chapter|unit|part|section|appendix|annexure|annex)\s*[-–—:]?\s*([0-9ivxlcdm]{1,5}|[a-z])\.?$/i

   183→   183→function headingScore(line: string, inTocTitles: Set<string>): { level: 1 | 2; score: number; title: string } | null {
   184→   184→  const raw = line.trim()
   185→   185→  if (raw.length < 3 || raw.length > HEADING_MAX_LEN) return null
   186→   186→  if (PURE_NUMBER_RE.test(raw)) return null
   187→   187→  if (/[.,;:]$/.test(raw) && !NUMBERED_L1_RE.test(raw) && !NUMBERED_L2_RE.test(raw)) return null
   188→   188→  if (NON_HEADING_RE.test(raw)) return null
   189→   189→
   190→   190→  // x.y numbered subsection — strongest hierarchical signal
   191→   191→  const l2 = NUMBERED_L2_RE.exec(raw)
   192→   192→  if (l2 && l2[3].length >= 3 && WORD_RE.test(l2[3])) {
   193→   193→    return { level: 2, score: 6, title: l2[3].trim() }
   194→   194→  }
   195→   195→
   196→   196→  // Chapter / Unit / Appendix … style — keep the document's own wording
   197→   197→  const ch = CHAPTER_WORD_RE.exec(raw)
   198→   198→  if (ch) {
   199→   199→    return { level: 1, score: 7, title: cleanTitle(raw) }
   200→   200→  }
   201→   201→
   202→   202→  // "Section A", "PART — I"
   203→   203→  const sec = SECTION_LABEL_RE.exec(raw)
   204→   204→  if (sec && (sec[3] ?? '').trim().length < 40) {
   205→   205→    return { level: 1, score: 5, title: cleanTitle(raw) }
   206→   206→  }
   207→   207→
   208→   208→  // Numbered top-level heading "12. Reservation Policy"
   209→   209→  const l1 = NUMBERED_L1_RE.exec(raw)
   210→   210→  if (l1) {
   211→   211→    const title = l1[2].trim()
   212→   212→    if (title.length >= 3 && WORD_RE.test(title) && !QUESTION_LINE_RE.test(raw)) {
   213→   213→      const bonus = inTocTitles.has(normKey(title)) ? 2 : 0
   214→   214→      return { level: 1, score: 4 + bonus, title: cleanTitle(title) }
   215→   215→    }
   216→   216→    return null
   217→   217→  }
   218→   218→
   219→   219→  // ALL-CAPS short line
   220→   220→  const ls = letters(raw)
   221→   221→  if (ls.length >= 4 && upperRatio(raw) >= 0.8 && raw.split(/\s+/).length <= 12) {
   222→   222→    const bonus = inTocTitles.has(normKey(raw)) ? 2 : 0
   223→   223→    return { level: 1, score: 3 + bonus, title: cleanTitle(raw) }
   224→   224→  }
   225→   225→
   226→   226→  // Title-Case short line (two+ capitalised words, no ending punctuation)
   227→   227→  const words = raw.split(/\s+/)
   228→   228→  if (words.length >= 2 && words.length <= 10 && titleCaseRatio(raw) >= 0.7 && raw.length >= 8) {
   229→   229→    const bonus = inTocTitles.has(normKey(raw)) ? 3 : 0
   230→   230→    // Without a TOC match, plain Title-Case lines are weak signals — they are
   231→   231→    // often names, table headers or list items. Only TOC-corroborated ones or
   232→   232→    // clearly heading-ish lines (no verb-ish trailing punctuation) pass.
   233→   233→    if (bonus > 0) return { level: 1, score: 3 + bonus, title: cleanTitle(raw) }
   234→   234→    return null
   235→   235→  }
   236→   236→
   237→   237→  return null
   238→   238→}

   240→   240→function cleanTitle(t: string): string {
   241→   241→  return t.replace(/\s+/g, ' ').replace(/[.:]\s*$/, '').trim().slice(0, 90)
   242→   242→}

   244→   244→/** Remove leading row numbers from a TOC entry title: "1.1 About NTA" → "About NTA". */
   245→   245→function stripEntryTitle(t: string): string {

   248→   248→    .replace(/^\d{1,2}\s*[.)]\s+/, '')
   249→   249→    .replace(/\s+/g, ' ')
   250→   250→    .trim()
   251→   251→}

   256→   256→/** Normalisation used to correlate TOC titles with body headings. */
   257→   257→export function normKey(s: string): string {
   258→   258→  return s
   259→   259→    .toLowerCase()
   260→   260→    .replace(/[^a-z0-9]+/g, ' ')
   261→   261→    .replace(/\b(the|a|an|of|for|and|in|to|on)\b/g, ' ')
   262→   262→    .replace(/\s+/g, ' ')
   263→   263→    .trim()
   264→   264→}

   268→   268→const ROMAN_VALUES: Record<string, number> = { i: 1, v: 5, x: 10, l: 50, c: 100, d: 500, m: 1000 }
   269→   269→
   270→   270→function romanToInt(s: string): number | null {
   271→   271→  const str = s.toLowerCase()
   272→   272→  if (!/^[ivxlcdm]+$/.test(str)) return null
   273→   273→  let total = 0
   274→   274→  let prev = 0
   275→   275→  for (let i = str.length - 1; i >= 0; i--) {
   276→   276→    const v = ROMAN_VALUES[str[i]]
   277→   277→    if (v < prev) total -= v
   278→   278→    else {
   279→   279→      total += v
   280→   280→      prev = v
   281→   281→    }
   282→   282→  }
   283→   283→  return total >= 1 ? total : null
   284→   284→}

   286→   286→function printedLabelFromEdge(lines: string[]): string | null {
   287→   287→  const edges = [...lines.slice(0, 2), ...lines.slice(-2)]
   288→   288→  for (const line of edges) {
   289→   289→    if (line.length > 40) continue
   290→   290→    // "Page 12 of 96", "Page |12", "Page | 12", "12", "- 12 -", "12."
   291→   291→    const m1 = /^page\s*[|:\-]?\s*(\d{1,4})(?:\s*(?:of|\/)\s*\d{1,4})?$/i.exec(line)
   292→   292→    if (m1) return m1[1]
   293→   293→    const m2 = /^[-–—\s]*(\d{1,4})[-–—\s.]*$/.exec(line)
   294→   294→    if (m2) return m2[1]
   295→   295→    if (ROMAN_RE.test(line) && line.length <= 7) {
   296→   296→      const v = romanToInt(line)
   297→   297→      if (v !== null && v <= 100) return line.toLowerCase()
   298→   298→    }
   299→   299→  }
   300→   300→  return null
   301→   301→}

   303→   303→export function detectPrintedPages(pages: ExtractedPage[]): PrintedPageInfo {
   304→   304→  const pairs: { physical: number; printed: number | null; roman: boolean }[] = []
   305→   305→  for (const p of pages) {
   306→   306→    if (p.scanned || !p.text) {
   307→   307→      pairs.push({ physical: p.page, printed: null, roman: false })
   308→   308→      continue
   309→   309→    }
   310→   310→    const label = printedLabelFromEdge(linesOf(p))
   311→   311→    if (label === null) {
   312→   312→      pairs.push({ physical: p.page, printed: null, roman: false })
   313→   313→    } else if (/^\d+$/.test(label)) {
   314→   314→      pairs.push({ physical: p.page, printed: Number(label), roman: false })
   315→   315→    } else {
   316→   316→      pairs.push({ physical: p.page, printed: romanToInt(label), roman: true })
   317→   317→    }
   318→   318→  }
   319→   319→
   320→   320→  // Roman front matter anywhere in the first 15% of the document
   321→   321→  const headCount = Math.max(1, Math.ceil(pairs.length * 0.15))
   322→   322→  const romanFrontMatter = pairs.slice(0, headCount).some(p => p.roman && p.printed !== null)
   323→   323→
   324→   324→  // Longest consistent arabic run: printed = physical + offset
   325→   325→  let bestOffset: number | null = null
   326→   326→  let bestRun = 0
   327→   327→  const byOffset = new Map<number, number>()
   328→   328→  for (const p of pairs) {
   329→   329→    if (p.printed === null || p.roman) continue
   330→   330→    const offset = p.physical - p.printed
   331→   331→    if (offset < 0 || offset > 60) continue // cover/plate offsets beyond 60 are implausible
   332→   332→    byOffset.set(offset, (byOffset.get(offset) ?? 0) + 1)
   333→   333→  }
   334→   334→  for (const [offset, count] of byOffset) {
   335→   335→    if (count > bestRun) {
   336→   336→      bestRun = count
   337→   337→      bestOffset = offset
   338→   338→    }
   339→   339→  }
   340→   340→  // Require the offset to hold for a meaningful share of pages (≥5 and ≥25%)
   341→   341→  const arabicPages = pairs.filter(p => p.printed !== null && !p.roman).length
   342→   342→  if (bestOffset !== null && !(bestRun >= 5 && bestRun >= arabicPages * 0.5 && arabicPages >= pages.length * 0.25)) {
   343→   343→    bestOffset = null
   344→   344→  }
   345→   345→  return { offset: bestOffset, romanFrontMatter }
   346→   346→}

   350→   350→export function detectStructure(pages: ExtractedPage[], numPages: number, opts: BuildUnitsOptions): DetectedStructure {
   351→   351→  const readable = pages.filter(p => !p.scanned && p.text.length > 0)
   352→   352→  const printed = detectPrintedPages(pages)
   353→   353→
   354→   354→  // 1 ── TOC pages
   355→   355→  const tocPages: TocPageResult[] = []
   356→   356→  for (const p of readable.slice(0, 15)) {
   357→   357→    const entries = detectTocOnPage(p)
   358→   358→    if (entries.length > 0) tocPages.push({ page: p.page, entries })
   359→   359→  }
   360→   360→  const allTocEntries = tocPages.flatMap(t => t.entries)
   361→   361→  const tocValidated = validateToc(allTocEntries, numPages)
   362→   362→  // Titles stripped of leading row numbers ("1.1 About NTA" → "About NTA") so
   363→   363→  // TOC entries correlate with the body's own headings.
   364→   364→  const tocTitleKeys = new Set(allTocEntries.map(e => normKey(stripEntryTitle(e.title))))
   365→   365→
   366→   366→  // 2 ── Heading candidates over all readable lines
   367→   367→  const hits: LineHit[] = []
   368→   368→  let questionLines = 0
   369→   369→  let contentLines = 0
   370→   370→  for (const p of readable) {
   371→   371→    const onTocPage = tocPages.some(t => t.page === p.page)
   372→   372→    const lines = linesOf(p)
   373→   373→    for (let i = 0; i < lines.length; i++) {
   374→   374→      const line = lines[i]
   375→   375→      if (line.length >= 12) contentLines++
   376→   376→      if (QUESTION_LINE_RE.test(line)) questionLines++
   377→   377→      if (onTocPage) continue // TOC lines look like headings but are not
   378→   378→      const h = headingScore(line, tocTitleKeys)
   379→   379→      if (!h) continue
   380→   380→      // Bare "CHAPTER 10" / "APPENDIX VII" headings: the real title often sits
   381→   381→      // on the next line — merge it so the topic keeps the document's wording.
   382→   382→      let title = h.title
   383→   383→      let consumedExtra = 0
   384→   384→      const bare = BARE_CHAPTER_RE.exec(h.title)
   385→   385→      if (bare) {
   386→   386→        const next = lines[i + 1]
   387→   387→        if (
   388→   388→          next && next.length >= 3 && next.length <= 90 && !PURE_NUMBER_RE.test(next) &&
   389→   389→          !NON_HEADING_RE.test(next) && !headingScore(next, tocTitleKeys) && !QUESTION_LINE_RE.test(next)
   390→   390→        ) {
   391→   391→          title = cleanTitle(`${h.title} ${next}`)
   392→   392→          consumedExtra = 1
   393→   393→        }
   394→   394→      }
   395→   395→      hits.push({ title, level: h.level, score: h.score, page: p.page, lineIdx: i })
   396→   396→      i += consumedExtra
   397→   397→    }
   398→   398→  }
   399→   399→
   400→   400→  // 3 ── Question-paper layout? High numbered-line density + explicit marks
   401→   401→  // instructions near the front — both together, so ordinary numbered lists in
   402→   402→  // textbooks (or a phone directory) are not misclassified.
   403→   403→  const marksMention = readable
   404→   404→    .slice(0, 8)
   405→   405→    .some(p => /marks?\s*[:：]|max(?:imum)?\s*marks|time\s*(?:allowed|:)\s*\d/i.test(p.text))
   406→   406→  const questionPaper =
   407→   407→    contentLines > 30 && questionLines / Math.max(1, contentLines) >= QUESTION_LINE_RATIO && marksMention
   408→   408→
   409→   409→  // 4 ── Heading density sanity: too many "headings" ⇒ they are not headings
   410→   410→  let headingHits = hits.filter(h => h.score >= 4)
   411→   411→  const density = contentLines > 0 ? headingHits.length / contentLines : 0
   412→   412→  if (density > HEADING_DENSITY_LIMIT) headingHits = headingHits.filter(h => h.score >= 6)
   413→   413→  if (contentLines > 0 && headingHits.length / contentLines > HEADING_DENSITY_LIMIT) headingHits = []
   414→   414→
   415→   415→  // 5 ── Corroborate with the TOC: boost + mark
   416→   416→  const headings: HeadingHit[] = []
   417→   417→  let lastKey = ''
   418→   418→  for (const h of headingHits) {
   419→   419→    const key = normKey(h.title)
   420→   420→    if (!key || key === lastKey) continue // consecutive duplicate line
   421→   421→    lastKey = key
   422→   422→    headings.push({
   423→   423→      title: h.title,
   424→   424→      page: h.page,
   425→   425→      lineIdx: h.lineIdx,
   426→   426→      level: h.level,
   427→   427→      score: h.score,
   428→   428→      fromToc: tocTitleKeys.has(key),
   429→   429→    })
   430→   430→  }
   431→   431→
   432→   432→  // Drop weak one-off headings when a TOC exists (they are usually noise);
   433→   433→  // keep everything when the TOC corroborates them.
   434→   434→  const filtered = headings.filter(h => !tocValidated || h.fromToc || h.score >= 4)
   435→   435→
   436→   436→  // 6 ── Units. When a validated TOC exists, the document's own contents table
   437→   437→  // (validated against the body) defines the major sections — that is more
   438→   438→  // faithful than any heading heuristic. Headings are the fallback.
   439→   439→  let units: AnalysisUnit[] | null = null
   440→   440→  let kind: StructureKind
   441→   441→  if (questionPaper) {
   442→   442→    kind = 'QUESTION_PAPER'
   443→   443→  } else {
   444→   444→    if (tocValidated) units = buildTocUnits(readable, allTocEntries, printed.offset, opts)
   445→   445→    if (units && units.length >= 2) {
   446→   446→      kind = 'TOC_GUIDED'
   447→   447→    } else {
   448→   448→      units = null
   449→   449→      kind = filtered.length >= 3 ? (tocValidated ? 'TOC_GUIDED' : 'HEADING_GUIDED') : 'UNSTRUCTURED'
   450→   450→    }
   451→   451→  }
   452→   452→  if (!units) units = buildUnits(readable, filtered, kind, opts)
   453→   453→
   454→   454→  return {
   455→   455→    kind,
   456→   456→    toc: allTocEntries,
   457→   457→    tocValidated,
   458→   458→    headings: filtered,
   459→   459→    printed,
   460→   460→    units,
   461→   461→    tocPages: tocPages.map(t => t.page),
   462→   462→  }
   463→   463→}
   464→   464→
   465→   465→// ─── Analysis units ───────────────────────────────────────────────────────────
   466→   466→
   467→   467→/**
   468→   468→ * Build units directly from a VALIDATED table of contents: each major (non
   469→   469→ * x.y) entry becomes a section whose physical start page is the entry's page
   470→   470→ * (printed + detected offset) — but only when the body page actually contains
   471→   471→ * the entry's words. If too few entries verify, returns null so the caller
   472→   472→ * falls back to heading-anchored units. This is what makes page references
   473→   473→ * land on the right page in documents with a real contents table.
   474→   474→ */
   475→   475→export function buildTocUnits(
   476→   476→  readable: ExtractedPage[],
   477→   477→  entries: TocEntry[],
   478→   478→  printedOffset: number | null,
   479→   479→  opts: BuildUnitsOptions,
   480→   480→): AnalysisUnit[] | null {
   481→   481→  const physicalOf = (printed: number) => (printedOffset !== null ? printed + printedOffset : printed)
   482→   482→  const pageByNumber = new Map(readable.map(p => [p.page, p] as const))
   483→   483→  const normText = (s: string) => s.toLowerCase().replace(/[^a-z0-9]+/g, ' ')
   484→   484→
   485→   485→  const majors = entries
   486→   486→    .filter(e => e.page !== null && !SUBSECTION_ROW_RE.test(e.title.trim()))
   487→   487→    .map(e => ({ title: stripEntryTitle(e.title), physical: physicalOf(e.page as number) }))
   488→   488→    .filter(x => x.title.length >= 2 && x.title.length <= 95)
   489→   489→  if (majors.length < TOC_MIN_ENTRIES) return null
   490→   490→
   491→   491→  // Wrapped TOC rows continue on the next line claiming the same page —
   492→   492→  // "…General Aptitude" + "Test under CUET (UG) – 2026 37-38" are ONE section.
   493→   493→  const mergedMajors: { title: string; physical: number }[] = []
   494→   494→  for (const m of majors) {
   495→   495→    const prev = mergedMajors[mergedMajors.length - 1]
   496→   496→    if (prev && prev.physical === m.physical) {
   497→   497→      prev.title = cleanTitle(`${prev.title} ${m.title}`)
   498→   498→      continue
   499→   499→    }
   500→   500→    mergedMajors.push({ ...m })
   501→   501→  }
   502→   502→  if (mergedMajors.length < TOC_MIN_ENTRIES) return null
   503→   503→
   504→   504→  // Verify each entry against the physical page it claims
   505→   505→  const verified: { title: string; physical: number }[] = []
   506→   506→  for (const m of mergedMajors) {
   507→   507→    const page = pageByNumber.get(m.physical)
   508→   508→    if (!page) continue
   509→   509→    const hay = normText(page.text)
   510→   510→    const tokens = normKey(m.title).split(' ').filter(t => t.length > 3)
   511→   511→    const hit = tokens.length === 0 || tokens.some(t => hay.includes(t))
   512→   512→    if (hit) verified.push(m)
   513→   513→  }
   514→   514→  if (verified.length < TOC_MIN_ENTRIES || verified.length < Math.ceil(majors.length * 0.5)) return null
   515→   515→
   516→   516→  verified.sort((a, b) => a.physical - b.physical)
   517→   517→  const cleaned = verified.filter((v, i) => i === 0 || v.physical > verified[i - 1].physical)
   518→   518→  if (cleaned.length < TOC_MIN_ENTRIES) return null
   519→   519→
   520→   520→  const units: AnalysisUnit[] = []
   521→   521→  const textOf = (from: number, to: number) =>
   522→   522→    readable
   523→   523→      .filter(p => p.page >= from && p.page <= to)
   524→   524→      .map(p => `[page ${p.page}]\n${p.text}`)
   525→   525→      .join('\n\n')
   526→   526→
   527→   527→  const pushRange = (title: string | null, from: number, to: number, kind: AnalysisUnit['kind']) => {
   528→   528→    const pagesSlice = readable.filter(p => p.page >= from && p.page <= to)
   529→   529→    if (pagesSlice.length === 0) return
   530→   530→    // oversize ranges split into consecutive chunks under the same title
   531→   531→    let start = pagesSlice[0]
   532→   532→    let prev = pagesSlice[0]
   533→   533→    let chars = 0
   534→   534→    const flush = () => {
   535→   535→      units.push({
   536→   536→        index: units.length, title, level: title ? 1 : null,
   537→   537→        fromPage: start.page, toPage: prev.page, text: textOf(start.page, prev.page),
   538→   538→        chars, kind,
   539→   539→      })
   540→   540→    }
   541→   541→    for (const p of pagesSlice) {
   542→   542→      if (chars > 0 && chars + p.text.length > opts.maxUnitChars) {
   543→   543→        flush()
   544→   544→        start = p
   545→   545→        chars = 0
   546→   546→      }
   547→   547→      chars += p.text.length
   548→   548→      prev = p
   549→   549→    }
   550→   550→    if (chars > 0) flush()
   551→   551→  }
   552→   552→
   553→   553→  // Front matter: everything before the first verified entry
   554→   554→  const first = cleaned[0].physical
   555→   555→  const front = readable.filter(p => p.page < first)
   556→   556→  if (front.length > 0) pushRange(null, front[0].page, front[front.length - 1].page, 'FRONT')
   557→   557→
   558→   558→  for (let i = 0; i < cleaned.length; i++) {
   559→   559→    const from = cleaned[i].physical
   560→   560→    const to = i + 1 < cleaned.length ? Math.min(cleaned[i + 1].physical - 1, readable[readable.length - 1].page) : readable[readable.length - 1].page
   561→   561→    if (to < from) continue
   562→   562→    pushRange(cleaned[i].title, from, to, 'SECTION')
   563→   563→  }
   564→   564→
   565→   565→  return units.length >= 2 ? units.map((u, i) => ({ ...u, index: i })) : null
   566→   566→}
   567→   567→
   568→   568→/**
   569→   569→ * Build analysis units that cover every readable page exactly once.
   570→   570→ * TOC/heading-guided documents are sliced at real heading boundaries (level-1
   571→   571→ * sections, split at level-2 subsections when oversize); everything else falls
   572→   572→ * back to whole-document page chunks. Tiny neighbouring sections are merged so
   573→   573→ * one-line headings don't each cost a model call, and oversize sections are
   574→   574→ * split further so no single call exceeds the char budget.
   575→   575→ */
   576→   576→export function buildUnits(
   577→   577→  readable: ExtractedPage[],
   578→   578→  headings: HeadingHit[],
   579→   579→  kind: StructureKind,
   580→   580→  opts: BuildUnitsOptions,
   581→   581→): AnalysisUnit[] {
   582→   582→  const units: AnalysisUnit[] = []
   583→   583→
   584→   584→  const pushUnit = (u: Omit<AnalysisUnit, 'index'>) => {
   585→   585→    units.push({ ...u, index: units.length })
   586→   586→  }
   587→   587→
   588→   588→  const textOf = (from: ExtractedPage, to: ExtractedPage) =>
   589→   589→    readable
   590→   590→      .slice(readable.indexOf(from), readable.indexOf(to) + 1)
   591→   591→      .map(p => `[page ${p.page}]\n${p.text}`)
   592→   592→      .join('\n\n')
   593→   593→
   594→   594→  if (kind === 'TOC_GUIDED' || kind === 'HEADING_GUIDED') {
   595→   595→    // Anchor each level-1 heading to its page; sections span to the next level-1 anchor.
   596→   596→    const anchors = headings
   597→   597→      .filter(h => h.level === 1)
   598→   598→      .sort((a, b) => a.page - b.page || a.lineIdx - b.lineIdx)
   599→   599→    // Collapse anchors that land on the same page with the same key (repeat prints)
   600→   600→    const deduped: HeadingHit[] = []
   601→   601→    for (const a of anchors) {
   602→   602→      const prev = deduped[deduped.length - 1]
   603→   603→      if (prev && prev.page === a.page && normKey(prev.title) === normKey(a.title)) continue
   604→   604→      deduped.push(a)
   605→   605→    }
   606→   606→
   607→   607→
   608→   608→    // Front matter: readable pages before the first anchor
   609→   609→    const firstAnchorPage = deduped[0]?.page ?? null
   610→   610→    const frontPages = firstAnchorPage ? readable.filter(p => p.page < firstAnchorPage) : []
   611→   611→    if (frontPages.length > 0) {
   612→   612→      const from = frontPages[0]
   613→   613→      const to = frontPages[frontPages.length - 1]
   614→   614→      pushUnit({ title: null, level: null, fromPage: from.page, toPage: to.page, text: textOf(from, to), chars: frontPages.reduce((s, p) => s + p.text.length, 0), kind: 'FRONT' })
   615→   615→    }
   616→   616→
   617→   617→    for (let i = 0; i < deduped.length; i++) {
   618→   618→      const anchor = deduped[i]
   619→   619→      const nextPage = deduped[i + 1]?.page ?? null
   620→   620→      const sectionPagesList = readable.filter(p => p.page >= anchor.page && (nextPage === null || p.page < nextPage))
   621→   621→      if (sectionPagesList.length === 0) continue
   622→   622→      // Split oversize sections at level-2 heading pages within the section
   623→   623→      const subAnchors = headings
   624→   624→        .filter(h => h.level === 2 && h.page >= anchor.page && (nextPage === null || h.page < nextPage))
   625→   625→        .sort((a, b) => a.page - b.page)
   626→   626→      const segments: { from: ExtractedPage; to: ExtractedPage; title: string | null }[] = []
   627→   627→      let cursorPage = anchor.page
   628→   628→      for (const sub of subAnchors) {
   629→   629→        if (sub.page <= cursorPage) continue
   630→   630→        const from = readable.find(p => p.page === cursorPage)
   631→   631→        const to = readable.find(p => p.page === sub.page - 1) ?? readable.find(p => p.page === cursorPage)
   632→   632→        if (from && to) segments.push({ from, to, title: null })
   633→   633→        cursorPage = sub.page
   634→   634→      }
   635→   635→      {
   636→   636→        const from = readable.find(p => p.page === cursorPage)
   637→   637→        const to = sectionPagesList[sectionPagesList.length - 1]
   638→   638→        if (from && to) segments.push({ from, to, title: null })
   639→   639→      }
   640→   640→
   641→   641→      let pendingFrom: ExtractedPage | null = null
   642→   642→      let pendingChars = 0
   643→   643→      let pendingStartPage = anchor.page
   644→   644→      const flushPending = (upto: ExtractedPage) => {
   645→   645→        if (!pendingFrom) return
   646→   646→        pushUnit({
   647→   647→          title: anchor.title,
   648→   648→          level: 1,
   649→   649→          fromPage: pendingStartPage,
   650→   650→          toPage: upto.page,
   651→   651→          text: textOf(pendingFrom, upto),
   652→   652→          chars: pendingChars,
   653→   653→          kind: 'SECTION',
   654→   654→        })
   655→   655→        pendingFrom = null
   656→   656→        pendingChars = 0
   657→   657→      }
   658→   658→      for (const seg of segments) {
   659→   659→        const segPages = readable.filter(p => p.page >= seg.from.page && p.page <= seg.to.page)
   660→   660→        const segChars = segPages.reduce((s, p) => s + p.text.length, 0)
   661→   661→        if (!pendingFrom) {
   662→   662→          pendingFrom = segPages[0]
   663→   663→          pendingStartPage = pendingFrom.page
   664→   664→          pendingChars = segChars
   665→   665→        } else {
   666→   666→          pendingChars += segChars
   667→   667→        }
   668→   668→        if (pendingChars >= opts.maxUnitChars) flushPending(seg.to)
   669→   669→      }
   670→   670→      flushPending(segments[segments.length - 1]?.to ?? sectionPagesList[sectionPagesList.length - 1])
   671→   671→    }
   672→   672→
   673→   673→    // Sanity: any readable pages beyond the last anchor (appendices etc.)
   674→   674→    const lastCovered = units.reduce((m, u) => Math.max(m, u.toPage), 0)
   675→   675→    const tail = readable.filter(p => p.page > lastCovered)
   676→   676→    if (tail.length > 0) {
   677→   677→      pushUnit({
   678→   678→        title: deduped.length > 0 ? 'Appendix & back matter' : null,
   679→   679→        level: deduped.length > 0 ? 1 : null,
   680→   680→        fromPage: tail[0].page,
   681→   681→        toPage: tail[tail.length - 1].page,
   682→   682→        text: textOf(tail[0], tail[tail.length - 1]),
   683→   683→        chars: tail.reduce((s, p) => s + p.text.length, 0),
   684→   684→        kind: 'SECTION',
   685→   685→      })
   686→   686→    }
   687→   687→
   688→   688→    // Merge tiny units forward (a 1-line heading should not cost a model call)
   689→   689→    return mergeTinyUnits(units, opts.maxUnitChars)
   690→   690→  }
   691→   691→
   692→   692→  // Fallback: UNSTRUCTURED / QUESTION_PAPER — page chunks across the WHOLE doc
   693→   693→  let cur: ExtractedPage[] = []
   694→   694→  let curChars = 0
   695→   695→  const flush = () => {
   696→   696→    if (cur.length === 0) return
   697→   697→    const from = cur[0]
   698→   698→    const to = cur[cur.length - 1]
   699→   699→    pushUnit({
   700→   700→      title: null,
   701→   701→      level: null,
   702→   702→      fromPage: from.page,
   703→   703→      toPage: to.page,
   704→   704→      text: textOf(from, to),
   705→   705→      chars: curChars,
   706→   706→      kind: kind === 'QUESTION_PAPER' ? 'CHUNK' : 'CHUNK',
   707→   707→    })
   708→   708→    cur = []
   709→   709→    curChars = 0
   710→   710→  }
   711→   711→  for (const p of readable) {
   712→   712→    if (p.text.length > opts.maxUnitChars) {
   713→   713→      flush()
   714→   714→      const text = `[page ${p.page}]\n${p.text.slice(0, opts.maxUnitChars)}`
   715→   715→      pushUnit({ title: null, level: null, fromPage: p.page, toPage: p.page, text, chars: text.length, kind: 'OVERFLOW' })
   716→   716→      continue
   717→   717→    }
   718→   718→    if (curChars + p.text.length > opts.maxUnitChars) flush()
   719→   719→    cur.push(p)
   720→   720→    curChars += p.text.length
   721→   721→  }
   722→   722→  flush()
   723→   723→  return units
   724→   724→}
   725→   725→
   726→   726→function mergeTinyUnits(units: AnalysisUnit[], maxUnitChars: number): AnalysisUnit[] {
   727→   727→  const out: AnalysisUnit[] = []
   728→   728→  for (const u of units) {
   729→   729→    const prev = out[out.length - 1]
   730→   730→    const tiny = u.chars < MIN_UNIT_CHARS
   731→   731→    if (
   732→   732→      tiny && prev && prev.kind === 'SECTION' && u.kind === 'SECTION' &&
   733→   733→      prev.title === u.title && // never fuse two different sections — a short chapter is still a chapter
   734→   734→      prev.chars + u.chars <= maxUnitChars
   735→   735→    ) {
   736→   736→      prev.toPage = u.toPage
   737→   737→      prev.text = `${prev.text}\n\n${u.text}`
   738→   738→      prev.chars += u.chars
   739→   739→      continue
   740→   740→    }
   741→   741→    out.push({ ...u })
   742→   742→  }
   743→   743→  return out.map((u, i) => ({ ...u, index: i }))
   744→   744→}
