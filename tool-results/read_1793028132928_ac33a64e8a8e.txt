   372→type WorkspaceTab = 'material' | 'analysis' | 'tests'
   373→
   374→function RiisheWorkspace({ docId, initialPage }: { docId: string; initialPage?: number }) {
   375→  const guestId = useAppStore(s => s.guestId)
   376→  const navigate = useAppStore(s => s.navigate)
   377→
   378→  const [reloadTick, setReloadTick] = useState(0)
   379→  const [loaded, setLoaded] = useState<{ key: string; data: MaterialDetailDTO } | null>(null)
   380→  const [failure, setFailure] = useState<{ key: string; message: string } | null>(null)
   381→
   382→  const loadKey = guestId ? `${guestId}|${docId}|${reloadTick}` : 'no-session'
   383→
   384→  useEffect(() => {
   385→    if (!guestId) return
   386→    let alive = true
   387→    api
   388→      .material(docId, guestId)
   389→      .then(d => {
   390→        if (!alive) return
   391→        setLoaded({ key: loadKey, data: d })
   392→        setFailure(null)
   393→      })
   394→      .catch((e: unknown) => {
   395→        if (!alive) return
   396→        setFailure({ key: loadKey, message: e instanceof Error ? e.message : 'Could not load this material.' })
   397→      })
   398→    return () => {
   399→      alive = false
   400→    }
   401→  }, [guestId, docId, loadKey])
   402→
   403→  const doc = loaded && loaded.key === loadKey ? loaded.data : null
   404→  const error = failure && failure.key === loadKey ? failure.message : null
   405→  const loading = !doc && !error
   406→  const notFound = !!error && /not found/i.test(error)
   407→
   408→  // Poll the material every 5s ONLY while it (or one of its tests) is in flight.
   409→  const shouldPoll =
   410→    !!doc && (isProcessing(doc.status) || doc.tests.some(t => t.status === 'QUEUED' || t.status === 'GENERATING'))
   411→  useEffect(() => {
   412→    if (!shouldPoll || !guestId || !docId) return
   413→    const timer = setInterval(() => {
   414→      api
   415→        .material(docId, guestId)
   416→        .then(d => setLoaded({ key: loadKey, data: d }))
   417→        .catch(() => {
   418→          /* keep the last honest state; the next tick retries */
   419→        })
   420→    }, 5000)
   421→    return () => clearInterval(timer)
   422→  }, [shouldPoll, guestId, docId, loadKey])
   423→
   424→  // Default tab: land on Practice when the document already has a READY set
   425→  // to take — decided ONCE per document (not on every background refresh,
   426→  // which would yank the student out of the tab they are working in).
   427→  const [tab, setTab] = useState<WorkspaceTab>('material')
   428→  const [appliedDocId, setAppliedDocId] = useState<string | null>(null)
   429→  if (doc && doc.id !== appliedDocId) {
   430→    setAppliedDocId(doc.id)
   431→    setTab(doc.status === 'READY' && doc.tests.some(t => t.status === 'READY') ? 'tests' : 'material')
   432→  }
   433→
   434→  const [renameOpen, setRenameOpen] = useState(false)
   435→  const [renameValue, setRenameValue] = useState('')
   436→  const [renaming, setRenaming] = useState(false)
   437→
   438→  // Integrated reader — hosted at workspace level so page references anywhere
   439→  // (analysis, provenance, explanations, material panel) can deep-link into it.
   440→  // A ?page=N route parameter opens the reader directly at that page once.
   441→  const [readerPage, setReaderPage] = useState<number | null>(null)
   442→  const openReaderAt = (page?: number) => setReaderPage(page ?? 1)
   443→  const closeReader = () => setReaderPage(null)
   444→  const [pageParamApplied, setPageParamApplied] = useState(false)
   445→  if (doc && initialPage && !pageParamApplied) {
   446→    setPageParamApplied(true)
   447→    setReaderPage(initialPage)
   448→  }
   449→
   450→  const refresh = () => setReloadTick(t => t + 1)
   451→
   452→  // ── Not found ──
   453→  if (notFound) {
   454→    return (
   455→      <div className="mx-auto w-full max-w-3xl px-4 py-16 sm:px-6 lg:px-8">
   456→        <EmptyState
   457→          icon={FileWarning}
   458→          title="Material not found"
   459→          description="This material does not exist in your library — it may have been removed from this device, or the link is stale."
   460→          actionLabel="Back to My Materials"
   461→          onAction={() => navigate({ view: 'materials' })}
   462→        />
   463→      </div>
   464→    )
   465→  }
   466→
   467→  // ── Load error ──
   468→  if (!loading && error && doc === null) {
   469→    return (
   470→      <div className="mx-auto w-full max-w-3xl px-4 py-16 sm:px-6 lg:px-8">
   471→        <div role="alert" className="rounded-xl border border-destructive/40 bg-destructive/10 p-6">
   472→          <p className="font-display text-lg font-semibold text-destructive">Could not load this material</p>
   473→          <p className="mt-2 text-sm text-destructive/90">{error}</p>
   474→          <div className="mt-5 flex flex-wrap gap-3">
   475→            <Button variant="outline" className="border-gold/40 text-gold hover:bg-gold/10" onClick={refresh}>
   476→              <RotateCcw className="mr-1.5 h-4 w-4" /> Try again
   477→            </Button>
   478→            <Button
   479→              variant="outline"
   480→              className="border-border text-muted-foreground hover:bg-elevated"
   481→              onClick={() => navigate({ view: 'materials' })}
   482→            >
   483→              Back to My Materials
   484→            </Button>
   485→          </div>
   486→        </div>
   487→      </div>
   488→    )
   489→  }
   490→
   491→  // ── Loading ──
   492→  if (loading || !doc) {
   493→    return (
   494→      <div className="mx-auto w-full max-w-5xl px-4 py-10 sm:px-6 lg:px-8" aria-hidden>
   495→        <Skeleton className="h-8 w-40 bg-elevated" />
   496→        <Skeleton className="mt-4 h-10 w-2/3 bg-elevated" />
   497→        <Skeleton className="mt-3 h-6 w-1/2 bg-elevated" />
   498→        <Skeleton className="mt-8 h-10 bg-elevated" />

   522→      .finally(() => setRenaming(false))
   523→  }

   527→      .deleteMaterial(doc.id, guestId)

   542→      .retryMaterial(doc.id, guestId)
   543→      .then(() => {
   544→        toast({ title: 'Reprocessing started', description: 'Riishe is reading the document again.' })

   546→      })

   591→                  {label}
   592→                </Badge>
   593→              ))}
   594→            </div>
   595→          </div>
   596→          <div className="flex shrink-0 flex-wrap items-center gap-2">
   597→            {doc.status === 'READY' && (
   598→              <Button
   599→                variant="outline"
   600→                size="sm"
   601→                className="border-gold/40 text-gold hover:bg-gold/10"

   623→                  <Trash2 className="mr-1.5 h-3.5 w-3.5" aria-hidden /> Remove

   630→                    This permanently deletes “{doc.title}” — the uploaded file, its analysis, and every practice set

   835→/** Honest processing-coverage card — real numbers, never a status badge alone. */
   836→function CoverageCard({
   837→  coverage,
   838→  contentHash,
   839→}: {
   840→  coverage: DocumentCoverageDTO | null
   841→  contentHash?: string
   842→}) {
   843→  if (!coverage) return null
   844→  const rows: { label: string; value: string; tone?: 'ok' | 'warn' | 'bad' }[] = [
   845→    { label: 'PDF pages', value: String(coverage.totalPages) },
   846→    { label: 'Text extracted', value: `${coverage.extractedPages}`, tone: 'ok' },
   847→    ...(coverage.noTextPages > 0
   848→      ? [{ label: 'Image-only (no text)', value: `${coverage.noTextPages}`, tone: 'warn' as const }]
   849→      : []),
   850→    ...(coverage.failedPages > 0
   851→      ? [{ label: 'Extraction failed', value: `${coverage.failedPages}`, tone: 'bad' as const }]
   852→      : []),
   853→    ...(coverage.unprocessedPages > 0
   854→      ? [{ label: 'Beyond size cap (not read)', value: `${coverage.unprocessedPages}`, tone: 'warn' as const }]
   855→      : []),
   856→    { label: 'Readable pages', value: String(coverage.readablePages) },
   857→    { label: 'Pages analysed by Riishe', value: `${coverage.analysedPages} (${coverage.analysisPct}%)`, tone: coverage.analysisState === 'COMPLETE' ? 'ok' : 'warn' },
   858→  ]
   859→  const toneClass = (t?: string) =>
   860→    t === 'ok' ? 'text-success' : t === 'warn' ? 'text-gold' : t === 'bad' ? 'text-destructive' : 'text-foreground'
   861→
   862→  return (
   863→    <section aria-label="Processing coverage" className="rounded-xl border border-border bg-card p-5 sm:p-6">
   864→      <div className="flex flex-wrap items-center justify-between gap-3">
   865→        <h2 className="flex items-center gap-2 font-display text-xl font-semibold text-foreground">
   866→          <ScanSearch className="h-4 w-4 text-gold" aria-hidden /> Processing coverage
   867→        </h2>
   868→        <Badge
   869→          variant="outline"
   870→          className={cn(
   871→            coverage.analysisState === 'COMPLETE'
   872→              ? 'border-success/40 bg-success/10 text-success'
   873→              : 'border-gold/40 bg-gold/10 text-gold',
   874→          )}
   875→        >
   876→          {coverage.analysisState === 'COMPLETE' ? 'Full analysis' : `Partial analysis — ${coverage.analysisPct}% of readable pages`}
   877→        </Badge>
   878→      </div>
   879→
   880→      <dl className="mt-4 grid grid-cols-2 gap-x-4 gap-y-2 sm:grid-cols-3">
   881→        {rows.map(r => (
   882→          <div key={r.label} className="rounded-lg border border-border bg-elevated px-3 py-2">
   883→            <dt className="text-[10px] font-semibold uppercase tracking-[0.12em] text-muted-foreground">{r.label}</dt>
   884→            <dd className={cn('mt-0.5 text-sm font-semibold tabular-nums', toneClass(r.tone))}>{r.value}</dd>
   885→          </div>
   886→        ))}
   887→      </dl>
   888→
   889→      {coverage.analysisState === 'PARTIAL' && (
   890→        <p className="mt-3 text-xs leading-relaxed text-muted-foreground">
   891→          {coverage.analysedPages} of {coverage.readablePages} readable page{coverage.readablePages === 1 ? ' was' : 's were'} analysed.
   892→          Topics and page references below only cover the analysed part — the rest of the document is still readable in the viewer,
   893→          and questions are generated only from the analysed content.
   894→        </p>
   895→      )}
   896→
   897→      {coverage.printedPageOffset !== null && coverage.printedPageOffset !== 0 && (
   898→        <p className="mt-2 text-xs leading-relaxed text-muted-foreground">
   899→          Note: this document prints its own page numbers ({coverage.printedPageOffset > 0 ? `${coverage.printedPageOffset} behind` : `${Math.abs(coverage.printedPageOffset)} ahead of`} the PDF page numbers).
   900→          All page references here use the PDF page numbers shown in the reader.
   901→        </p>
   902→      )}
   903→
   904→      {coverage.pageErrors.length > 0 && (
   905→        <>
   906→          <Separator className="my-4" />
   907→          <h3 className="flex items-center gap-2 text-sm font-semibold text-destructive">
   908→            <FileWarning className="h-3.5 w-3.5" aria-hidden /> Page-level extraction errors
   909→          </h3>
   910→          <ul className="mt-2 max-h-40 space-y-1 overflow-y-auto pr-1">
   911→            {coverage.pageErrors.map(e => (
   912→              <li key={e.page} className="flex items-start gap-2 text-xs text-muted-foreground">
   913→                <XCircle className="mt-0.5 h-3 w-3 shrink-0 text-destructive" aria-hidden />
   914→                <span>
   915→                  Page {e.page}: {e.error}
   916→                </span>
   917→              </li>
   918→            ))}
   919→          </ul>
   920→        </>
   921→      )}
   922→
   923→      {contentHash && (
   924→        <p className="mt-4 font-mono text-[10px] text-muted-foreground/70" title="SHA-256 content hash (first 12 chars)">
   925→          content {contentHash} · this analysis belongs to exactly this version of the file
   926→        </p>
   927→      )}
   928→    </section>
   929→  )
   930→}

  1151→function TestsPanel({
  1152→  doc,
  1153→  guestId,
  1154→  onDocRefresh,
  1155→  onOpenPage,
  1156→}: {
  1157→  doc: MaterialDetailDTO
  1158→  guestId: string
  1159→  onDocRefresh: () => void
  1160→  onOpenPage: (p: number) => void
  1161→}) {
  1162→  const navigate = useAppStore(s => s.navigate)
  1163→
  1164→  // Explicitly selected set; falls back to in-flight → newest READY.
  1165→  const [selectedId, setSelectedId] = useState<string | null>(null)
  1166→  const sortedTests = [...doc.tests].sort((a, b) => b.createdAt.localeCompare(a.createdAt))
  1167→  const inflight = doc.tests.find(t => t.status === 'QUEUED' || t.status === 'GENERATING')
  1168→  const latestReady = sortedTests.find(t => t.status === 'READY')
  1169→  const activeId = selectedId ?? inflight?.id ?? latestReady?.id ?? null
  1170→  const { test: activeTest, error: testError } = useRiisheTestPoll(activeId, guestId)
  1171→  const [startingId, setStartingId] = useState<string | null>(null)
  1172→
  1173→  const startTest = (testId: string) => {
  1174→    if (startingId) return
  1175→    setStartingId(testId)
  1176→    api
  1177→      .startRiisheTest(testId, guestId)
  1178→      .then(r => {
  1179→        if (r.resumed) {
  1180→          toast({ title: 'Resuming your saved attempt', description: 'Your earlier answers were restored.' })
  1181→        }
  1182→        navigate({ view: 'test', attemptId: r.attemptId })
  1183→      })
  1184→      .catch((e: unknown) => {
  1185→        toast({ title: 'Could not start the test', description: e instanceof Error ? e.message : 'Try again.', variant: 'destructive' })
  1186→      })
  1187→      .finally(() => setStartingId(null))
  1188→  }
  1189→
  1190→  const canGenerate = doc.status === 'READY'
  1191→  const regenerate = () => {
  1192→    if (!activeId) return
  1193→    setSelectedId(activeId)
  1194→  }
  1195→
  1196→  return (
  1197→    <div className="space-y-6">
  1198→      {/* Generator — locked to this document */}
  1199→      <RiisheGeneratorPanel
  1200→        guestId={guestId}
  1201→        docs={[{ id: doc.id, title: doc.title, topics: doc.analysis?.topics.map(t => ({ name: t.name, pages: t.pages })) ?? [] }]}
  1202→        locked
  1203→        onGenerated={testId => {
  1204→          setSelectedId(testId)
  1205→          onDocRefresh()
  1206→        }}
  1207→      />
  1208→
  1209→      {!canGenerate && (
  1210→        <p className="flex items-start gap-2 rounded-lg border border-border bg-elevated px-4 py-3 text-xs leading-relaxed text-muted-foreground">
  1211→          <Info className="mt-0.5 h-3.5 w-3.5 shrink-0 text-gold" aria-hidden />
  1212→          {doc.status === 'NEEDS_OCR'
  1213→            ? 'Questions cannot be generated — this document could not be analysed (scanned pages, OCR unavailable).'
  1214→            : doc.status === 'FAILED'
  1215→              ? 'Processing failed — retry it from the Material tab before generating a set.'
  1216→              : 'Generation unlocks once the document analysis completes.'}
  1217→        </p>
  1218→      )}
  1219→
  1220→      {/* Active set: live progress / failure / pre-test summary */}
  1221→      {activeId && (
  1222→        <section aria-label="Selected practice set">
  1223→          {!activeTest && !testError && <TestSkeletonCard />}
  1224→
  1225→          {testError && (
  1226→            <div className="rounded-xl border border-destructive/40 bg-destructive/10 p-5 sm:p-6" role="alert">
  1227→              <p className="text-sm text-destructive">{testError}</p>
  1228→            </div>
  1229→          )}
  1230→
  1231→          {activeTest && (activeTest.status === 'QUEUED' || activeTest.status === 'GENERATING') && (
  1232→            <TestPendingCard status={activeTest.status} statusDetail={activeTest.statusDetail} />
  1233→          )}
  1234→
  1235→          {activeTest && activeTest.status === 'FAILED' && (
  1236→            <TestFailedCard statusDetail={activeTest.statusDetail} onRetry={regenerate} disabled={!canGenerate} />
  1237→          )}
  1238→
  1239→          {activeTest && activeTest.status === 'READY' && (
  1240→            <PreTestSummaryCard
  1241→              test={activeTest}
  1242→              starting={startingId === activeTest.id}
  1243→              onStart={() => startTest(activeTest.id)}
  1244→              onOpenPage={(page) => onOpenPage(page)}
  1245→            />
  1246→          )}
  1247→        </section>
  1248→      )}
  1249→
  1250→      {/* History */}
  1251→      <section aria-label="Practice history" className="rounded-xl border border-border bg-card p-5 sm:p-6">
  1252→        <h2 className="flex items-center gap-2 font-display text-xl font-semibold text-foreground">
  1253→          <ClipboardList className="h-4 w-4 text-gold" aria-hidden /> Practice history
  1254→        </h2>
  1255→
  1256→        {sortedTests.length === 0 ? (
  1257→          <p className="mt-5 rounded-lg border border-border bg-elevated px-4 py-3 text-sm text-muted-foreground">
  1258→            No practice sets generated for this document yet.
  1259→          </p>
  1260→        ) : (
  1261→          <ul className="mt-5 space-y-3">
  1262→            {sortedTests.map(t => (
  1263→              <TestHistoryRow
  1264→                key={t.id}
  1265→                test={t}
  1266→                starting={startingId === t.id}
  1267→                onSelect={() => setSelectedId(t.id)}
  1268→                onStart={() => startTest(t.id)}
  1269→              />
  1270→            ))}
  1271→          </ul>
  1272→        )}
  1273→        <p className="mt-4 text-xs text-muted-foreground">
  1274→          Completed attempts and their scores appear in My Progress. You can also combine topics from several
  1275→          documents in <button type="button" className="text-gold hover:underline" onClick={() => navigate({ view: 'practice' })}>Practice</button>.
  1276→        </p>
  1277→      </section>
  1278→    </div>
  1279→  )
  1280→}
